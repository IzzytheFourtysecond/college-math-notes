\exTwo\ul{Lemma 3.2:} The map described above is surjective.
\begin{myIndent}\exThreeP
	Proof:\\
	Let $w \in T_a \mathbb{R}^n$ be any derivation. Then  having let $e_1, \ldots, e_n$ be the standard basis, for all $1 \leq i \leq n$ let $c_i \coloneqq w(x \mapsto x \cdot e_i)$. I claim $w = \Df_{v}|_a$ where $v = c_1 e_1 + \ldots + c_n e_n$.\newpage

	To see why, let $f$ be any $C^\infty$ function on $\mathbb{R}^n$. Next for any $x \in \mathbb{R}^n$ let\\ $h(t) \coloneqq f(a + t(x - a))$. If we say $a = (a_1, \ldots, a_n)$ and $x = (x_1, \ldots, x_n)$,\\ then note that:
	
	{\centering\begin{tabular}{l}
		$f(x) - f(a) = h(1) - h(0) = \int_0^1 \frac{\df h}{\df t}(s)\df s$\\ [6pt]
		$\phantom{f(x) - f(a) = h(1) - h(0)} = \int_0^1 (x - a) \cdot \nabla f(a + s(x - a))\df s$\\ [6pt]
		$\phantom{f(x) - f(a) = h(1) - h(0)} = \sum_{i=1}^n  (x_i - a_i) \int_0^1 \frac{\partial}{\partial x_i}f(a + s(x - a))\df s$.
	\end{tabular}\retTwo\par}

	In particular, we set $h_{i}(x) \coloneqq \int_0^1 \frac{\partial}{\partial x_i}f(a + s(x - a))\df s$. By Leibniz's rule applied $h_i$ (which we can apply to $x$ in bounded subsets of $\mathbb{R}^n$ since $f$ is a $C^\infty$ function), we know $h_i$ is also a $C^\infty$ function on $\mathbb{R}^n$.
	\begin{myIndent}\myComment
		As a side note, if $f$ were only finitely many times differentiable then this paragraph would not work. This hints at how we need an alternate definition of tangent vectors for when we are working on $C^r$ manifolds where $r < \infty$.\retTwo
	\end{myIndent}

	And now we have that $w(f) = w\left(f(a) + \sum_{i=1}^n (x_i - a_i)g_i(x)\right)$. Yet $w(f(a)) = 0$ since $f(a)$ is a constant. Meanwhile:
	
	{\centering\begin{tabular}{l}
		$w((x_i - a_i)g_i(x)) = (a_i - a_i)w(g_i) + g_i(a)w(x_i - a_i)$\\ [6pt]
		$\phantom{w((x_i - a_i)g_i(x))} = 0 + (w(x \mapsto x \cdot e_i) - w(a_i))\int_0^1 \frac{\partial}{\partial x_i}f(a)\df s = (c_i - 0)\frac{\partial}{\partial x_i}f(a)$.
	\end{tabular}\retTwo\par}

	So, $w(f) = \sum_{i=1}^n c_i \frac{\partial}{\partial x_i}f(a) = D_v|_a f$. $\blacksquare$\retTwo

	\begin{myIndent}\myComment
		As a side note, if we consider "derivations" whose domain is the vector space\\ $C^k(M, \mathbb{R})$ where $M$ is a $C^r$ manifold and $k < \min(r + 1, \infty)$, then we can ask\\ about "exotic derivations" which aren't described by directional derivatives.\retTwo
	\end{myIndent}
\end{myIndent}

\exTwo\ul{Lemma 3.11:} Suppose $\iota: H^n \hookrightarrow \mathbb{R}^n$ is the inclusion map. (Hopefully it is obvious that this is a $C^\infty$-map.) For any $a \in \partial H^n$ the differential $\df \iota_a: T_a H^n \to T_a \mathbb{R}^n$ is an isomorphism.
\begin{myIndent}\exThreeP
	Proof:\\
	First I shall show injectivity. Suppose $v \in T_a H^n$ and $\df \iota_a(v) = 0 \in T_a \mathbb{R}^n$. Then consider any smooth function $f: H^n \to \mathbb{R}$. By definition we know that $f$ extends to a smooth function $g: U \to \mathbb{R}$ where $U \subseteq \mathbb{R}^n$ is open and contains $H^n$. Furthermore, $H^n$ is a closed subset of $U$. Thus by identical reasoning as in the proof of proposition 3.9, we can find a smooth function $\widetilde{f}: \mathbb{R}^n \to \mathbb{R}$ such that $\widetilde{f}|_{H^n} = g|_{H^n} = f$.\retTwo

	But now $v(f) = v(\widetilde{f}|_{H^n}) = v(\widetilde{f}\circ \iota) = \df \iota_a(v)\widetilde{f} = 0$. And since $f$ was arbitrary, we've proven that $v = 0$. So, $\df \iota_a$ has a trivial kernel and is thus injective.\retTwo

	Next we show surjectivity. Suppose $w \in T_a \mathbb{R}^n$ and then define $v \in T_a H^n$ by\\ $v(f) \coloneqq w(\widetilde{f})$ where $\widetilde{f}$ is any smooth extension of $f$ to all of $\mathbb{R}^n$.\newpage
	\begin{myIndent}\exPPP
		To see that $v$ is well-defined, note by the prior corollary that $w = \sum_{i=1}^n c_i \frac{\partial}{\partial x_i}|_{a}$\\ for some $c_1, \ldots, c_n \in \mathbb{R}$. Yet since $\widetilde{f}$ has a continuous derivative, $\frac{\partial}{\partial x_i}\widetilde{f}(a)$ is\\ uniquely determined by $\widetilde{f}$'s restriction to $H^n$.\retTwo
	\end{myIndent}

	Now its easy to check that $v$ is a derivation and that $\df \iota_a(v) = w$. $\blacksquare$\retTwo
\end{myIndent}

\begin{myIndent}\exThreeP
	Proof:\\

I want to switch to covering some category theory related notes about manifolds.\retTwo Firstly, we let $\mcateg{Man}$ be the category whose objects are topological manifolds without\\ boundaries and whose morphisms are continuous maps. If we want to allow our manifolds to have a boundary, we instead consider the category $\mcateg{Man}_b$.\retTwo

\exTwo\ul{Proposition:} Suppose $M_1, \ldots, M_{k-1}$ are topological manifolds without boundary and $M_k$ is a topological manifold with boundary. Also suppose each $M_i$ has dimension $n_i$. Then the topological product space $M \coloneqq M_1 \times \cdots \times M_k$ is an $n_1 + \ldots + n_k$ dimensional topological manifold with boundary. 
\begin{myIndent}\exThreeP
	Proof:\\
	Countable products of second countable topological spaces are still second countable. Hence $M$ is second countable.
	\begin{myIndent}\exPPP
		This is easy to verify.\retTwo
	\end{myIndent}
	
	Also arbitrary products of Hausdorff spaces are still Hausdorff. Hence, $M$ is Haudorff.\retTwo

	Finally, to show $M$ is locally Euclidean, let $(p_1, \ldots, p_k) \in M$. Then for each $1 \leq i \leq k$ we can pick a chart $(U_i, \varphi_i)$ on $M_i$ containing $p_i$. Now we identify $\mathbb{R}^{n_1} \times \cdots \times \mathbb{R}^{n_k}$ with $\mathbb{R}^{n_1 + \ldots + n_k}$ in the obvious way. Then we consider the chart $(U_1 \times \ldots \times U_k, \varphi_1 \times \cdots \times \varphi_k)$. This will be a homeomorphism to an open subset of $\mathbb{R}^{n_1 + \ldots + n_k}$ or $H^{n_1 + \ldots + n_k}$. $\blacksquare$\retTwo

	\begin{myIndent}\myComment
		Remark: By the proposition on \inLinkRap{Proposition page 1010}{pages 1010-1011}, we know that $(p_1, \ldots, p_k)$ is a boundary point of $M$ if and only if $p_k$ is a boundary point of $M_k$. Hence if $M_k$ has an empty boundary then our constructed chart from before will always be an interior chart instead of a boundary chart.\retTwo
	\end{myIndent}
\end{myIndent}

\hTwo Since $M = M_1 \times \cdots \times M_k$ is also a product in the category of topologies, we know $M$\\ interacts with morphisms (i.e. continuous maps) in the way that a category theoretic\\ product should. And by the above remark, we've thus proven that $\mcateg{Man}$ admits finite\\ products.\newpage

As for whether $\mcateg{Man}_b$ admits finite products, we can actually assume in the prior\\ proposition that $M_1, \ldots, M_{k-1}$ have boundaries. To see why, let:

{\centering$\mathbb{R}^k_+ \coloneqq \{(x_1, \ldots, x_k) : x_i \geq 0 \text{ for all }i\}$.\retTwo\par}

\exTwo\ul{Proposition:} $\mathbb{R}^k_+$ is homeomorphic to $H^k$.
\begin{myIndent}\exThreeP
	Proof:\\ 
	Firsty, note that $\mathbb{R}^1_+ = H^1$. Also $\mathbb{R}^2_+$ is homeomorphic to $H^2$ via the map:
	
	{\centering$\varphi(x, y) \coloneqq (x^2 - y^2, 2xy)$.\retTwo\par}

	\begin{myIndent}\exPPP
		Why?\\
		It's clear that $\varphi$ is continuous and also that it maps $\mathbb{R}^2_+$ into $H^2$. After all, $x \geq 0$ and $y \geq 0$ implies that $2xy \geq 0$. As for showing the other requirements for $\varphi$ to be a homeomorphism, I want to explicitely solve for the inverse function $\varphi^{-1}$.\retTwo

		Suppose $\varphi(x, y) = (z, w)$.
		\begin{myIndent}\color{orange}
			Suppose $y \neq 0$. Then $w = 2xy \Longrightarrow x^2 = \frac{w^2}{4y^2}$. In turn, we get that\\ $z = x^2 - y^2 = \frac{w^2}{4y^2} - y^2$. Or in other words, $4y^4 + 4zy^2 - w^2 = 0$. By\\ using the quadratic formula, we can then solve that:

			{\centering $y^2 = \frac{-4z \pm \sqrt{16z^2 + 16w^2}}{8} = \frac{-z \pm \sqrt{z^2 + w^2}}{2}$ \retTwo\par}

			We know $y^2 \geq 0$ though. So, we can actually further solve that:
			
			{\centering$y^2 = \frac{-z \cyPen{+} \sqrt{z^2 + w^2}}{2}$\retTwo\par}

			And then since $y$ is nonnegative, we can finally say that $y = \sqrt{\frac{-z + \sqrt{z^2 + w^2}}{2}}$.\\ And this is a well-defined real number because $\sqrt{z^2 + w^2} \geq \max(|-z|, 0)$.\\ Also, since $y \neq 0$ by assumption we can say that:
			
			{\centering$x = \frac{w}{2y} = \frac{w}{2\sqrt{\frac{-z + \sqrt{z^2 + w^2}}{2}}}$.\retTwo\par}

			So, we define: $\psi: H^2 - \{(z, 0) : z \geq 0\} \to \mathbb{R}^2_+ - \{(x, 0) : x \geq 0\}$ by:
			
			{\centering$\psi(z, w) = \left(\frac{w}{2\sqrt{\frac{-z + \sqrt{z^2 + w^2}}{2}}}, \phantom{.}\sqrt{\frac{-z + \sqrt{z^2 + w^2}}{2}}\right)$\retTwo\par}

			Clearly $\psi$ is a continuous function. And by how we solved for $\psi$ we know that $\psi(\varphi(x, y)) = (x, y)$. Similarly, we can solve that $\varphi(\psi(z, w)) = (z, w)$.\retTwo

			Finally, note that if $y = 0$ then $w = 0$ and $z = x^2$. Therefore, we want to define $\psi(z, 0) \coloneqq (\sqrt{z}, 0)$ when $z \geq 0$. And once we define this, we know that $\psi = \varphi^{-1}$. That said, we need to show our extension of $\psi$ to the ray $\{(z, 0) : z \geq 0\}$ is still continuous.\retTwo

			To do this, consider any $z \geq 0$ and pick $R > \sqrt{z}$. Then $N \coloneqq \overline{B_R(0)} \cap \mathbb{R}^2_+$ is a compact neighborhood of $\varphi^{-1}(z, 0)$ relative to $\mathbb{R}^2_+$. Also, we can see that:
			
			{\centering$\varphi(N) = \varphi(\overline{B_R(0)} \cap \mathbb{R}^2_+) = \overline{B_{R^2}(0)} \cap H^2 \eqqcolon N^\prime$.\retTwo\par}

			This is because $|\varphi(x, y)| = |(x, y)|^2$ and we already know from earlier that $\varphi$\\ surjectively maps $\mathbb{R}_+^2$ into $H^2$. So, $\varphi$ is a bijective continuous map from the compact neighborhood $N$ of $\varphi^{-1}(z, 0)$ onto a compact neighborhood $N^\prime$ of $(z, 0)$. It follows from the compactness of $N$ and the Hausdorffness of $N^\prime$ that $\varphi^{-1}$ is a continuous map from $N^\prime$ to $N$. In other words, $\varphi^{-1}$ is continuous at $(z, 0)$.\retTwo
		\end{myIndent}
	\end{myIndent}

	But now its easy to conclude by induction that $\mathbb{R}^k_+$ is homeomorphic to $H^k$. After all,\\ $\mathbb{R}^k_+ = \mathbb{R}^{k-2}_+ \times \mathbb{R}^2_+$ is homeomorphic to $\mathbb{R}^{k-2}_+ \times H^2$ by the prior reasoning, and then\\ that is trivially homeomorphic to $\mathbb{R} \times \mathbb{R}^{k-1}_+$. And now we just conclude by induction that\\ $\mathbb{R}^{k-1}_+ \cong H^{k-1}$. So, $\mathbb{R} \times \mathbb{R}^{k-1}_+ \cong \mathbb{R} \times H^{k-1} \cong H^k$. $\blacksquare$\retTwo
\end{myIndent}

\ul{Corollary:} Suppose $M_1, \ldots, M_{k}$ are topological manifolds with boundary. Also suppose each $M_i$ has dimension $n_i$. Then the topological product space $M \coloneqq M_1 \times \cdots \times M_k$ is an $n_1 + \ldots + n_k$ dimensional topological manifold with boundary. 
\begin{myIndent}\exThreeP
	Proof:\\
	Second countability and Hausdorffness is obvious. Hence, we only need to worry about if $M$ is locally Euclidean.\retTwo

	Let $(p_1, \ldots, p_k)$ be any point in $M$ and let $(U_i, \varphi_i)$ be a chart in $M_i$ containing $p_i$. By shrinking each $U_i$, we can assume without loss of generality that $\widehat{U}_i \coloneqq \varphi(U_i)$ is a rectangle in $H^{n_i}$ whose ends are all open intervals in $\mathbb{R}$ or $[0, \infty]$. Then, we consider the function $\varphi_1 \times \cdots \times \varphi_n : U_1 \times \cdots \times U_k \longrightarrow \widehat{U}_1 \times \cdots \times \widehat{U}_k$. By composing this function with some permutation of our coordinates, we get that there are open sets $V \subseteq \mathbb{R}^{n_1 + \ldots + n_k - k}$ and $W \subseteq \mathbb{R}^k_+$, as well a homeomorphism $\psi: U \to V \times W$. Finally, transform $W$ into an open set of $H^k$ via a homeomorphism we know exists from the prior proposition. $\blacksquare$\retTwo
\end{myIndent}

	Proof:\\

	{\centering$(\psi_1 \circ \varphi_1^{-1}) \times \cdots \times (\psi_k \circ \varphi^{-1}_k)$.\retTwo\par}

\begin{myIndent}\exThreeP
	Proof:\\

	\begin{itemize}
		\item $\psi(V)$ is open in $H^n$ but not $\mathbb{R}^n$ and $\psi(p) \in \partial H^n$,
		\item $\psi(V)$ is open in $\overline{\mathbb{R}_+}^n$ but not $H^n$ or $\mathbb{R}^n$ and $\psi(p) \in \partial \overline{\mathbb{R}_+}^n$.\retTwo

	\begin{myIndent}\exPPP

One may notice that the definition of a derivation on a $C^\infty$ manifold $M$ which was given prior specifically required us to work with maps in $C^\infty(M, \mathbb{R})$ maps. Also, we only defined differentials of $C^\infty$ maps between maps. But what if we want to work with less smooth manifolds.\retTwo

To start off, I want to recall a comment I made on \inLinkRap{Side note page 1035}{page 1035}. Namely, suppose we\\ are working with $C^r$ manifold $M$ (where $r \in \mathbb{Z}_{\geq 1} \cup \{\infty\}$). Then given any $p \in M$ and\\ $k < \min(\infty, r + 1)$, what is the space of "derivations at $p$" mapping $C^k(M, \mathbb{R})$ to $\mathbb{R}$?
\begin{myIndent}\pracTwo
	I'm only slightly modifying a proof that I read from math stack exchange (see \inLinkRap{citation 11}{citation 11}\\ in my bibliography).\retTwo

	Recall that $C^k(M, \mathbb{R})$ is a unital real algebra. So, let us consider the set:
	
	{\centering$\mathfrak{m} \coloneqq \{f \in C^k(M, \mathbb{R}) : f(p) = 0\}$.\retTwo\par}
	
	I claim this is a maximal ideal of $C^k(M, \mathbb{R})$.
	\begin{myIndent}\exPPP
		To see that this is an ideal, just note that if $f \in \mathfrak{m}$ and $g \in C^k(M, \mathbb{R})$ then\\ $fg \in C^k(M, \mathbb{R})$ with $(fg)(p) = 0 \cdot g(p) = 0$. So, $fg \in \mathfrak{m}$. Similarly, if\\ $f, g \in \mathfrak{m}$ then $f + g \in C^k(M, \mathbb{R})$ with $(f + g)(p) = f(p) + g(p) = 0 + 0 = 0$.\\ So, $f + g \in \mathfrak{m}$.\retTwo

		As for showing that $\mathfrak{m}$ is a maximal ideal, suppose $g \in C^k(M, \mathbb{R}) - \mathfrak{m}$. Then\newpage consider any $f \in C^k(M, \mathbb{R})$. Since $g(p) \neq 0$, we can say that $f - \frac{f(p)}{g(p)}g \in \mathfrak{m}$. In turn, $f \in \mathbb{R}g + \mathfrak{m}$. And since $f$ was arbitrary, we've proven that:
		
		{\center$C^k(M, \mathbb{R}) = \mathbb{R}g + \mathfrak{m} \subseteq \langle \{g\} \cup \mathfrak{m}\rangle \subseteq C^k(M, \mathbb{R})$.\retTwo\par}
		
		Also since $g$ was arbitrary, we've shown that $\mathfrak{m}$ is a maximal ideal (and even a maximal proper vector subspace) of $C^k(M, \mathbb{R})$.\retTwo
	\end{myIndent}

	Now $\mathfrak{m}^2$ will be an ideal (and thus also vector subspace) of $C^k(M, \mathbb{R})$ contained in $\mathfrak{m}$.\\ Hence, we can consider the quotient space $V \coloneqq \mathfrak{m}/\mathfrak{m}^2$. I claim that the space of\\ "derivations at $p$" mapping $C^k(M, \mathbb{R})$ to $\mathbb{R}$ is isomorphic to the algebraic dual space $V^*$.
	\begin{myIndent}\exPPP
		Proof:\\
		Firstly suppose $\varphi \in V^* = (\mathfrak{m}/\mathfrak{m}^2)^*$. Equivalently, we can view $\varphi$ as linear\\ functional on $\mathfrak{m}$ such that $\varphi(\mathfrak{m}^2) = \{0\}$. Next, we know from before that\\ $C^k(M, \mathbb{R}) = \{\text{constant functions}\} \oplus \mathfrak{m}$. So, we can uniquely extend $\varphi$ to being\\ a linear functional on $C^k(M, \mathbb{R})$ by just setting $\varphi(\text{constant function}) \coloneqq 0$. For\\ convenience, I'll give this extended linear functional the name $\Df_\varphi$.\retTwo

		I claim $\Df_{\varphi}$ is a derivation at $p$. After all, we already know its a linear map from $C^k(M, \mathbb{R})$ to $\mathbb{R}$. Furthermore, suppose $f, g \in C^k(M, \mathbb{R})$. Then we know that $\widetilde{f} \coloneqq f - f(p)$ and $\widetilde{g} \coloneqq g - g(p)$ are both in $\mathfrak{m}$. Also:

		{\centering\begin{tabular}{l}
			$\Df_\varphi(fg) = \Df_\varphi((\widetilde{f} + f(p))(\widetilde{g} + g(p)))$\\ [4pt]
			$\phantom{\Df_\varphi(fg)} = \Df_\varphi(\widetilde{f}\widetilde{g}) + f(p)\Df_\varphi(g - g(p)) + g(p)\Df_\varphi(f - f(p)) + \Df_\varphi(f(p)g(p))$.
		\end{tabular}\retTwo\par}

		But now as $\Df_{\varphi}(\mathfrak{m}^2) = \{0\}$, we know that $\Df_\varphi(\widetilde{f}\widetilde{g}) = 0$. Also, $\Df_\varphi(f(p)g(p)) = 0$, $\Df_\varphi(g - g(p)) = \Df_\varphi(g)$, and $\Df_\varphi(f - f(p)) = \Df_\varphi(f)$ since $\Df_\varphi$ is a linear map sending constant functions to zero. Hence, we can conclude that:

		{\centering $\Df_\varphi(fg) = f(p)\Df_\varphi(g) + g(p)\Df_\varphi(f)$.\retTwo\par}

		Now after letting $\mathcalli{D}_p M$ denote the space of derivations at $p$ mapping $C^k(M, \mathbb{R})$ to $\mathbb{R}$ define the map $\Phi: V^* \to \mathcalli{D}_p M$ by $\varphi \mapsto \Df_\varphi$. It is obvious from the construction of $\Df_\varphi$ that $\Phi$ is a linear map. Meanwhile, to show that $\Phi$ is bijective we shall note that $\Phi$ has an incredibly simple inverse.\retTwo

		Suppose $w \in \mathcalli{D}_p M$ and consider $w|_{\mathfrak{m}}$. Since $w$ is a derivation, we know from the lemma on \inLinkRap{lemma page 1028}{page 1028} that $w(\mathfrak{m}^2) = \{0\}$. {\color{red}[Technically that lemma assumes all the functions being input into the derivation are $C^\infty$. That said, nothing in the proof of that lemma actually requires any smoothness conditions.]} It follows that $w|_{\mathfrak{m}}$\\ quotients to a well defined map $\Psi(w) \in (\mathfrak{m}^2/\mathfrak{m})^*$. Also, we can easily see that\\ $\Psi = \Phi^{-1}$. $\blacksquare$\retTwo
	\end{myIndent}

	You may note that none of our reasoning in this section so far has required that $k$ is finite. However, a fact one can prove is that if $k$ is finite then $V$ will be infinite dimensioned. In turn, we'll also have that $V^*$ is infinite dimensioned. I'm tired, its almost 1am, and proving this fact isn't the most insanely interesting thing ever. So, I'm going to skip showing this. But the T.L.D.R. is that if $k < \infty$ then there are way more derivations from $C^k(M, \mathbb{R})$ into $\mathbb{R}$ then there should be if derivations are purely an abstraction of directional derivatives. And we don't want this.\newpage
\end{myIndent}

{\hTwo\dispDate{7/19/2026}}

If our goal is to have the tangent space to a $C^r$-manifold $M$ (where $r$ is potentially finite) only contain directional derivatives, then we can try to formalize a tangent space in the following different way from how we were doing so before. (Also for this section I'll be assuming by default that our manifolds can have corners and a boundary.)\retTwo

Pick a point $p \in M$. Next, we consider any $C^1$ path $\gamma: J \to M$ where $J \subseteq \mathbb{R}$ is a closed interval containing $0$ and having a nonempty interior, and $\gamma(0) = p$.
\begin{myIndent}\myComment
	As a side note, I'm allowing $0$ to be either endpoint of $J$.\retTwo
	
	Furthermore, as $J$ is a $1$-dimensional manifold (with a boundary) it does make sense to talk about $\gamma$ as being $C^1$ or not. In fact, given any smooth chart $(U, \varphi)$ on $M$, we can say that $\gamma$ is a $C^1$ map if and only if $\gamma$ is continuous and $\varphi \circ \gamma$ is a $C^1$ map from $\gamma^{-1}(U) \subseteq \mathbb{R}$.\retTwo
\end{myIndent}

We shall consider the set $\mathcalli{C}(M, p)$ of all such $C^1$-paths $\gamma$ described priorly. And then, we define an equivalence relation on $\mathcalli{C}(M, p)$. Namely, we say $\gamma_1 \sim \gamma_2$ iff:

{\centering$(f \circ \gamma_1)^\prime(0) = (f \circ \gamma_2)^\prime(0)$ for all $f \in C^1(M, \mathbb{R})$.\retTwo\par}

The fact that this is an equivalence relation is trivial to show. Hence, we can consider the space $\mathcalli{V}_p M$ of equivalence classes of $\mathcalli{C}(M, p)$. Plus, if $\gamma \in \mathcalli{C}(M, p)$ then we consider its equivalence class in $\mathcalli{V}_p M$ to be $[\gamma]$.\retTwo

\begin{myIndent}
	\ul{Lemma:} Suppose $F: M \to N$ is a $C^1$ map and $\gamma_1: J_1 \to M$ and $\gamma_2: J_2 \to M$ are paths in $\mathcalli{C}(M, p)$ such that $[\gamma_1] = [\gamma_2]$.
	\begin{enumerate}
		\item[(a)] We know $(F \circ \gamma_i): J_i \to N$ is a path in $\mathcalli{C}(N, F(p))$.
		\begin{myIndent}\pracTwo
			Hopefully this is obvious. After all, $F \circ \gamma_i$ is a composition of $C^1$ maps and is thus a $C^1$ map itself. Also $(F \circ \gamma_i)(0) = F(p)$.\retTwo
		\end{myIndent}

		\item[(b)] Furthermore, $[F \circ \gamma_1] = [F \circ \gamma_2]$ in $\mathcalli{V}_{F(p)}N$.
		\begin{myIndent}\pracTwo
			Why?\\
			Suppose $g \in C^1(N, \mathbb{R})$. Then $g \circ F \in C^1(M, \mathbb{R})$. Hence, we know that:
			
			{\centering$((g \circ F) \circ \gamma_1)^\prime(0) = ((g \circ F) \circ \gamma_2)^\prime(0)$. $\blacksquare$\retTwo\par}
		\end{myIndent}
	\end{enumerate}
\end{myIndent}

Based on this, we'll define the \udefine{differential at $p$} of a $C^1$ map $F: M \to N$ to be the\\ function $\df F_p : \mathcalli{V}_p M \to \mathcalli{V}_{F(p)} N$ given by $\df F_p [\gamma] \coloneqq [F \circ \gamma]$.
\begin{myTindent}\myComment
	(Ignore the conflict in notation now. I'll deal with it later\dots)\retTwo
\end{myTindent}

\begin{myIndent}
	\ul{Proposition:} Let $M, N, P$ be $C^r$ manifolds, let $F: M \to N$ and $G: N \to P$ be\\ $C^1$ maps, and let $p \in M$.
	\begin{enumerate}
		\item[(a)] $\df(G \circ F)_p = \df G_{F(p)} \circ \df F_p$.
		\begin{myIndent}\pracTwo
			Why?\\
			$\df(G \circ F)_p([\gamma]) = [G \circ F \circ \gamma] = \df G_{F(p)}([F \circ \gamma]) =  \df G_{F(p)}( \df F_{p}([\gamma]))$.\newpage
		\end{myIndent}
		\item[(b)] $\df (\myId_M)_p = \myId_{\mathcalli{V}_p M}$.
		\begin{myIndent}\pracTwo
			Hopefully this is obvious from the fact $[\myId_M \circ \gamma] = [\gamma]$.\retTwo
		\end{myIndent}
		\item[(c)] If $F: M \to N$ is a $C^1$-diffeomorphism then $\df F_p$ is a bijection and\\ $(\df F_p)^{-1} = \df (F^{-1})_{F(p)}$.
		\begin{myIndent}\pracTwo
			To see why, just combine parts (a) and (b). $\blacksquare$.\retTwo
		\end{myIndent}
	\end{enumerate}
\end{myIndent}

Now a downside of viewing the tangent space $\mathcalli{V}_p M$ as a collection of equivalence classes of curves is that the vector space structure on $\mathcalli{V}_p M$ is not at all obvious. So my next goal will be to make that structure clear.\retTwo

\begin{myIndent}
	\ul{Lemma:} Let $M$ be an $n$-dimensional $C^r$ manifold, let $p \in M$, and let $(U, \varphi)$ be any smooth chart on $M$ containing $p$. Then $\gamma_1, \gamma_2 \in \mathcalli{C}(M, p)$ satisfy that $[\gamma_1] = [\gamma_2]$ if and only if $(\varphi \circ \gamma_1)^\prime(0) = (\varphi \circ \gamma_2)^\prime(0)$.
	\begin{myIndent}\pracTwo
		Proof:\\
		Note for any $f \in C^1(M, \mathbb{R})$ that:
		
		{\centering$(f \circ \gamma_i)^\prime(0) = \nabla(f \circ \varphi^{-1})(\varphi(p)) \cdot (\varphi \circ \gamma_i)^\prime(0)$\retTwo\par}

		Since $\nabla(f \circ \varphi^{-1})(\varphi(p))$ depends only on $f$ and not on $\gamma_i$, it's now clear that if $(\varphi \circ \gamma_1)^\prime(0) = (\varphi \circ \gamma_2)^\prime(0)$ then we'll also have that $(f \circ \gamma_1)^\prime(0) = (f \circ \gamma_2)^\prime(0)$.\retTwo
		
		Conversely, suppose $v_1 \coloneqq (\varphi \circ \gamma_1)^\prime(0) \neq (\varphi \circ \gamma_2)^\prime(0) \eqqcolon v_2$. Then it suffices to find some $f \in C^1(M, \mathbb{R})$ such that:

		{\centering $\nabla(f \circ \varphi^{-1})(\varphi(p)) \cdot v_1 \neq \nabla(f \circ \varphi^{-1})(\varphi(p)) \cdot v_2$ \retTwo\par}

		Fortunately, because $v_1 \neq v_2$ in $\mathbb{R}^n$, we know there is some $w \in \mathbb{R}^n$ with\\ $w \cdot v_1 \neq w \cdot v_2$. Next we can define $g: \mathbb{R}^n \to \mathbb{R}$ by $g(x) = x \cdot w$. Clearly $g$ is a\\ $C^\infty$ function on $\varphi(U)$ with $\nabla g(\varphi(p)) = w$. If we then further set $\widetilde{g} \coloneqq g \circ \varphi$, we have that $\widetilde{g} \in C^r(M, \mathbb{R})$.\retTwo

		By standard arguments on LCH spaces as well as with bump functions on $C^r$\\ manifolds, we can find a precompact open set $V \subseteq M$ with $p \in V \subseteq \overline{V} \subseteq U$\\ as well as a function $f \in C^r(M, \mathbb{R}) \subseteq C^1(M, \mathbb{R})$ such that $f = \widetilde{g}$ on $\overline{V}$. And\\ at last, we have on a neighborhood of $\varphi(p)$ that:
		
		{\centering\begin{tabular}{l}
			$\nabla(f \circ \varphi^{-1})(\varphi(p)) = \nabla(\widetilde{g} \circ \varphi^{-1})(\varphi(p))$\\ [4pt]
			$\phantom{\nabla(f \circ \varphi^{-1})(\varphi(p))} = \nabla(g \circ \varphi \circ \varphi^{-1})(\varphi(p)) = \nabla g(\varphi(p)) = w$. $\blacksquare$
		\end{tabular}\retTwo\par}
	\end{myIndent}
\end{myIndent}

One consequence of the prior lemma is that for any smooth chart $(U, \varphi)$ on an $n$-dimensional manifold $M$ containing a point $p$, there is a well-defined injective map $T_\varphi: \mathcalli{V}_p M \to \mathbb{R}^n$ given by $T_\varphi([\gamma]) \coloneqq (\varphi \circ \gamma)^\prime(0)$.\retTwo

\begin{myIndent}
	\ul{Proposition:} Suppose $p$ is \textit{not} a corner point of $M$. Then the map $T_\varphi$ defined above is also surjective.
	\begin{myIndent}\pracTwo
		Proof:\\
		Consider any $w \in \mathbb{R}^n$ and note that since $\varphi(U)$ must be an open neighborhood\\ of $\varphi(p)$ in either $H^n$ or $\mathbb{R}^n$, we know for a small enough $R > 0$ that either\\ $\varphi(p) + tw \in \varphi(U)$ or $\varphi(p) - tw \in \varphi(U)$ for all $t \in [0, R]$.\newpage

		Now we can define a map $\gamma$ going from either $[0, R]$ or $[-R, 0]$ to $M$ by setting $\gamma(t) \coloneqq \varphi^{-1}(\varphi(p) + tw)$. In either case, we have that $\gamma \in \mathcalli{C}(M, p)$ and:
		
		{\centering$(\varphi \circ \gamma)^\prime(0) = \left.\frac{\df}{\df t}\right|_{t = 0}(\varphi(p) + tw) = w$. $\blacksquare$\retTwo\par}

		\begin{myIndent}\myComment
			Unfortunately, this proof doesn't work if $p$ is a corner point of $M$. In fact, if $p$ is a corner point of $M$ then we'll not even have that $T_\varphi(\mathcalli{T}_p M)$ is a vector subspace of $\mathbb{R}^n$. So from here on out in this section I'll be assuming my $C^r$ manifolds do not have corners.\retTwo
		\end{myIndent}
	\end{myIndent}
\end{myIndent}

Now that we have a bijection $T_\varphi: \mathcalli{V}_p M \to \mathbb{R}^n$, we can let this bijection induce a vector\\ space structure on $\mathcalli{V}_p M$. In other words, given any $[\gamma_1], [\gamma_2] \in \mathcalli{V}_p M$ as well as any\\ $c_1, c_2 \in \mathbb{R}$, we define:

{\centering$c_1[\gamma_1] + c_2[\gamma_2] \coloneqq T_\varphi^{-1}(c_1T_\varphi([\gamma_1]) + c_2 T_\varphi([\gamma_2]))$.\retTwo\par}

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\begin{myIndent}
	\ul{Proposition:} The vector space structure on $\mathcalli{V}_p M$ is defined independently of any particular smooth chart $(U, \varphi)$ on $M$ containing $p$. In particular, suppose $(V, \psi)$ is another smooth chart on $M$ containing $p$. Then for any $[\gamma_1], [\gamma_2] \in \mathcalli{V}_p M$ as well as any $c_1, c_2 \in \mathbb{R}$ we have that:

	{\centering$T_\varphi^{-1}(c_1T_\varphi([\gamma_1]) + c_2 T_\varphi([\gamma_2])) = T_\psi^{-1}(c_1T_\psi([\gamma_1]) + c_2 T_\psi([\gamma_2]))$\retTwo\par}
	\begin{myIndent}\pracTwo
		Proof:\\
		Define the vectors:
		\begin{itemize}
			\item $v \coloneqq c_1T_\varphi([\gamma_1]) + c_2 T_\varphi([\gamma_2]) = c_1(\varphi \circ \gamma_1)^\prime(0) + c_2(\varphi \circ \gamma_2)^\prime(0)$,
			\item $w \coloneqq c_1T_\psi([\gamma_1]) + c_2 T_\psi([\gamma_2]) = c_1(\psi \circ \gamma_1)^\prime(0) + c_2(\psi \circ \gamma_2)^\prime(0)$.\retTwo
		\end{itemize}

		Next let $[\widetilde{\gamma}] \coloneqq T_\varphi^{-1}(v)$. Thus $\widetilde{\gamma}$ is a $C^1$ path on $M$ with $\widetilde{\gamma}(0) = p$ and $T_\varphi([\widetilde{\gamma}]) = (\varphi \circ \widetilde{\gamma})^\prime(0) = v$. To prove the proposition, we need to show that $[\widetilde{\gamma}] = T_\psi^{-1}(w)$. Or in other words, we need to show that $T_\psi([\widetilde{\gamma}]) = (\psi \circ \widetilde{\gamma})^\prime(0) = w$.\retTwo
