\hTwo To address this short coming we instead consider derivations acting on a stalk. Recall\\ the definition of sheaves, stalks, and germs given on \inLinkRap{Page 1026 reference}{pages 1026-1027}. Then given any\\ $C^r$ manifold $M$ (with $r \in \mathbb{Z}_{\geq 1} \cup \{\infty\}$) and any $0 \leq k \leq r$, we consider the sheaf\\ $\mathcalli{F}: \mcateg{Op}(M) \to \mcateg{CAlg}_\mathbb{R}$ such that $\mathcalli{F}(U) = C^k(U, \mathbb{R})$ and for any open sets $V \subseteq U \subseteq M$\\ we have that $r^U_V(f) = f|_V$.
\begin{myTindent}\myComment
	Here $C^0(U, \mathbb{R}) \coloneqq C(U, \mathbb{R})$.\retTwo
\end{myTindent}

For any $p \in M$ we denote the stalk of $\mathcalli{F}$ at $p$ to be the set $C_p^k(M)$. Also, given any open set $U$ and $C^k$ map $f: U \to \mathbb{R}$, we typically denote their germ as either $[(U, f)]$ or $[f]$.\retTwo

Given any germ $[f] \in C^k_p(M)$, it is well-defined to evaluate $f(p)$.
\begin{myIndent}\pracTwo
	This is because if $[(U, f)] = [(V, g)]$ then there is some open set $W \subseteq U \cap V$ containing $p$ such that $f|_W = g|_W$. But in turn we know that $f(p) = g(p)$.\retTwo
\end{myIndent}

Now, we say a \udefine{derivation} on $C^k_p(M)$ is an $\mathbb{R}$-linear map $v: C^k_p(M) \to \mathbb{R}$ which also satisfies that if $[f]$ and $[g]$ are germs then $v([f][g]) = f(p)v([g]) + g(p)v([f])$. The set of all derivations on $C^k_p(M)$ is easily checked to be a real vector space and we denote that vector space by $\mathcalli{D}_p(M)$.\retTwo

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\begin{myIndent}\mySepThree\end{myIndent}

Let $M$ be an $n$-dimensional $C^r$ manifold (where $r \in \mathbb{Z}_{\geq 1} \cup \{\infty\}$). Then let $U \subseteq M$ be an open set, let $p \in M$, and consider any smooth chart $(V, \varphi)$ such that $p \in V \subseteq U$. Note that $\varphi$ defines local coordinates $(x_1, \ldots, x_n)$ on $V$. I'll also write $\varphi(p) = (x_1(p), \ldots, x_n(p))$.\retTwo

\exTwo\ul{Lemma 1:} If $f \in C^k(U, \mathbb{R})$ (where $k \geq 1$) and $f(p) = 0$ then there is some open subset $W \subseteq V$ containing $p$ such that:

{\centering$f|_W = \sum\limits_{i=1}^n (x_i - x_i(p))h_i$,\\ [4pt]\par}

where $h_i \in C^{k-1}(W, \mathbb{R})$ for all $i$.
\begin{myIndent}\exThreeP
	Proof:\\
	To start off, if $\varphi(V)$ is not open in $\mathbb{R}^n$, then by definition $f \circ \varphi^{-1}$ can be extended to\\ being a $C^k$ function $\widetilde{f}$ from an open set $\widetilde{V} \subseteq \mathbb{R}^n$.\retTwo 

	Next, let $0 < r$ be a real number such that $\overline{B_r(\varphi(p))} \subseteq \widetilde{V}$. Then for any\\ $x = (x_1, \ldots, x_n) \in \overline{B_r(\varphi(p))}$ we can write:

	{\centering $\widetilde{f}(x) = \sum\limits_{i=1}^n (x_i - x_i(p))\int_0^1 \frac{\partial}{\partial x_i} \widetilde{f}(\varphi(p) + s(x - \varphi(p)))\df s$ \retTwo\par}

	By Leibniz's rule, we can conclude that $\widetilde{h}_i : B_r(\varphi(p)) \to \mathbb{R}$ given by:
	
	{\center$\widetilde{h_i}(x) = \int_0^1 \frac{\partial}{\partial x_i} \widetilde{f}(\varphi(p) + s(x - \varphi(p)))\df s$\\ [4pt]\par}
	
	is a $C^{k-1}$ map. And now we let $W = \varphi^{-1}(\varphi(V) \cap B_r(\varphi(p)))$ and $h_i: W \to \mathbb{R}$ be defined by $h_i = \widetilde{h}_i \circ \varphi$. And then each $h_i$ will be a $C^{k-1}$ map and $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Now let $(V, \varphi)$ be the smooth chart from before and suppose $f \in C^1(U, \mathbb{R})$ with $f(p) = 0$. If there exists an open set $W \subseteq U \cap V$ such that $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ where each $h_i \in C^{0}(W)$ and $h_i(\varphi(p)) = 0$ for each $i$, we say $f$ \udefine{vanishes to first order at $p$}.\retTwo

\exTwo\ul{Lemma 2:} Let $W, W^\prime \subseteq U \cap V$ both be open sets containing $p$ and then suppose\\ $h_1, \ldots, h_n \in C^{0}(W, \mathbb{R})$ and $\overline{h_1}, \ldots, \overline{h_n} \in C^{0}(W^\prime, \mathbb{R})$ satisfy that:

{\centering$f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ and $f|_{W^\prime} = \sum_{i=1}^n (x_i - x_i(p))\overline{h_i}$.\\ [4pt]\par}

Then $h_i(p) = \overline{h_i}(p)$ for all $i$.
\begin{myIndent}\exThreeP
	Proof:\\
	By considering the set $W \cap W^\prime$ we can assume without loss of generality that $W = W^\prime$. Also, we can assume without loss of generality that $W$ is an open set in $\mathbb{R}_{\geq 0}^n$, $H^n$, or $\mathbb{R}^n$ since $\overline{h_i}(p) = h_i(p) = 0$ iff $(\overline{h_i} \circ \varphi^{-1})(\varphi(p)) = (h_i \circ \varphi^{-1})(\varphi(p)) = 0$. Finally, note that\\ $f = \sum_{i=1}^n (x_i - x_i(p))h_i = \sum_{i=1}^n (x_i - x_i(p))\overline{h}_i$ means that:
	
	{\center$0 \equiv \sum_{i=1}^n (x_i - x_i(p)) \cdot (h_i - \overline{h_i})$.\retTwo\par}
	
	Thus, we will be done once we show that if $0 \equiv \sum_{i=1}^n (x_i - x_i(p)) g_i$ where\\ $g_i \in C^{0}(W, \mathbb{R})$ then $g_i(x(p)) = 0$ for all $i$ (where $x(p) = (x_1(p), \ldots, x_n(p))$).\retTwo

	Fortunately, consider any $s > 0$ such that $x(p) + se_i \in W$. (Here $e_1, \ldots, e_n$ is the standard basis for $\mathbb{R}^n$). Then $0 = \int_0^s 0\df t = \int_0^s tg_i(x(p) + te_i)\df t$. But now if $g_i(x(p)) \neq 0$ then by continuity we know there is some $s > 0$ and $\varepsilon > 0$ such that $|g_i(x(0) + te_i)| > \varepsilon$ when $0 \leq t \leq s$. In turn, we have a contradiction as $\left|\int_0^s tg_i(x(p) + te_i)\df t\right| \geq s\varepsilon$ means that $0 = \int_0^s tg_i(x(p) + te_i)\df t \neq 0$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo More specifically, what the prior lemma shows is that if $f \in C^1(U, \mathbb{R})$ with $f(p) = 0$\\ and $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ for some open set $W \subseteq U \cap V$ containing $p$ and\\ some $C^{0}$ maps $h_1, \ldots, h_n$ then:

{\centering $h_i(p) = \left.\frac{\partial}{\partial x_i}\right|_p f$ for each $i$.\retTwo\par}

After all, the above equation holds for the functions $h_1, \ldots, h_n$ constructed in lemma 1, and lemma 2 then says that all other possible $h_1, \ldots, h_n$ must agree with the functions constructed in lemma 1. As a result:

{\centering $f$ vanishes to first order iff $\left.\frac{\partial}{\partial x_i}\right|_p f = 0$ for all $i$ iff $v(f) = 0$ for all $v \in T_p U$\retTwo\par}

This leads naturally to another result:\retTwo


\exTwo\ul{Lemma 3:} Now suppose $(V^\prime, \psi)$ is another smooth chart containing $p$. We'll let $\psi$ define the local coordinates $(y_1, \ldots, y_n)$ on $V^\prime$. Also I'll write $\psi(p) = y(p) = (y_1(p), \ldots, y_n(p))$. If $W \subseteq U \cap V$ is an open set containing $p$ and there exists functions $h_i \in C^{0}(W, \mathbb{R})$ such that $f|_W = \sum_{i=1}^n(x_i - x_i(p))h_i$ and $h_i(p) = 0$ for each $i$, then for any open set $\Omega \subseteq U \cap V^\prime$ containing $p$ as well as functions $g_i \in C^{0}(\Omega, \mathbb{R})$ such that $f|_\Omega = \sum_{i=1}^n(y_i - y_i(p))g_i$, we know that $g_i(p) = 0$ for each $i$.\newpage
\begin{myIndent}\exThreeP
	Proof:\\
	We can conclude via lemmas 1 and 2 that $g_i(p) = \left.\frac{\partial}{\partial y_i}\right|_p f$ for each $i$. But now note from\\ [-6pt] \inLinkRap{page 1052 reference}{page 1052} that each linear operator $\left.\frac{\partial}{\partial y_i}\right|_p$ is a linear combination of the operators $\left.\frac{\partial}{\partial x_j}\right|_p$.\\ [-6pt] And since $\left.\frac{\partial}{\partial x_j}\right|_p f = h_j(p) = 0$ for each $j$ by assumption, we can conclude $g_i(p) = 0$\\ for each $i$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo As a result of the prior lemmas, we can say that $f$ vanishing to first order at $p$ is independent of our chosen local coordinates or the particular functions in our expression:

{\centering$f|_W = \sum\limits_{i=1}^n (x_i - x_i(p))h_i$.\retTwo\par}

Also as is hopefully obvious, if $[(U, f)] = [(\widetilde{U}, \widetilde{f})]$ in $C_p^1(M)$ and $f$ vanishes to first order at $p$ then so does $\widetilde{f}$. So, we can talk about a germ in $C^1_p(M)$ vanishing to first order.\retTwo

Now here's the new and improved definition of the tangent plane of $p$. For now I'll denote my old construction as $T_p M$. Meanwhile, I'll define a new construction by:

{\centering $\mathcal{T}_p M \coloneqq \{v \in \mathcalli{D}_p(M) : v([f]) = 0 \text{ if } [f] \text{ vanishes to first order}\}$. \retTwo\par}

Clearly $\mathcal{T}_p M$ is a vector subspace of $\mathcalli{D}_p(M)$.\retTwo

\exTwo\ul{Theorem:} There is a bijective linear map $\Phi: T_p M \to \mathcal{T}_p M$ such that $(\Phi(v))([f]) \coloneqq v(f)$ for all $f \in C^k(M, \mathbb{R})$.

\begin{myIndent}\exThreeP
	Proof:\\
	Suppose $[(U, f)]$ is any germ and note that by multiplying on a $C^r$ bump function, we can find a function $g \in C^1(M, \mathbb{R})$ such that $[(M, g)] = [(U, f)]$. So, for any $v \in T_p M$ we let $(\Phi(v))([f]) \coloneqq v(g)$.
	\begin{myIndent}\exPPP
		To see that $\Phi$ is well-defined, note that if $[(M, g)] = [(U, f)] = [(M, \widetilde{g})]$ then $g = f = \widetilde{g}$ on a neighborhood of $p$. So by proposition 3.8 on \inLinkRap{page 1033 reference}{page 1033} we know that $v(g) = v(\widetilde{g})$.\retTwo
	\end{myIndent}

	Showing that $\Phi(v) \in \mathcalli{D}_p(M)$ is then easy. Furthermore, if $[(U, f)]$ vanishes to first\\ order then we know for any $g \in C^k(M, \mathbb{R})$ with $[(M, g)] = [(U, f)]$ that $v(g) = 0$. So,\\ $\Phi(v) \in \mathcal{T}_p M$. Finally, it is easy to verify that $\Phi(c_1 v_1 + c_2 v_2) = c_1 \Phi(v_1) + c_2 \Phi(v_2)$\\ for all $c_1, c_2 \in \mathbb{R}$ and $v_1, v_2 \in T_p M$. So, $\Phi$ is a linear map from $T_p M$ to $\mathcal{T}_p M$.\retTwo

	To show that $\Phi$ is an isomorphism, let $(U, \varphi)$ be a smooth chart containing $p$ and defining\\ local coordiantes $\varphi(m) = x(m) = (x_1(m), \ldots, x_n(m))$. Then recall $\left.\frac{\partial}{\partial x_1}\right|_p, \ldots, \left.\frac{\partial}{\partial x_n}\right|_p$\\ [-4pt] is a basis for $T_p M$. Meanwhile, we can define a basis for $\mathcal{T}_p M$ as follows:
	\begin{myIndent}\exPPP
		Let $[(V, f)] \in C^1_p(M)$. By lemma 1 from a few pages ago can find an open subset $W \subseteq V \cap U$ as well as functions $h_i \in C^{0}(W, \mathbb{R})$ such that:
		
		{\centering$f|_W = f(p) + (x_i - x_i(p))h_i$.\newpage\par}

	Proof:\\

\exTwo\ul{Proposition 4.22:} Suppose $M$ and $N$ are $C^r$ manifolds with or without boundary, and\\ $F: M \to N$ is an injective $C^k$ immersion (where $k \leq r$). Then $F$ is a $C^k$ embedding if any of the following conditions hold:
\begin{enumerate}
	\item[(a)] $F$ is an open or closed map into $N$.
	\begin{myIndent}\exThreeP
		Why?\\
		$F$ being an open map into $N$ trivially means $F$ is an open map into the subspace topology of $F(M) \subseteq N$.\retTwo

		Meanwhile, if $F$ is a closed map then let $U \subseteq M$ be an open set. Then $M - U$ is a closed set so $F(M - U)$ is a closed subset of $N$. It follows that $N - F(M - U)$ is an open subset of $N$. And so $F(M) \cap (N - F(M - U)) = F(U)$ (which holds because $F$ is injective) is an open set in the subspace topology of $F(M) \subseteq N$.
		\begin{myIndent}\myComment
			More generally, this reasoning shows that if $X$ and $Y$ are topological spaces then any injective continuous map $f: X \to Y$ that is also open or closed is a \udefine{topological embedding} (i.e. a homeomorphism from $X$ to $f(X) \subseteq Y$).\retTwo
		\end{myIndent}
	\end{myIndent}

	\item[(b)] $F$ is a proper map.
	\begin{myIndent}\exThreeP
		Why?\\
		Let $E$ be any closed subset of $M$ and then suppose $q$ is any accumulation point of $F(E)$ in $N$. Since $N$ is an LCH space, we can pick a precompact open set $V \subseteq N$\\ [-1pt] containing $q$. And now $\overline{V} \cap F(E)$ also has $q$ as an accumulation point. Yet because\\ $F$ is proper, we know that $F^{-1}(\overline{V})$ is compact. Then since closed subsets of compact sets are also compact, we know that $E \cap F^{-1}(\overline{V})$ is compact. In turn, as $F$ is\\ continuous we have that $F(E \cap F^{-1}(\overline{V})) = F(E) \cap \overline{V}$ is compact in $N$ (and thus closed since $N$ is Hausdorff). So $F(E) \cap \overline{V}$ must contain all its accumulation points and so $q \in F(E)$.\retTwo

		Since $q$ was arbitrary, we've proven that $F(E)$ is a closed subset of $N$. So, $F$ is a closed map into $N$ and we can finish by citing part (a).
		\begin{myIndent}\myComment
			More generally, this reasoning shows that if $X$ is a topological space and $Y$ is an LCH space then any continuous proper map $f: X \to Y$ is also a closed map.\retTwo
		\end{myIndent}
	\end{myIndent}

	\item[(c)] $M$ is a compact manifold.
	\begin{myIndent}\exThreeP
		Why?\\
		$F: M \to F(N)$ is a continuous bijection, $M$ is compact, and $F(N)$ is Hausdorff. Thus $F$ is a homeomorphism.\retTwo

		Suppose $U \subseteq Y$ again satisfies that $f^{-1}(U)$ is an open subset of $X$. Then we know that $X - f^{-1}(U)$ is a closed subset of $X$. So because $f$ is a closed map, we know that $f(X - f^{-1}(U)) = f(X) - U$ is a closed subset of $Y$. Since $f$ is surjective, we in other words know that $Y - U$ is a closed set. Equivalently, $U$ is an open set. $\blacksquare$\retTwo
	\end{myIndent}
\end{itemize}

\exTwo\ul{Theorem 4.25 (Local Embedding Theorem):} Suppose $M$ and $N$ are $C^r$ manifolds where $M$ has no corners and $F: M \to N$ is a $C^k$ map. Then $F$ is a $C^k$ immersion if and only if for all $p \in M$ there exists an open set $U \subseteq M$ such that $F|_U : U \to N$ is a $C^k$ embedding.
\begin{myIndent}\exThreeP
	$(\Longleftarrow)$\\
	This direction is immediate.\retTwo

	$(\Longrightarrow)$\\
	Suppose $F$ is a $C^k$ immersion. Without loss of generality we can assume $M$ and $N$ are equipped with the unique $C^k$ structures containing their original $C^r$ structures since that would have no effect on whether $F$ is a $C^k$ immersion or $F|_U$ is a $C^k$ embedding. And this assumption is useful cause it let's us apply the rank theorem.\newpage

	Firstly we show that any $p \in M$ has a neighborhood $U$ on which $F$ is injective.
	\begin{myIndent}\exPPP
		First suppose $F(p) \notin \partial N$. Then as the set of interior points $\Int(N)$ is an open\\ set and $F$ is continuous, we know that there is an open set  $\Omega \subseteq M$ containing $p$\\ with $F(\Omega) \subseteq \Int(N)$. Then by either the rank theorem on \inLinkRap{The Rank Theorem}{page 1062} or the local\\ immersion theorem on \inLinkRap{Local Immersion Theorem boundary page 1101}{page 1101}, we know there is an open set $U_1 \subseteq M$ of $p$\\ where $F$ has the coordinate representation:
		
		{\centering$(x_1, \ldots, x_m) \mapsto (x_1, \ldots, x_m, 0, \ldots, 0)$.\retTwo\par}
		
		So $F$ is injective on $U_1$.\retTwo

		Meanwhile, suppose $F(p) \in \partial N$ and let $(W, \psi)$ be a smooth chart on $N$\\ containing $F(p)$. Then $U_0 \coloneqq F^{-1}(W)$ is an open subset of $M$ containing $p$.\\ Furthermore, let $\iota: H^n (\text{ or }\mathbb{R}_{\geq 0}^n) \hookrightarrow \mathbb{R}^n$ be the inclusion map (here $n = \dim(N)$).\\ Then $\iota \circ \psi \circ F|_{U_0}$ is a $C^k$ map from $M$ to a $C^k$ manifold without boundary.\\ Also, as $\df \iota_{a}$ and $\df \psi_{b}$ are bijections for all $a \in H^n$ and $b \in W$, we can conclude that $\iota \circ \psi \circ F|_{U_0}$ is a $C^k$-immersion still. Thus, we can apply the earlier reasoning to this map to find an open set $U_1 \subseteq U_0$ containing $p$ such that $(\iota \circ \psi \circ F|_{U_0})|_{U_1}$ is an injection. Equivalently, this implies $F|_{U_1}$ is an injection.\retTwo
	\end{myIndent}

	Now consider any $p \in M$ and let $U_1 \subseteq M$ be an open set containing $p$ on which $F$\\ is injective. Since $M$ is an LCH space, we can find a precompact open set $U$ such that\\ $p \in U \subseteq \overline{U} \subseteq U_1$. But now $F|_{\overline{U}}$ is an injective continuous function from a compact\\ set. Therefore, $F|_{\overline{U}}$ is a topological embedding. Next since restrictions of topological\\ embeddings are still topological embeddings, we know $F|_U$ is a topological embedding. And since $F|_U$ is a $C^k$ immersion, we can conclude that $F|_U: U \to N$ is a $C^k$ embedding. $\blacksquare$\retTwo
\end{myIndent}

Suppose $M$ and $N$ are $C^r$ manifolds. If $\pi: M \to N$ is a continuous map then a \udefine{section for $\pi$} is a continuous right inverse function $\sigma: N \to M$ (i.e. $\pi \circ \sigma = \myId_N$). A \udefine{local section for $\pi$} is a continuous map $\sigma: U \to M$ where $U \subseteq N$ is an open set and $\pi \circ \sigma = \myId_U$.\retTwo

\exTwo\ul{Theorem 4.26 (Local Section Theorem):} Suppose $M$ and $N$ are $C^r$ manifolds without\\ boundaries, and $\pi: M \to N$ is a $C^k$ map (where $k \leq r$). Then $\pi$ is a $C^k$ submersion\\ if and only if every $p \in M$ is in the image of a $C^k$ local section of $\pi$.
\begin{myIndent}\exThreeP
	$(\Longleftarrow)$\\
	Given any $p \in M$, let $\sigma: U \to M$ be a $C^k$ local section such that $q \in U$ and $\sigma(q) = p$. In other words, $q = \pi(\sigma(q)) = \pi(p)$. But now since $\pi \circ \sigma = \myId_U$, we can conclude that $\df \pi_p \circ \df \sigma_q = \myId_{T_q N}$. In turn, we must have that $\df \pi_p$ is surjective. And since $p$ was arbitrary, we can conclude $\pi$ is a submersion.\retTwo

	$(\Longrightarrow)$\\
	Suppose $\pi$ is a $C^k$ submersion. Then without loss of generality we can assume $M$ and $N$ are equipped with the unique $C^k$ structures containing their original $C^r$ structures. After all, this change won't effect whether $\pi$ is a $C^k$ submersion, or whether some local section of $\pi$ is $C^k$ or not. Also, this assumption lets us apply the rank theorem.\newpage

	Pick any $p \in M$ and let $q = \pi(p)$. By the rank theorem, we can choose smooth charts $(V_0, \varphi)$ on $M$ containing $p$ and $(U_0, \psi)$ containing $q$ such that $\varphi(p) = 0$, $\pi(V_0) \subseteq U_0$, and $\widehat{\pi} \coloneqq \psi \circ \pi \circ \varphi^{-1}$ has the form $\widehat{\pi}(x_1, \ldots, x_n, x_{n+1}, \ldots, x_m) = (x_1, \ldots, x_n)$.\retTwo

	For a small enough $\varepsilon > 0$ we have that the coordinate cube
	
	{\centering$C_\varepsilon \coloneqq \{(x_1, \ldots, x_m) : |x_i| < \varepsilon \text{ for all }i\}$\\ [4pt]\par}

	is an open subset of $\varphi(V_0)$ containing $0 = \varphi(p)$. In turn we have that:
	
	{\centering$C_\varepsilon^\prime \coloneqq \{(y_1, \ldots, y_n) : |y_i| < \varepsilon \text{ for all }i\}$\\ [4pt]\par}
	
	is an open subset of $\psi(U_0)$ containing $0 = \widehat{\pi}(0) = \widehat{\pi} \circ \varphi(p) = \psi(q)$ and satisfying that $\widehat{\pi}(C_\varepsilon) = C_\varepsilon^\prime$. Let $U = \psi^{-1}(C_\varepsilon^\prime)$. Then define $\iota: C_\varepsilon^\prime \to C_\varepsilon$ by:
	
	{\centering$\iota(y_1, \ldots, y_n) = (y_1, \ldots, y_n, \underbrace{0, \ldots, 0}_{m - n \text{ zeros}})$.\retTwo\par}
	
	Finally, let $\sigma = \varphi^{-1} \circ \iota \circ \psi|_U$. Then if $q^\prime \in U$ and $\varphi(q^\prime) = (y_1, \ldots, y_n)$ we have that:

	{\center\begin{tabular}{l}
		$\pi \circ \sigma(q^\prime) = \psi^{-1} \circ \widehat{\pi} \circ \varphi \circ (\varphi^{-1} \circ \iota \circ \psi)(q^\prime)$\\ [4pt]
		$\phantom{\pi \circ \sigma(q^\prime)} = \psi^{-1} \circ \widehat{\pi} \circ \iota(y_1, \ldots, y_n) = \psi^{-1} \circ \widehat{\pi}(y_1, \ldots, y_n, 0, \ldots, 0)$\\ [4pt]
		$\phantom{\pi \circ \sigma(q^\prime) = \psi^{-1} \circ \widehat{\pi} \circ \iota(y_1, \ldots, y_n)} = \psi^{-1}(y_1, \ldots, y_n) = q^\prime$\\ [4pt]
	\end{tabular}\retTwo\par}

	So $\pi \circ \sigma = \myId_U$. Also $\sigma$ is a $C^k$ map and 
	
	{\centering$\sigma(q) = \varphi^{-1} \circ \iota(0, \ldots, 0) = \varphi^{-1}(0, \ldots, 0) = p$. $\blacksquare$\retTwo\par}
\end{myIndent}

\hTwo Based on theorems 4.25 and 4.26, we also get the following purely topological definitions. Suppose $X$ and $Y$ are topological spaces and $f: X \to Y$ is a continuous function.
\begin{itemize}
	\item $f$ is a \udefine{topological immersion} if every point $x \in X$ has a neighborhood $U \subseteq X$ such that $f|_U : U \to Y$ is a topological embedding.
	\item $f$ is a \udefine{topological submersion} if every point $x \in X$ is in the image of a (continuous) local section for $f$.\retTwo
\end{itemize}

The prior two theorems say that any $C^k$ immersion / submersion is a topological\\ immersion / submersion. That said, be aware that a topological immersion or submersion that happens to also be a $C^k$ map is not necessarily a "$C^k$ immersion" or "$C^k$ submersion".
\begin{myIndent}\pracTwo
	For example, let $f: \mathbb{R} \to \mathbb{R}$ be defined by $f(t) = t^3$. Then $f$ is a $C^\infty$ map which is also a homeomorphism. And clearly homeomorphisms are trivially topological immersions and submersions. That said, as $\df f_0$ is the trivial linear map as $f^\prime(0) = 0$, we know $f$ is not a $C^\infty$ immersion or submersion.\retTwo
\end{myIndent}

\exTwo\ul{Theorem 4.27:} Let $M$ and $N$ be $C^r$ manifolds without boundaries, and suppose\\ $\pi: M \to N$ is a $C^k$ submersion. Then $\pi$ is an open map. Also consequently, if\\ $\pi$ is surjective then $\pi$ is a quotient map.
\begin{myIndent}\exThreeP
	Proof:\\
	Suppose $W \subseteq M$ is an open set and $q \in \pi(W) \subseteq N$. We can pick $p \in W$ with $\pi(p) = q$. And then by the local section theorem, we know there is an open subset $U \subseteq N$ containing $q$ as well as a $C^k$ map $\sigma: U \to M$ such that $\sigma(q) = p$ and $\pi \circ \sigma = \myId_U$. Yet now note\newpage that $\sigma^{-1}(W)$ is an open subset of $U$ (since $\sigma$ is continuous) and it contains $q$. Also note that if $y \in \sigma^{-1}(W)$ then $y = \pi(\sigma(y)) \in \pi(W)$ Therefore, we've shown that $q$ is in the topological interior of $\pi(W)$. And since $q$ was arbitrary, this proves that $\pi(W)$ is an open set. $\blacksquare$\retTwo
\end{myIndent}

\Hstatement\blab{Exercise 4-6:} Let $M$ be a nonempty compact $C^r$ manifold without a boundary, and let $N$ be a connected non-compact $C^r$ without a boundary. Then there is no $C^k$ submersion $\pi: M \to N$ (where $k \leq r$).
\begin{myIndent}\HexOne
	Proof:\\
	By the prior theorem we must have that such a function $\pi$ is an open map. Thus in particular we can conclude that $\pi(M)$ is an open subset of $N$. That said, as $M$ is compact and $\pi$ is continuous, we must also have that $\pi(M)$ is a compact subset of $N$. Since $N$ is Hausdorff, we can conclude $\pi(M)$ is a clopen set. And since $N$ is connected, we thus must have that $\pi(M)$ is either the empty set or all of $N$.\retTwo

	$\pi(M)$ can't be the empty set however because $M \neq \emptyset$. Simultaneously, since $N$ is not compact we can't have that $\pi(M) = N$. Thus we have a contradiction. $\blacksquare$\retTwo
\end{myIndent}

Suppose $M$ is a $C^r$ manifold. A \udefine{$C^r$ embedded submanifold of $M$} is a subset $S \subseteq M$ which is a topological manifold when equipped with the subspace topology and which has a $C^r$ structure such that the inclusion map $\iota: S \to M$ is a $C^r$ embedding.\retTwo

\myComment Remarks:\newpage
\begin{itemize}
	\item If $k \leq r$ then $M$ has a unique $C^k$ structure containing its $C^r$ structure. Based on this structure on $M$, if $S$ is a $C^k$ manifold then we can ask if $\iota: S \to M$ is a $C^k$ embedding. And if so we call $S$ a $C^k$ embedded submanifold of $M$.
	\item Some authors also refer to embedded submanifolds as \udefine{regular submanifolds}.\retTwo
\end{itemize}

\hTwo If $M$ is a $C^r$ manifold and $S \subseteq M$ is a $C^k$ embedded submanifold of $M$ then the\\ \udefine{codimension of $S$ in $M$} is defined to be the difference $\dim(M) - \dim(S)$. Here $M$ is\\ called the \udefine{ambient manifold for $S$}. If $S$ has codimension $1$ we say $S$ is an \udefine{embedded\\ hypersurface}.\retTwo

\exTwo\ul{Proposition 5.1:} Suppose $M$ is a $C^r$ manifold without a boundary. Then the embedded $C^k$\\ submanifolds of codimension $0$ (without boundaries) in $M$ are precisely the open\\ submanifolds of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	If $U \subseteq M$ is an open submanfold then we know $\dim(U) = \dim(M)$. Furthermore, $U$ is equipped with the subspace topology and we know from before that the inclusion map $\iota: U \hookrightarrow M$ is a $C^k$ immersion and submersion. It follows that $\iota$ is a $C^k$ embedding. And as $M$ has no boundary, neither does $U$.\retTwo

	Conversely, suppose $U \subseteq M$ is an embedded $C^k$ submanifold in $M$ of codimension $0$. By definition the inclusion map $\iota: U \hookrightarrow M$ is a $C^k$ embedding. Yet as $\dim(M) = \dim(U)$ and $\iota$ is an immersion, we know from proposition 4.8(b) on \inLinkRap{Proposition 4.8 page 1061}{page 1061} that $\iota$ is a local $C^k$-diffeomorphism. In turn, by proposition 4.6(c) on \inLinkRap{Properties of local $C^k$-diffeomorphisms page 1060}{page 1060} we know $\iota$ is an open map. So, $\iota(U) = U$ is an open subset of $M$.\retTwo

	Finally, suppose $(V, \varphi)$ is a smooth chart on $M$ (equipped with its unique $C^k$ structure). Then $U \cap V$ is an open subset of $U$ and I claim that $(U \cap V, \varphi)$ is a smooth map on $U$. After all, suppose $(W, \psi)$ is any other smooth chart on $U$ as an embedded submanifold. Then as $\iota: U \hookrightarrow M$ is a $C^k$ map we know that $\varphi \circ \iota \circ \psi^{-1} = \varphi \circ \psi^{-1}$ is $C^k$ on $\psi((U \cap V) \cap W)$.\retTwo

	Meanwhile, as $\iota$ is a local $C^k$-diffeomorphism, we know for every $p \in (U \cap V) \cap W$ that there is some open set $\Omega \subseteq U$ containing $p$ such that $\iota$ is a $C^k$-diffeomorphism from $\Omega$ (as an open submanifold of $U$) to $\Omega$ (as an open submanifold of $M$). In turn, $\psi \circ \varphi^{-1}$ is a $C^k$ map on $\psi((U \cap V) \cap W \cap \Omega)$. So, $\psi \circ \varphi^{-1}$ is $C^k$ at every $p \in (U \cap V) \cap W$. And with that, we've proven that the $C^k$ structure on $U$ as an embedded submanifold must be the same as the $C^k$ structure on $U$ as an open submanifold. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition 5.2:} Suppose $M$ and $N$ are $C^r$ manifolds, and $F: N \to M$ is a $C^r$ embedding. Let $S = F(N)$. With the subspace topology $S$ is a topological manifold. Also $S$ has a unique $C^r$ structure making it into an embedded submanifold of $M$ and satisfying that $F$ is a $C^r$-diffeomorphism from $N$ to $S$.
\begin{myIndent}\exThreeP
	Proof:\\
	$S$ being a subspace of $M$ and $M$ being second countable and Hausdorff means that $S$ is also second countable and Hausdorff. Also by assumption $F$ is a homeomorphism from $N$ to $S$. Therefore, $N$ being locally Euclidean of dimension $\dim(N)$ means that $S$ is as well.\newpage

	Now we equip $S$ with the $C^r$ structure given by the charts $(F(U), \varphi \circ F^{-1})$ where $(U, \varphi)$ is any smooth chart on $N$. Them being $C^r$ compatible with each other is a result of the smooth charts on $N$ being $C^r$ compatible with each other. So, $S$ is a $C^r$ manifold. It's also easy to see now that $F$ is a $C^r$-diffeomorphism. After all, for any $p \in S$ we can pick a smooth chart $(U, \varphi)$ on $N$ containing $F^{-1}(p)$. Then $(F(U), \varphi \circ F^{-1})$ is a smooth chart on $S$ containing $p$, and:

	{\centering $\varphi \circ F \circ (\varphi \circ F^{-1})^{-1} = \myId_{\varphi(U)}$ is a $C^r$ map.\\ [4pt]\par}

	So, $F^{-1}: S \to N$ is a $C^r$ map.\retTwo

	Next, to show uniqueness just note that if $F$ is to be a diffeomorphism from $N$ to $S$ then for any smooth chart $(U, \varphi)$ on $N$ we must have that $(F(U), \varphi \circ F^{-1}$) is a smooth chart on $S$. Similarly, for every smooth chart $(V, \psi)$ on $S$ we must have that $(F^{-1}(V), \psi \circ F)$ is a smooth chart on $N$. So the atlas described in the prior paragraph is maximal and $F$ being a $C^r$-diffeomorphism from $N$ to $S$ automatically implies that $S$ is equipped with that maximal structure.\retTwo
	
	Finally, I need to show that the map $\iota: S \to M$ is a $C^r$-embedding. Fortunately, consider that $\iota$ is given by the composition:

	\[\begin{tikzcd}
		S & N & M
		\arrow["{F^{-1}}", from=1-1, to=1-2]
		\arrow["\iota"', curve={height=12pt}, from=1-1, to=1-3]
		\arrow["F", from=1-2, to=1-3]
	\end{tikzcd}\]
	
	Since $F^{-1}$ is a $C^r$-diffeomorphism and $F$ is a $C^r$-embedding, we can conclude $\iota$ is a $C^r$ embedding. So, $S$ is an embedded submanifold of $M$. $\blacksquare$\retTwo

	\begin{myIndent}\myComment
		As a side note, if $F: N \to M$ is a $C^k$ embedding (where $k \leq r$) we can also now say that $S = F(N)$ has a unique $C^k$ structure making it into an embedded submanifold of $M$ such that $F$ is a $C^k$-diffeomorphism from $N$ to $S$. After all, we can just re-equip $M$ and $N$ with the unique $C^k$ structures containing their original $C^r$ structures and then everything works the same.\retTwo
	\end{myIndent}

\ul{Proposition 5.3:} Suppose $M$ and $N$ are $C^r$ manifolds. For each $p \in N$ the subset\\ $M \times \{p\} \subseteq M \times N$ (called a \udefine{slice} of the product manifold) is an embedded submanifold of $M \times N$ diffeomorphic to $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	The map $F: M \to M \times N$ given by $F(m) \coloneqq (m, p)$ is a $C^r$ embedding. After all, it's a $C^r$ map that is a topological embedding. Furthermore, recall from \inLinkRap{Proposition 3.14 page 1059}{page 1059} how we standardly identify $T_{(m, p)}(M \times N)$ with $T_m M \oplus T_p N$.\retTwo

	\ul{Lemma:} Suppose $N, M_1, \ldots, M_k$ are $C^r$ manifolds. Then let $M \coloneqq \prod_{i=1}^k M_i$ and\\ for each $i$ let $\pi_i: M \to M_i$ be the projection map. Given any $(p_1, \ldots, p_k) \in M$ we\\ identify $T_{(p_1, \ldots, p_k)}M$ with $\bigoplus_{i=1}^k T_{p_i}M_i$ via the standard identification. Now suppose for\\ each $1 \leq i \leq k$ that $F_i: N \to M_i$ is a $C^1$ map. Then let $F: N \to M$ be defined\\ by $F(q) = (F_1(q), \ldots, F_k(q))$. Now for any $q \in N$ we have that:
	
	{\centering$\df F_q(v) = (\df (F_1)_q (v),\phantom{.} \ldots,\phantom{.} \df (F_k)_q (v))$.\\ [4pt]\par}
	\begin{myIndent}\exPPP
		Why?\\
		Suppose $v \in T_q N$ and $f: M \to \mathbb{R}$ is a $C^1$ map. Then $\df F_q(v) f$ is identified with:
		
		{\centering\begin{tabular}{l}
			$(\df (\pi_1)_{F(q)} \circ \df F_q(v), \ldots \df (\pi_k)_{F(q)} \circ \df F_q(v)) = (\df (\pi_1 \circ F)_q (v), \ldots, \df (\pi_k \circ F)_q (v))$\\ [4pt]
			$\phantom{(\df (\pi_1)_{F(q)} \circ \df F_q(v), \ldots \df (\pi_k)_{F(q)} \circ \df F_q(v))} = (\df (F_1)_q (v),\phantom{.} \ldots,\phantom{.} \df (F_k)_q (v))$
		\end{tabular}\retTwo\par}
	\end{myIndent}

	Based on the prior lemma, we can see for any $v \in T_m M$ that:
	
	{\centering$\df F_m(v) = (v, 0) \in T_m M \oplus T_p N$.\retTwo\par}

	It follows that $\df F_m$ is injective for all $m \in M$. So, $F$ is an immersion. $\blacksquare$\retTwo
\end{myIndent}

\exTwo\ul{Proposition 5.4:} Suppose $M$ is a $C^r$ $m$-dimensional manifold, $N$ is a $C^r$ $n$-dimensional manifold, $U \subseteq M$ is open, and $f: U \to N$ is a $C^k$ map. Let $\Gamma(f) \subseteq M \times N$ denote the graph of $f$:

{\centering $\Gamma(f) \coloneqq \{(m, f(M)) \in M \times N : m \in U\}$ \\ [4pt]\par}

Then $\Gamma(f)$ is a $C^k$ $m$-dimensional embedded submanifold of $M \times N$.
\begin{myIndent}\exThreeP
	Proof:\\
	Define a map $\gamma_f: U \to M \times N$ by $\gamma_f(m) = (m, f(m))$. This will be a $C^k$ map because the inclusion map $\iota: U \to M$ is $C^r$ and similarly the map $f: U \to N$ is $C^k$. Also let $\pi_M: M \times N \to M$ be the projection map. Then $\myId_{U} = \pi_M \circ \gamma_f$ implies that $\gamma_f$ injective. Also for any $m \in U$ we have $\df (\pi_M)_{(m, f(m))} \circ \df (\gamma_{f})_m = \df (\myId_{U})_m = \myId_{T_m U}$. So, for every $m \in U$ we have that $\df (\gamma_{f})_m$ is injective. In other words, $\gamma_{f}$ is an immersion. Finally, $\gamma_f$ is a topological embedding since its inverse function is given by $\pi_M |_{\Gamma(f)}$.\retTwo
	
	And now we've shown that $\Gamma(f)$ is the image of a $C^k$ embedding $\gamma_f$ from $U$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo An embedded submanifold $S \subseteq M$ is said to be \udefine{properly embedded} if the inclusion map $S \hookrightarrow M$ is a proper map.\retTwo

\exTwo\ul{Proposition 5.5:} Suppose $M$ is a $C^r$ manifold and $S \subseteq M$ is an embedded submanifold. $S$ is properly embedded if and only if $S$ is a closed subset of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	Suppose the inclusion map $\iota: S \to M$ is a proper map. Then we know that it is a closed map (see \inLinkRap{Page 1104 reference}{page 1104}). In turn, $S = \iota(S)$ must be a closed subset of $M$.\retTwo

	Conversely, note that $\iota$ being a topological embedding with a closed image implies that $\iota$ is proper.
	\begin{myIndent}\exPPP
		This is because if $K \subseteq M$ is a compact set, then $S \cap K$ will also be compact since $S$ is closed. And now as $\iota^{-1}$ is a continuous map, we know that $\iota^{-1}(K) = \iota^{-1}(S \cap K)$ is compact. So, $\iota$ is proper. \exThreeP $\blacksquare$\retTwo
	\end{myIndent}
\end{myIndent}

\ul{Corollary 5.6:} Every compact embedded submanifold is properly embedded.
\begin{myIndent}\exThreeP
	Proof:\\
	Compact subsets of Haudorff spaces are closed. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition 5.7:} Suppose $M$ and $N$ are $C^r$ manifolds, and $f: M \to N$ is a $C^k$ map. Then after equipping $\Gamma(f) \coloneqq \{(m, f(M)) \in M \times N : m \in M\}$ with the smooth structure defined in proposition 5.4, we have that $\Gamma(f)$ is properly embedded.\newpage
\begin{myIndent}\exThreeP
	Proof:\\
	Note that $\gamma_f : M \to M \times N$ given by $\gamma_f(m) = (m, f(m))$ has the map\\ $\pi_M : M \times N \to M$ given by $\pi_M(m, n) = m$ as a continuous left inverse. In other words, $\pi_M \circ \gamma_f = \myId_M$. As a result we can conclude that $\gamma_f : M \to M \times N$ is a proper map. To see why, I claim the following:\retTwo

	\ul{Lemma:} If $X, Y$ are topological spaces and $g: X \to Y$ and $h: Y \to X$ are continuous functions such that $h \circ g = \myId_X$, then $g$ must be proper.
	\begin{myIndent}\exPPP
		Why?\\
		Suppose $K \subseteq Y$ is a compact set. Since $h$ is continuous, we know that $h(K) \subseteq X$ is a compact set. But also let $F \coloneqq g^{-1}(K)$. Then:
		
		{\centering\begin{tabular}{l}
			$F = \myId_X(F) = (h \circ g)(F) = h(K \cap g(X))$\\ [4pt]
			$\phantom{F = \myId_X(F) = (h \circ g)(F)} = h(K) \cap (h \circ g)(X)$\\ [4pt]
			$\phantom{F = \myId_X(F) = (h \circ g)(F)} = h(K) \cap \myId_X(X) = h(K) \cap X = h(K)$.
		\end{tabular}\retTwo\par}

		So, we've shown that $g^{-1}(K)$ is compact.\retTwo
	\end{myIndent}

	Since $\gamma_f$ is a continuous proper map though and $M \times N$ is an LCH space, we can conclude that $\gamma_f$ is a closed map. So $\gamma_f(M) = \Gamma(f)$ is a closed subset of $M \times N$. And by proposition 5.5, we are done. $\blacksquare$\retTwo
\end{myIndent}

If $U \subseteq \mathbb{R}^n$ is an open set and $k \in \{0, 1, \ldots, n\}$ then a \udefine{$k$-slice} of $U$ is a set of the form 

{\centering$S \coloneqq \{(x_1, \ldots, x_k, x_{k + 1}, \ldots, x_n) \in U : x_{k+1} = c_{k+1}, \ldots, x_n = c_n\}$\\ [4pt]\par}

where $c_{k+1}, \ldots, c_n \in \mathbb{R}$. In practice we'll usually have $c_{k+1}, \ldots, c_n = 0$.\retTwo

More generally, if $M$ is an $n$-dimensional $C^r$ manifold without a boundary and $(U, \varphi)$ is a smooth chart on $M$, then we say a set $S \subseteq U$ is a \udefine{$k$-slice of $U$} if $\varphi(S)$ is a $k$-slice of $\varphi(U) \subseteq \mathbb{R}^n$.\retTwo

Given a subset $S$ of a $C^r$ $n$-dimensional manifold $M$ and a nonnegative integer $k$, we say $S$ satisfies the \udefine{local $k$-slice condition} if every $p \in S$ is contained in a smooth chart $(U, \varphi)$ such that $S \cap U$ is a $k$-slice of $U$. Such a chart $(U, \varphi)$ is then called a \udefine{slice chart} and the corresponding local coordinates are called \udefine{slice coordinates}.\retTwo

\exTwo\ul{Theorem 5.8 (Local Slice Criterion):} Let $M$ be a $C^r$ $n$-dimensional manifold without a boundary.
\begin{itemize}
	\item If $S \subseteq M$ is an embedded $k$-dimensional $C^r$ submanifold without a boundary then $S$ satisfies the local $k$-slice condition.
	\begin{myIndent}\exThreeP
		Proof:\\
		Let $p \in S$. Since the inclusion map $\iota: S \hookrightarrow M$ is an immersion, we can use the rank theorem to conclude that there are smooth charts $(U, \varphi)$ on $S$ and $(V, \psi)$ on $M$ such that $p \in U$, $U \subseteq V \cap S$, $\varphi(p) = 0$, and:\newpage
		
		{\centering$\psi \circ \iota \circ \varphi^{-1}(x_1, \ldots, x_k) = (x_1, \ldots, x_k, \underbrace{0, \ldots, 0}_{n-k\text{ zeros}})$.\retTwo\par}

		Choose $\varepsilon > 0$ small enough so that $U$ and $V$ both contain coordinate cubes $(-\varepsilon, \varepsilon)^k$ and $(-\varepsilon, \varepsilon)^n$ respectively centered at $p$ and denote these coordinate balls as $U_0$\\ and $V_0$. Then $U_0$ is a $k$-slice of $V_0$.\retTwo

		Finally, as $S$ has the subspace topology, we know there is an open set $W \subseteq M$ such that $W \cap S = U_0$. Set $V_1 = V_0 \cap W$. Then $(V_1, \psi|_{V_1})$ is a smooth chart for $M$ containing $p$ such that $V_1 \cap S = U_0$ and $U_0$ is still a $k$-slice of $V_1$.
		\begin{myIndent}\myComment
			\hypertarget{idk reference 7 back}{As} a side note if $\pi: \mathbb{R}^n \to \mathbb{R}^k$ is the projection map onto the first $k$ coordinates, then $\varphi = \pi \circ \psi|_{U_0}$. So we've proven for all $p \in S$ can find a slice chart $(\Omega, \phi)$ on $M$ containing $p$ such that $(\Omega \cap S, \pi \circ \phi)$ is a smooth chart on $S$ containing $p$. This observation will be referenced later on \inLinkRap{idk reference 7}{page \_\_\_\_}.\retTwo
		\end{myIndent}
	\end{myIndent}
	\item Conversely, if $S \subseteq M$ is a subset satisfying the local $k$-slice condition, then it is a\\ $k$-dimensional topological manifold without a boundary when equipped with the\\ subspace topology relative to $M$. Also, there is a $C^r$ structure on $S$ which makes\\ $S$ into a $C^r$ embedded submanifold of $M$.
	\begin{myIndent}\exThreeP
		Proof:\\
		If we give $S$ the subspace topology, then we know $S$ is second countable and\\ Hausdorff because $M$ is. Meanwhile, the idea for constructing an atlas for $S$ is that if $(x_1, \ldots, x_n)$ is the coordinates of a slice chart for $S$ then we can define a chart with the coordinates $(x_1, \ldots, x_k)$.\retTwo

		We let $\pi: \mathbb{R}^n \to \mathbb{R}^k$ be the projection onto the first $k$ coordinates. Now suppose $(U, \varphi)$ is any slice chart on $M$. We let:
		
		{\centering$V \coloneqq U \cap S$, $\widehat{V} \coloneqq (\pi \circ \varphi)(V)$, and $\psi \coloneqq \pi \circ \varphi|_V$.\retTwo\par}

		Note that as $S$ is equipped with the subspace topology, $V$ is an open subset of $S$. Also, by the definition of a slice chart, $\varphi(V) = A \cap \varphi(U)$ for some set:
		
		{\centering$A = \{(x_1, \ldots, x_n) \in \mathbb{R}^n : x_{k+1} = c_{k+1}, \ldots, x_n = c_n\}$\\ [4pt]\par}
		
		where $c_{k+1}, \ldots, c_n \in \mathbb{R}$ are constants. It follows that $\varphi(V)$ is an open subset of $A$. And since $\pi|_A : A \to \mathbb{R}^k$ is a $C^\infty$ diffeomorphism from $A$ to $\mathbb{R}^k$, we know that $\widehat{V} = \pi(\varphi(V))$ is an open subset of $\mathbb{R}^k$.\\ [-4pt]

		Moreover note that $\psi: V \to \widehat{V}$ has a continuous inverse function $\varphi^{-1} \circ j|_{\widehat{V}}$ where $j: \mathbb{R}^k \to \mathbb{R}^n$ is the $C^\infty$ function $(x_1, \ldots, x_k) \mapsto (x_1, \ldots, x_k, c_{k+1}, \ldots, c_n)$. So, $(V, \psi)$ is a well-defined interior chart on $S$.\retTwo

		Since $S \subseteq M$ has the local $k$-slice condition, we know every $p \in S$ is contained in some chart constructed above. So, $S$ is a topological manifold without a boundary. Also trivially we have that the inclusion map $\iota: S \hookrightarrow M$ is a topological embedding.

		\mySepFour\newpage

		To equip $S$ with a $C^r$ structure, we prove that the charts on $S$ constructed priorly are all $C^r$ compatible with each other. So let $(U, \varphi)$ and $(U^\prime, \varphi^\prime)$ be two different slice charts for $S$ in $M$ and then let $(V, \psi)$ and $(V^\prime, \psi^\prime)$ be the corresponding charts for $S$ constructed priorly. The transition map $\psi^\prime \circ \psi^{-1}$ will be given by $\pi \circ \varphi \circ \varphi^{-1} \circ j$ where $j: \mathbb{R}^k \to \mathbb{R}^n$ is the map defined priorly. Thus, $\psi^\prime \circ \psi^{-1}$ is a $C^r$ map since it is a composition of four $C^r$ maps. And since $\psi$ and $\psi^\prime$ were arbitrary, we've proven that our priorly constructed atlas defines a $C^r$ structure on $S$.\retTwo

		Finally, consider any $p \in S$ and let $(U, \varphi)$ be a slice chart for $S$ in $M$ containing\\ $p$. Then let $(V, \psi)$ be the corresponding smooth chart on $S$. We know $p \in V$,\\ $\iota(V) \subseteq U$ and also that there exists $c_{k+1}, \ldots, c_n \in \mathbb{R}$ such that $\varphi \circ \iota \circ \psi^{-1}$ is\\ given by $(x_1, \ldots, x_k) \mapsto (x_1, \ldots, x_k, c_{k+1}, \ldots, c_n)$. It follows that $\iota$ is a $C^r$ map. And by looking at the Jacobian of $(x_1, \ldots, x_k) \mapsto (x_1, \ldots, x_k, c_{k+1}, \ldots, c_n)$, we can also conclude that $\df \iota_p$ is full rank (and injective). $\blacksquare$\retTwo
	\end{myIndent}
\end{itemize}

\exTwo\ul{Theorem 5.11:} Let $M$ be a $C^r$ $n$-dimensional manifold without corners. Then $\partial M$ equipped with the subspace topology is an $(n-1)$-dimensional topological manifold without a boundary that can be equipped with a $C^r$ structure such that $\partial M$ is an embedded\\ submanifold of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	$\partial M$ is trivially Hausdorff and second countable when equipped with the subspace topology, and similarly the inclusion map $\iota: \partial M \hookrightarrow M$ is trivially a topological embedding when $\partial M$ is equipped with the subspace topology.\retTwo

	To show that $\partial M$ is locally Euclidean, we construct an atlas. Let $\pi: \mathbb{R}^n \to \mathbb{R}^{n-1}$ be given by $\pi(x_1, \ldots, x_n) = (x_1, \ldots, x_{n-1})$ and let $j: \mathbb{R}^{n-1} \to \mathbb{R}^n$ be given by:
	
	{\centering$j(x_1, \ldots, x_{n-1}) = (x_1, \ldots, x_{n-1}, 0)$.\retTwo\par}

	Now given any smooth boundary chart $(U, \varphi)$ for $M$, we define a chart $(V, \psi)$ on $\partial M$ by $V = U \cap \partial M $ and $\psi = \pi \circ \varphi|_{V}$.
	\begin{myIndent}\exPPP
		By the invariance of the boundary, we know that:
		
		{\centering$\varphi(U \cap \partial M) = \{(x_1, \ldots, x_n) \in \varphi(U) : x_n = 0\} = \varphi(U) \cap \partial H^n$.\newpage\par}
		
		It follows that $\varphi(U \cap \partial M)$ is an open subset of $\partial H^n$. And since $\pi$ is a\\ homeomorphism from $\partial H^n$ to $\mathbb{R}^{n-1}$, we can conclude that $\psi(V)$ is an open\\ subset of $\mathbb{R}^{n-1}$.\retTwo

		Furthermore, note that $\psi$ has the continuous inverse $\varphi^{-1} \circ j|_{\psi(V)}$. So, $\psi$ is a\\ homeomorphism. And since every $p \in \partial M$ is contained in some boundary chart, we can conclude that the charts constructed here form an atlas for $\partial M$.\retTwo
	\end{myIndent}

	By identical reasoning as in the prior theorem, we can show that the charts constructed prior are $C^r$ compatible with each other, and also that $\partial M$ equipped with the $C^r$ structure they generate is an embedded submanifold of $M$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Suppose $M$ and $N$ are $C^k$ manifolds without boundaries, $\Phi: M \to N$ is any map, and $c \in N$ is any point. The set $\Phi^{-1}(\{c\})$ is called a \udefine{ level set of $\Phi$}. If $N = \mathbb{R}^n$ and $c = 0$, the level set $\Phi^{-1}(\{c\})$ is called the \udefine{zero set of $\Phi$}.\retTwo

In general, a level set or zero set of a $C^k$ map is not going to be a submanifold.\retTwo

\exTwo\ul{Theorem 2.29:} Let $M$ be a $C^k$ manifold and suppose $K \subseteq M$ is any closed set. Then there is a $C^k$ nonnegative function $f: K \to \mathbb{R}$ such that $f^{-1}(\{0\}) = K$.
\begin{myIndent}\exThreeP
	Proof:\\
	We start by considering the case where $M = \mathbb{R}^n$ and $K \subseteq \mathbb{R}^n$ is a closed set. Then for every $x \in K^\comp$ let $r_x \in (0, 1)$ be such that $B_{r_x}(x) \subseteq K^\comp$. Thus $\{B_{r_x / 2}(x) : x \in K^\comp\}$ is an open cover of $K^\comp$, and since $K^\comp$ is second countable, we know there is a countable subcover. So, we've proven there are sequences $\{x_i\}_{i \in \mathbb{N}} \subseteq \mathbb{R}^n$ and $\{r_i\}_{i \in \mathbb{N}} \subseteq (0, 1)$ satisfying that:
	
	{\center$K^\comp = \bigcup_{i = 1}^\infty B_{\sfrac{r_i}{2}}(x_i) = \bigcup_{i = 1}^\infty B_{r_i}(x_i)$.\retTwo\par}

	Now let $h: \mathbb{R}^n \to [0, 1]$ be a $C^\infty$ function such that $h \equiv 1$ on $B_{1/2}(0)$ and\\ $\supp(h) \subseteq B_1(0)$. Also for each $i \in \mathbb{Z}_{\geq 0}$ let $C_i \geq 1$ be some constant such that\\ $\|\partial^\alpha h\|_u \leq C_i$ whenever $|\alpha| \leq i$. And finally, define $f: \mathbb{R}^n \to [0, 1]$ by:

	{\centering $f(x) = \sum_{i=1}^\infty \frac{(r_i)^i}{2^i C_i} h\left(\frac{x - x_i}{r_i}\right)$ \retTwo\par}

	More generally, for any multi-index $\alpha$ let $g_\alpha(x) = \sum_{i=1}^\infty \frac{(r_i)^{i - |\alpha|}}{2^i C_i} \partial^\alpha h\left(\frac{x - x_i}{r_i}\right)$. Clearly\\ $g_{(0, \ldots, 0)} = f$. By the Weierstrass M-test we can see that each $g_\alpha$ is continuous. Also I\\ [2pt] claim $\partial^\alpha f = g_\alpha$ for every multi-index $\alpha$. This specifically follows as an immediate\\ consequence of the following lemma.\newpage

	\ul{Lemma:} Suppose $U \subseteq \mathbb{R}^n$ is an open set and let $e_1, \ldots, e_n$ be the standard basis for $\mathbb{R}^n$. Also denote $\partial_j = \frac{\partial}{\partial x_j}$. Next suppose $\{f_k\}_{k \in \mathbb{N}} \in C^1(U, \mathbb{R})$, and also that $g, h \in C(U, \mathbb{R})$ satisfy that $f_k \to g$ pointwise and $\partial_j f_k \to h$ uniformly on compact sets. Then $\partial_j g = h$.
	\begin{myIndent}\exPPP
		Why?\\
		Let $x \in U$ be any point. Then for small enough $t > 0$ we have that $B_t(x) \subseteq U$. And then by the fundamental theorem of calculus we have that:

		{\centering $f_k(x + te_j) - f_k(x) = \int_0^t \partial_j f_k(x + se_j)\df s$ \retTwo\par}

		Since $[-t, t]$ is compact, $\partial_j f$ is continuous, and $\partial_j f_k \to h$ uniformly, we can\\ conclude by the dominated convergence theorem that:
		
		{\centering$\lim\limits_{k \to \infty}\int_0^t \partial_j f_k(x + se_j)\df s = \int_0^t h(x + se_j)\df s$.\retTwo\par}

		Also since $f_k \to g$ pointwise we know:
		
		{\centering$g(x + te_j) - g(x) = \int_0^t h(x + se_j)\df s$\retTwo\par}

		And now by the fundamental theorem of calculus, we can conclude that $\frac{\df }{\df t}(g(x + te_j)) = h(x + te_j)$ exists and equals $h$ when $|t|$ is small enough. In other words, $\partial_j g = h$ at $x$.\retTwo
	\end{myIndent}

	By the prior lemma, we can conclude that if $\beta$ and $\alpha$ are multi-indices such that\\ $\partial^\alpha = \frac{\partial}{\partial x_j} \circ \partial^\beta$ then $\frac{\partial}{\partial x_j} g_\beta = g_\alpha$. Then the fact that $\partial^\alpha f = g_\alpha$ immediately follows\\ by induction.\retTwo

	To finish, off, note that if $x \in K$ then $x \notin B_{r_i}(x_i)$ for any $i$ and so $f(x_i) = 0$. Meanwhile, if $x \notin K$ then $x \in B_{\sfrac{r_i}{2}}(x_i)$ for some $i$ and hence $h(\frac{x - x_i}{r_i}) = 1$. It follows that $f(x) > 0$. So, $K = f^{-1}(\{0\})$.\\ [-6pt]
	
	\mySepFour 

	Now that this theorem has been shown in the case that $M = \mathbb{R}^n$, we can now consider the general case for a $C^k$ manifold. Suppose $K \subseteq M$ is any closed set and let $\{(B_\alpha, \varphi_\alpha)\}_{\alpha \in A}$ be a covering of $M$ with smooth coordinate balls, half balls, and corner balls. We can let $\{\psi_\alpha\}_{\alpha \in A}$ be a $C^r$ subordinate partition of unity as constructed on \inLinkRap{Existence of Partitions of Unity}{page 1032}.\retTwo
	
	Note that $B_\alpha \cap K$ is a closed subset relative to $B_\alpha$ for every $\alpha \in A$. In turn, we can see that $\varphi(B_\alpha \cap K)$ is closed relative to $\varphi(B_\alpha)$. Then there is some closed set $F_\alpha \subseteq \mathbb{R}^n$ such that $F_\alpha \cap \varphi(B_\alpha) = \varphi(B_\alpha \cap K)$. And now by the prior reasoning, we let $f_\alpha: \mathbb{R}^n \to [0, \infty)$ be a $C^\infty$ map such that $f_\alpha^{-1}(\{0\}) = F_\alpha$.\retTwo

	But now $f = \sum_{\alpha \in A} \psi_\alpha \cdot (f_\alpha \circ \varphi_\alpha)$ defines a $C^r$ map on $M$ with $f^{-1}(\{0\}) = K$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo However, by making assumptions about the rank of a $C^k$ map $\Phi$, we can guarentee that the level sets of $\Phi$ are embedded submanifolds.\retTwo
% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\exTwo\ul{Theorem 5.12 (Constant-Rank Level Set Theorem):} Let $M$ and $N$ be $C^k$ manifolds without boundaries and let $\Phi: M \to N$ be a $C^k$ map with constant rank $r$. Then each level set of $\Phi$ is a properly embedded submanifold of codimension $r$ in $M$ without a boundary.
\begin{myIndent}\exThreeP
	Proof:\newpage
	Write $m = \dim(M)$ and $n = \dim(N)$. Also let $c \in N$ be arbitrary and let\\ $S \coloneqq \Phi^{-1}(\{c\}) \subseteq M$. Now for any $p \in S$, the rank theorem gives us smooth\\ charts $(U, \varphi)$ on $M$ and $(V, \psi)$ on $N$ such that $\varphi(p) = 0$ and:
	
	{\centering$\psi \circ \Phi \circ \varphi^{-1}(x_1, \ldots, x_r, x_{r+1}, \ldots, x_m) = (x_1, \ldots, x_r, \underbrace{0, \ldots, 0}_{n - r \text{ zeros}})$.\retTwo\par}

	Yet note that $\Psi(p) = c$ implies that $\psi(c) = 0$. And now:
	
	{\centering$U \cap S = U \cap \Phi^{-1}(\{c\}) = \varphi^{-1}(\{(x_1, \ldots, x_m) \in \varphi(U) : x_1 = \cdots = x_r = 0\})$.\retTwo\par}

	It follows that $U \cap S$ is a $(m - r)$-slice of $(U, \varphi)$. Since $p \in S$ was arbitrary, we can conclude that $S$ satisfies the local $(m - r)$-slice condition. So, $S$ is an embedded $(m - r)$-dimensional $C^k$ submanifold of $M$ without a boundary.\retTwo

	Also as $S$ is a closed set (since $\{c\}$ is a closed subset of $N$ and $\Phi$ is continuous), we know $S$ is properly embedded. $\blacksquare$\retTwo
\end{myIndent}

\ul{Corollary 5.13 (Submersion Level Set Theorem):} If $M$ and $N$ are $C^k$ manifolds without boundaries and $\Phi: M \to N$ is a $C^k$ submersion, then each level set of $\Phi$ is a properly embedded submanifold of $M$ without a boundary whose codimension is $\dim(N)$ 
\begin{myIndent}\exThreeP
	Proof:\\
	Apply the prior theorem and use the fact that $\mRank(\Phi) = \dim(N)$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Suppose $M$ and $N$ are $C^k$ manifolds without boundaries and $\Phi: M \to N$ is a $C^k$ map. We say a point $p \in M$ is a \udefine{regular point of $\Phi$} if $\df \Phi_p$ is surjective. Otherwise we say $p$ is a \udefine{critical point of $\Phi$}. Also we say a point $c \in N$ is a \udefine{regular value of $\Phi$} if $\Phi^{-1}(\{c\})$ if every $p \in \Phi^{-1}(\{c\})$ is a regular point of $\Phi$. Otherwise, we say $c$ is a \udefine{critical value of $\Phi$}. Finally, a level set $\Phi^{-1}(\{c\})$ is a \udefine{regular level set} if $c$ is a regular value of $\Phi$.
\begin{myIndent}\myComment
	As a side note, the set of regular points in $M$ is always an open subset of $U$. This is\\ because of proposition 4.1 on \inLinkRap{Proposition 4.1 pages 1054-1056}{pages 1054-1056}. So, we can always restrict the domain\\ of $\Phi$ to just an open-submanifold where $\Phi$ is a submersion.\retTwo
\end{myIndent}

\exTwo\ul{Corollary 5.14 (Regular level set Theorem):} Suppose $M$ and $N$ are $C^k$ manifolds without boundaries and $\Phi: M \to N$ is a $C^k$ map. Every regular level set of $\Phi$ is a properly embedded $C^k$ submanifold without a boundary and with codimension equal to $\dim(N)$.
\begin{myIndent}\exThreeP
	Proof:\\
	Let $c \in N$ be a regular value of $\Phi$. The set $U \subseteq M$  of regular points of $\Phi$ is open in $M$ and contains $\Phi^{-1}(\{c\})$ since $c$ is a regular value. So, $\Phi|_U$ is a $C^k$ submersion and the prior corollary shows that $\Phi^{-1}(\{c\})$ is a $C^k$ submanifold of $U$ without a boundary and with codimension equal to $\dim(N)$. Since the composition of two $C^k$ embeddings is a $C^k$ embedding, and the inclusion $\Phi^{-1}(\{c\}) \hookrightarrow M$ is the composition of the inclusions\\ $\Phi^{-1}(\{c\}) \hookrightarrow U$ and $U \hookrightarrow M$, we can also conclude $\Phi^{-1}(\{c\})$ is an embedded $C^k$\\ submanifold of $M$.\retTwo
	
	Also once again, $\Phi$ being continuous and $\{c\}$ being a closed subset of $N$ means that\\ $\Phi^{-1}(\{c\})$ is a closed subset of $M$. So, $\Phi^{-1}(\{c\})$ is a properly embedded submanifold. $\blacksquare$\newpage
\end{myIndent}

\exTwo\ul{Proposition 5.16:} Let $S$ be a subset of a $C^r$ $m$-dimensional manifold $M$ without a boundary. Then $S$ is a $k$-dimensional embedded submanifold of $M$ without a boundary if and only if every $p \in S$ has an open neighborhood $U \subseteq M$ such that $U \cap S$ is a zero set of a $C^r$ submersion $\Phi: U \to \mathbb{R}^{m - k}$.
\begin{myIndent}\exThreeP
	$(\Longrightarrow)$\\
	Consider any $p \in S$ and let $(U, \varphi)$ be a slice chart for $S$ in $M$ containing $p$. By translating the codomain of $\varphi$, we can without loss of generality assume that:
	
	{\centering$\varphi(U \cap S) \coloneqq \{(x_1, \ldots, x_m) \in \varphi(U) : x_{k+1} = 0, \ldots, x_m = 0\}$.\retTwo\par}
	
	Now let us write $\varphi(q) = (x_1, \ldots, x_m)$ for all $q \in U$. Next define $\pi: \mathbb{R}^{m} \to \mathbb{R}^{m - k}$ by $(x_1, \ldots, x_m) \mapsto (x_{k+1}, \ldots, x_m)$. And finally, define $\Phi \coloneqq \pi \circ \varphi$. Since both $\pi$ and $\varphi$ are $C^r$ submersions, we can conclude that $\Phi: U \to \mathbb{R}^{m - k}$ is also a $C^r$ submersion. Also, $\Phi^{-1}(\{(0, \ldots, 0)\}) = U \cap S$.\retTwo

	$(\Longleftarrow)$\\
	By the submersion level set theorem, we can conclude that for every $p \in S$ there is an open set $U \subseteq M$ containing $p$ such that $S \cap U$ is a $k$-dimensional $C^r$ embedded submanifold of $M$ without a boundary. And now we just apply the comment on \inLinkRap{Page 1116 comment}{page 1116}. $\blacksquare$\retTwo
\end{myIndent}

Up until now, we've worked with submanifolds equipped with the subspace topology. However, this can be limiting. So, we adopt the following terminology.\retTwo

Let $M$ be a $C^r$ manifold. A \udefine{$C^r$ immersed submanifold of $M$} is a subset $S \subseteq M$ endowed with some topology (not necessarily the subspace topology) turning it into a topological manifold, as well a $C^r$ structure such that the inclusion map $\iota: S \hookrightarrow M$ is a $C^r$ immersion.\retTwo

\myComment Remarks:
\begin{itemize}
	\item If $k \leq r$ then we can define immersed $C^k$ submanifolds of $M$ identically as on \inLinkRap{Page 1111 remark}{page 1111}.
	\item Since all embedded submanifolds are also immersed submanifolds, we take the convention that a "$C^r$ submanifold" of $M$ is only understood by default to be immersed. If we want that submanifold to also be embedded then we will add that adjective.
	\item The \udefine{codimension} of a $C^r$ submanifold $S$ of $M$ is still defined as $\dim(M) - \dim(S)$. And $S$ is a \udefine{$C^r$ hypersurface} if $S$ has codimension $1$.\retTwo
\end{itemize}

\exTwo\ul{Proposition 5.18:} Suppose $M$ and $N$ are $C^r$ manifolds, and $F: N \to M$ is an injective $C^r$ immersion. Also let $S \coloneqq F(N)$. Then $S$ has a unique topology and $C^r$ structure making $S$ into a $C^r$ submanifold of $M$ such that $F$ is a $C^r$-diffeomorphism from $N$ to $S$.
\begin{myIndent}\exThreeP
	Proof:\\
	To start off, we define a topology on $S$ by saying that $U \subseteq S$ is open if $F^{-1}(U)$ is an open subset of $N$.\newpage
	\begin{myIndent}\exPPP
		Note that this is a well-defined topology. After all, $F^{-1}(S) = N$ and\\ $F^{-1}(\emptyset) = \emptyset$ are both open in $N$ and hence $S$ and $\emptyset$ are open subsets of $S$.\\ Also suppose $\{U_\alpha\}_{\alpha \in A}$ is a collection of subsets of $S$ such that $F^{-1}(U_\alpha)$ is open\\ in $N$ for all $\alpha$. Then $F^{-1}(\bigcup_{\alpha \in A} U_\alpha) = \bigcup_{\alpha \in A}F^{-1}(U_\alpha)$ is open in $N$. Similar\\ reasoning shows the topology on $S$ is closed under finite intersections.\retTwo

		Importantly, this topology on $S$ is the \textit{unique} topology $S$ must be equipped with\\ in order for $F$ to be a homeomorphism from $N$ to $S$ (which is necessary if we want $F$ to also be a $C^r$-diffeomorphism).\retTwo
	\end{myIndent}

	Now $N$ being second countable and Hausdorff means that $S$ is as well. Plus we can equip $S$ with an atlas given by the charts $(F(U), \varphi \circ F^{-1})$ where $(U, \varphi)$ is any chart on $N$. By identical reasoning as in the proof of proposition 5.2 on \inLinkRap{proposition 5.2 page 1111}{pages 1111-1112}, this atlas defines a $C^r$ structure on $S$ such that $S$ is a topological manifold and $F: N \to S$ is a $C^r$-diffeomorphism. (In fact this atlas is itself maximal and $F$ being a $C^r$-diffeomorphism from $N$ to $S$ would automatically imply $S$ is equipped with this maximal atlas).\retTwo

	Finally, the inclusion map $S \hookrightarrow M$ is the composition of the diffeomorphism\\ $F^{-1}: S \to N$ and the immersion $F: N \to M$. So, $S \hookrightarrow M$ is an immersion. $\blacksquare$\retTwo
\end{myIndent}

\exTwo\ul{Proposition 5.22:} If $M$ is a $C^r$ manifold and $S \subseteq M$ is a $C^r$ (immersed) submanifold\\ without corners, then for every $p \in S$ there is an open set $U \subseteq S$ containing $p$ such that $U$ is a $C^r$ embedded submanifold of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	By the local embedding theorem on \inLinkRap{local embedding theorem page 1105}{pages 1105-1106}, since the inclusion $\iota: S \hookrightarrow M$ is an immersion we know every $p \in S$ has an open neighborhood $U \subseteq S$ such that $\iota|_U$ is an embedding. And then the image of an embedding is always an embedded submanifold. $\blacksquare$\retTwo
\end{myIndent}

It only says we can find an open set $U$ in the topology of $S$ containing $p$ such that $U$ is an embedded submanifold.\\ [-6pt]

Let $M$ be an $n$-dimensional $C^r$ manifold without a boundary (where $n \geq 1$) and let $(U, \varphi)$ be a smooth chart on $M$. Given an integer $k \in \{0, 1, \ldots, n\}$, we say a \udefine{$k$-dimensional half slice of $U$} is a set $V \subseteq U$ such that:

{\centering $\varphi(V) = \{(x_1, \ldots, x_n) \in \varphi(U) : x_{k+1} = c_{k+1}, \ldots, x_n = c_n\}$,\\ [4pt]\par}

where $c_{k+1}, \ldots, c_n \in \mathbb{R}$.\retTwo

We say a subset $S \subseteq M$ satisfies the \udefine{local $k$-slice condition for a manifold with a boundary} (what a mouthful) if every $p \in S$ is contained in a smooth chart $(U, \varphi)$ such that $S \cap U$ is either a $k$-slice or a $k$-half slice of $U$. In the former case, $(U, \varphi)$ is called an \udefine{interior slice chart} for $S$ in $M$. Meanwhile, in the latter case $(U, \varphi)$ is called a \udefine{boundary slice chart} for $S$ in $M$.\retTwo

By using the local immersion theorem for manifolds with boundaries (see \inLinkRap{Local Immersion Theorem boundary page 1101}{pages 1101-1102}) in place of the rank theorem, we can prove an analogous result to theorem 5.8 on \inLinkRap{Theorem 5.8 page 1114}{pages 1114-1116}. \exTwo\ul{Theorem 5.51:} Let $M$ be a $C^r$ $n$-dimensional manifold without a boundary and let $S \subseteq M$ be a $C^r$ embedded $k$-dimensional submanifold without corners. Then $S$ satisfies the local $k$-slice condition for a manifold with a boundary.
\begin{myIndent}\color{red}

\end{myIndent}

Conversely, if $S \subseteq M$ satisfies the local $k$-slice condition for a manifold with a boundary, then $S$ equipped with the subspace topology is a topological manifold. Furthermore, it has a $C^r$ structure making it into an embedded $C^r$ submanifold in $M$ without corners.
\begin{myIndent}\color{red}

I'd like to take a break from John Lee chapter 5 and introduce a different important topic in the theory of manifolds.\retTwo

Let $M$ be a topological space. A \udefine{(real) vector bundle of rank $k$ over $M$} is a topological space $E$ with a surjective continuous map $\pi: E \to M$ satisfying that:
\begin{enumerate}
	\item[(a)] For each $p \in M$ the fiber $E_p \coloneqq \pi^{-1}(\{p\})$ over $p$ is endowed with a structure turning it into a $k$-dimensional real vector space.
	\item[(b)] For each $p \in M$ there is an open neighborhood $U$ of $p$ in $M$ and a homeomorphism $\Phi: \pi^{-1}(U) \to U \times \mathbb{R}^k$ (called a \udefine{local trivialization of $E$ over $U$}) satisfying that:\newpage
	\begin{itemize}
		\item $\pi_U \circ \Phi = \pi$ (where $\pi_U: U \times \mathbb{R}^k \to U$ is the projection);
		\item For each $q \in U$ the restriction of $\Phi$ to $E_q$ is a vector space isomorphism from $E_q$ to $\{q\} \times \mathbb{R}^k \cong \mathbb{R}^k$.
	\end{itemize}
\end{enumerate}

If $E$ and $M$ are $C^r$ manifolds, $\pi: E \to M$ is a $C^r$ map, and the local trivializations can be chosen to be $C^r$ diffeomorphisms, then $E$ is said to be a \udefine{$C^r$ vector bundle over $M$}. Also, any local trivialization that is a $C^r$ diffeomorphism is called a "smooth" local trivialization.\retTwo

We call $E$ the \udefine{total space of the bundle}, $M$ the \udefine{base space of the bundle}, and $\pi: E \to M$ the \udefine{projection of the bundle}.\retTwo

\exTwo\ul{Exercise 10.1:} Suppose $\pi: E \to M$ is a $C^r$ vector bundle (of rank $k$). Then the projection\\ map $\pi$ is a $C^r$ submersion.
\begin{myIndent}\exThreeP
	Proof:\\
	Let $q \in E$ be any point, and then set $p = \pi(q)$. We can find an open set $U \subseteq M$ containing $p$ and a $C^r$ diffeomorphism $\Phi: \pi^{-1}(U) \to U \times \mathbb{R}^k$ that is a local trivialization. Because $\Phi$ is continuous, we know $V \coloneqq \pi^{-1}(U)$ is an open subset of $E$ containing $q$.\retTwo

	But now note that $\pi|_V = \pi_U \circ \Phi$ where $\pi_U: U \times \mathbb{R}^k \to U$ is the projection map. Therefore, $\df \pi_q = \df (\pi_U)_{\Phi(q)} \circ \df \Phi_q$. Yet we can also see that $\pi_U$ is a submersion. After all, for any $x \in U \times \mathbb{R}^k$ we have that:
	
	{\centering$T_x (U \times \mathbb{R}^k) \cong T_{\pi_U(x)} U \oplus T_{\pi_{\mathbb{R}^k}(x)}\mathbb{R}^k$ via the map sending $v$ to $(\df (\pi_U)_{x}v, \hspace{0.2em}\df (\pi_{\mathbb{R}^k})_x v)$.\retTwo\par}

	Thus, $\df (\pi_U)_x$ must be a surjection onto $T_{\pi_U(q)}U$. And since $x$ was arbitrary, we can\\ conclude $\pi_U$ is a submersion.
	\begin{myIndent}\myComment
		More generally, if $M_1, \ldots, M_k$ are any manifolds, $1 \leq i \leq k$, and\\ $\pi: M_1 \times \cdots \times M_k \to M_i$ is the projection map, then this reasoning\\ shows that $\pi$ is a submersion.\retTwo
	\end{myIndent}

	Also $\df \Phi_q$ must be an invertible map since $\Phi$ is a diffeomorphism. So, $\df \pi_q$ is the composition of two surjective linear maps, and in turn must be a surjective linear map itself. Since $q$ was arbitrary, this proves that $\pi: E \to M$ is a submersion. $\blacksquare$\retTwo
\end{myIndent}

\ul{Lemma 1.35:} Let $M$ be a set, $r \in \mathbb{Z}_{\geq 0} \cup \{\infty\}$ and suppose we are given a collection\\ $\{U_\alpha\}_{\alpha \in A}$ of subsets of $M$ together with maps $\varphi_\alpha: U_\alpha \to \mathbb{R}^n$ such that:
\begin{itemize}
	\item[(i)] For each $\alpha \in A$, $\varphi_\alpha$ is a bijection between $U_\alpha$ and an open subset $\varphi_\alpha(U_\alpha)$ of\\ $[0, \infty)^k \times \mathbb{R}^{n - k}$ (where $0 \leq k \leq n$);
	\item[(ii)] For every $\alpha, \beta \in A$ we have that $\varphi_\alpha(U_\alpha \cap U_\beta)$ is an open subset of $\varphi_\alpha(U_\alpha)$; 
	\item[(iii)] Whenever $U_\alpha \cap U_\beta \neq \emptyset$ then the map $\varphi_\beta \circ \varphi_\alpha^{-1}: \varphi_\alpha(U_\alpha \cap U_\beta) \to \varphi_\beta(U_\alpha \cap U_\beta)$ is a $C^r$ map;
	\item[(iv)] Countably many of the $U_\alpha$ cover $M$;\newpage
	\item[(v)] For any $p, q \in M$ either there exists $U_\alpha$ with both $p, q \in U_\alpha$ or there exists disjoint $U_\alpha$ and $U_\beta$ with $p \in U_\alpha$ and $q \in U_\beta$. 
\end{itemize}

Then $M$ has a unique topology such that each $(U_\alpha, \varphi_\alpha)$ is a chart. And if $r \geq 1$ then $M$ has a unique $C^r$ structure such that each $(U_\alpha, \varphi_\alpha)$ is a smooth chart.

\begin{myIndent}\pracTwo
	To start off, we equip $M$ with the topology generated by the collection:
	
	{\centering$\{\varphi_\alpha^{-1}(V) : \alpha \in A \text{ and } V \text{ is an open subset of } \varphi_\alpha(U_\alpha)\}$.\retTwo\par}
	
	To prove that the latter set is a basis, first note by (iv) that every $x \in M$ is contained in\\ some $U_\alpha$. Also suppose that $x \in \varphi_\alpha^{-1}(V) \cap \varphi_\beta^{-1}(W)$. Then (iii) implies that:
	
	{\centering$(\varphi_\beta \circ \varphi_\alpha^{-1})^{-1}(W)$ is an open subset of $\varphi_\alpha(U_\alpha \cap U_\beta)$.\retTwo\par} 
	
	By (ii), we also know $\varphi_\alpha(U_\alpha \cap U_\beta)$ is an open subset of $\varphi_\alpha(U_\alpha)$. Thus, $(\varphi_\beta \circ \varphi_\alpha^{-1})^{-1}(W)$ is also an open subset of $\varphi_\alpha(U_\alpha)$. And now, we can say that:
	
	{\centering$\varphi_\alpha^{-1}(V \cap (\varphi_\beta \circ \varphi_\alpha^{-1})^{-1}(W)) = \varphi_\alpha^{-1}(V) \cap \varphi_\beta^{-1}(W)$\\ [4pt]\par}
	
	is also a basis set. Equality here is due to (i) saying $\varphi_\alpha$ and $\varphi_\beta$ are bijections.\retTwo
	
	Having equipped $M$ with this topology, it's now clear that each $\varphi_\alpha$ is a homeomorphism between $U_\alpha$ and $\varphi_\alpha(U_\alpha)$.
	\begin{myIndent}\exPPP
		To see why, first note by the definition of the topology that each $\varphi_\alpha$ is\\ continuous. I also claim each $\varphi_\alpha$ is an open map. After all, suppose $V \subseteq U_\alpha$\\ is any open set and $x \in V$ is any point. Then we know there exists some open\\ set $B \coloneqq \varphi_\beta^{-1}(W)$ where $W$ is an open subset of $\varphi_\beta(U_\beta)$ and $x \in \varphi_\beta^{-1}(W) \subseteq V$. Yet also note that as $B \subseteq V \subseteq U_\alpha$, we can further say that $W$ is an open subset of $\varphi_\beta(U_\beta \cap U_\alpha)$. In turn, by (ii) and (iii) we know that $(\varphi_\beta \circ \varphi^{-1}_\alpha)^{-1}(W) = \varphi_\alpha(B)$ is\\ an open subset of $\varphi_\alpha(U_\alpha)$. And since $\varphi_\alpha(B) \subseteq \varphi_\alpha(V)$, we've proven that $\varphi_\alpha(x)$ is in the topological interior of $\varphi_\alpha(V)$. Yet $x$ was arbitrary. So, $\varphi_\alpha(V)$ is an open subset of $\varphi_\alpha(U_\alpha)$.\retTwo
	\end{myIndent}

	It follows that $M$ is locally Euclidean. Furthermore, (iv) guarentees that $M$ is second\\ countable. Also, (v) guarentees that $M$ is Hausdorff. So, we've equipped $M$ with a\\ topology turning it into a topological $n$-dimensional manifold.
	\begin{myIndent}\myComment
		As a side note, this topology is the coarsest topology such that every $\varphi_\alpha$ is\\ continuous. Also suppose $\mathcal{T}^\prime$ is another topology on $M$ turning it into a\\ topological manifold with each $(U_\alpha, \varphi_\alpha)$ as a chart. Then suppose $V \in \mathcal{T}^\prime$. Since each $\varphi_\alpha$ is a homeomorphism, we must have that $\varphi_\alpha(V \cap U_\alpha)$ is an open subset of $\varphi_\alpha(U_\alpha)$ for all $\alpha$. Yet in turn $V \cap U_\alpha$ must be in our original topology. And since $V = V \cap M = V \cap \bigcup_{\alpha \in A}U_\alpha = \bigcup_{\alpha \in A} (V \cap U_\alpha)$, we can conclude that $V$ is open in our original topology. So, our original topology is also the finest topology that works. And this proves the topology we equipped $M$ with is unique.\retTwo
	\end{myIndent}

	Finally, if $r \geq 1$ then (iii) also tells us the $(U_\alpha, \varphi_\alpha)$ are $C^r$ compatible. And since they cover all of $M$, we know that they define a unique $C^r$ structure on $M$. $\blacksquare$\retTwo
\end{myIndent}

Suppose $M$ is a $C^r$ manifold (where $r \geq 1$) and $(U, \varphi)$ and $(V, \psi)$ are smooth charts such that $U \cap V = \emptyset$. We'll write $\varphi(p) = (x_1(p), \ldots, x_n(p))$ and $\psi(p) = (\widetilde{x_1}(p), \ldots, \widetilde{x_n}(p))$ Then for any $p \in U \cap V$ we can write $v \in T_p M$ as either $\sum_{i=1}^n c_i \left.\frac{\partial}{\partial x_i}\right|_p$ or as $\sum_{i=1}^n \widetilde{c_i} \left.\frac{\partial}{\partial \widetilde{x}_i}\right|_p$\\ [-2pt] (where each representation is unique). Importantly though, by taking coordinate\\ representations of the identity map $\myId_M$, we can see that:

{\centering$| \widetilde{c}_1, \ldots, \widetilde{c}_n {\rangle} = (\Df (\psi \circ \varphi^{-1})(\varphi(p)))| c_1, \ldots, c_n{\rangle}$.\retTwo\par}

\hTwo The most important vector bundle is the \udefine{tangent bundle} $TM$ of a $C^r$ manifold $M$.\retTwo
\begin{myIndent}\pracTwo
	We'll define $TM \coloneqq \coprod_{p \in M} T_p M = \{(p, v) : p \in M \text{ and } v \in T_p M\}$. Then we'll also define $\pi: TM \to M$ by $\pi(p, v) = p$.\retTwo

	\exTwo\ul{Lemma 3.18:} Suppose $r \geq 1$. For any $C^r$ $n$-dimensional manifold $M$, $TM$ has a natural topology (and $C^{r-1}$ structure if $r \geq 2$) turning $TM$ into a $2n$-dimensional $C^{r-1}$ manifold. Also, $\pi: TM \to M$ is a $C^{r-1}$ map.
	\begin{myIndent}\exThreeP
		Proof:\\
		Let $(U, \varphi)$ be any smooth chart and let $\varphi(p) = (x_1(p), \ldots, x_n(p))$ be the local coordinates on that chart. Then we define a map $\widetilde{\varphi}: \pi^{-1}(U) \to \varphi(U) \times \mathbb{R}^n$ by:
		
		{\centering$\widetilde{\varphi}(p, \sum\limits_{i=1}^n c_i \left.\frac{\partial}{\partial x_i}\right|_p) = (x_1(p), \ldots, x_n(p), c_1, \ldots, c_n)$.\retTwo\par}

		This is a bijection to the open set $\varphi(U) \times \mathbb{R}^n$. So, we let $(\pi^{-1}(U), \widetilde{\varphi})$ be a chart. By constructing such charts for all smooth charts on $M$, we get a collection of smooth charts for which condition (i) in the prior lemma is satisfied.\retTwo

		Next suppose $(U, \varphi)$ and $(V, \psi)$ are smooth charts on $M$. Then we can easily see that
		\begin{itemize}
			\item $\widetilde{\varphi}(\pi^{-1}(U) \cap \pi^{-1}(V)) = \varphi(U \cap V) \times \mathbb{R}^n$;
			\item $\widetilde{\psi}(\pi^{-1}(U) \cap \pi^{-1}(V)) = \psi(U \cap V) \times \mathbb{R}^n$.
		\end{itemize}

		The latter sets are all open so condition (ii) is satisfied.\retTwo

		Furthermore, $\widetilde{\psi} \circ \widetilde{\varphi}^{-1} : \widetilde{\varphi}(\pi^{-1}(U) \cap \pi^{-1}(V)) \to \widetilde{\psi}(\pi^{-1}(U) \cap \pi^{-1}(V))$ is\\ given by the map:
		
		{\centering\fontsize{12}{13}\begin{tabular}{l}$((x_1, \ldots, x_n),\hspace{0.2em} (c_1, \ldots, c_n)) \mapsto (\psi \circ \varphi^{-1}(x_1, \ldots, x_n),\hspace{0.2em} (\Df (\psi \circ \varphi^{-1})(\varphi(p)))| c_1, \ldots, c_n{\rangle})$\end{tabular}\retTwo\par}

		Since $\psi \circ \varphi^{-1}$ is a $C^r$ map, $\widetilde{\psi} \circ \widetilde{\varphi}^{-1}$ is thus a $C^{r-1}$ map. And this fulfills condition (iii).\retTwo

		To show condition (iv), note that a countable collection $\{(U_n, \varphi_n)\}_{n \in \mathbb{N}}$ of smooth charts covers $M$. In turn, we can see that $\{(\pi^{-1}(U_n), \widetilde{\varphi_n})\}$ is a countable collection of charts covering $TM$.\newpage

		Finally, to show condition (v) just note that if $p, q \in TM$ are in the same fiber of $\pi$ then there is a single chart $(\pi^{-1}(U), \widetilde{\varphi})$ containing $p$ and $q$. Meanwhile, if $\pi(p) \neq \pi(q)$ then we can find disjoint charts $(U, \varphi)$ and $(V, \psi)$ on $M$ such that $\pi(p) \in U$ and $\pi(q) \in V$. And then $\pi^{-1}(U)$ and $\pi^{-1}(V)$ are disjoint sets containing $p$ and $q$ respectively.\retTwo

		And now we just apply lemma 1.35 to get our topology and $C^{r-1}$ structure on $TM$ turning it into a  $2n$-dimensional $C^{r-1}$ manifold.\retTwo

		To see that $\pi: TM \to M$ is a $C^{r-1}$ map, just note for any smooth chart $(U, \varphi)$ on $M$ that $\pi(\pi^{-1}(U)) = U$ (since $\pi$ is surjective) and $\varphi \circ \pi \circ \widetilde{\varphi}^{-1} = \pi_U$ where $\pi_U$ is the projection map $U \times \mathbb{R}^n \to U$. It follows that every $q \in TM$ has a neighborhood on which $\pi$ is a $C^{r-1}$ map. So, $\pi$ is a $C^{r-1}$ map.\retTwo
	\end{myIndent}
\end{myIndent}

Now I claim that $\pi: TM \to M$ (where $TM$ is equipped with the topology described priorly) is a vector bundle of rank $n \coloneqq \dim(M)$ over $M$.

\end{myIndent}

Given an immersed $C^r$ submanifold $S$ of a $C^r$ manifold $M$, we can view its tangent space $T_p S$ at a point $p \in S$ as a subspace of the tangent space $T_p M$ of $M$ at $p$. To see how, recall by definition that the inclusion map $\iota: S \hookrightarrow M$ is an immersion. Thus at every $p \in S$ we have that $\df \iota_p : T_p S \to T_p M$ is an injection.\retTwo

In particular, suppose $v \in T_p S$ and $\widetilde{v} = \df \iota_p v \in T_p M$. Then:

{\centering$\widetilde{v}([f]) = \widetilde{v}([f, U]) = v([f \circ \iota, U \cap S]) = v([f|_{S}])$.\retTwo\par}

Now we shall view the injection $\df \iota_p$ as an identification of $T_p S$ as a subspace of $T_p M$.\retTwo

\exTwo\ul{Proposition 5.35:} Suppose $M$ is a $C^r$ manifold, $S \subseteq M$ is an immersed $C^r$ submanifold without corners, and $p \in S$. A vector $v \in T_p M$ is in $T_p S$ if and only if there is a $C^r$ curve $\gamma: J \to M$ such that:\\ [-20pt]
\begin{itemize}
	\item $J \subseteq \mathbb{R}$ is a compact interval containing $0$,
	\item $\gamma(J) \subseteq S$,
	\item $\gamma(0) = p$, 
	\item $\gamma^\prime(0) = v$, 
	\item $\gamma$ is a $C^r$ curve when viewed as a function from $J$ into $S$.
\end{itemize}

\begin{myIndent}\exThreeP
	$(\Longrightarrow)$\\
	Suppose there exists $\widetilde{v} \in T_p S$ such that $v = \df \iota_p \widetilde{v}$. Then we know there is a $C^r$ curve $\gamma: J \to S$ such that $J \subseteq \mathbb{R}$ is a compact interval containing $0$, $\gamma(0) = p$, and $\gamma^\prime(0) = \widetilde{v}$. But now $\iota \circ \gamma: J \to M$ is also a $C^r$ curve (since it is a composition of two $C^r$ functions). Also $(\iota \circ \gamma)^\prime(0) = \df \iota_p \circ \df \gamma_0(\left.\frac{\df}{\df t}\right|_0) = \df \iota_p(\gamma^\prime(0)) = \df \iota_p (\widetilde{v}) = v$\retTwo
	
	$(\Longleftarrow)$\\
	If we let $\widetilde{\gamma}$ be the curve $\gamma$ with it's codomain restricted to $S$ and then set $\widetilde{v} = \gamma^\prime(0)$, we have that:
	
	{\centering$\df \iota_p(\widetilde{v}) = \df \iota_p \circ \df \widetilde{\gamma}_0(\left.\frac{\df}{\df t}\right|_0) = \df (\iota \circ \widetilde{\gamma})_0(\left.\frac{\df}{\df t}\right|_0) = (\iota \circ \widetilde{\gamma})^\prime(0) = \gamma^\prime(0) = v$.\retTwo\par}

	So, $v \in T_p M$ is also in $T_p S$. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition 5.37:} Suppose $M$ is a $C^r$ manifold without a boundary, $S \subseteq M$ is an embedded $C^r$ submanifold (without corners), and $p \in S$. As a subspace of $T_p M$, we have that:

{\centering $T_p S = \{v \in T_p M : v(f) = 0 \text{ if } f \in C^r(M, \mathbb{R}) \text{ with } f \equiv 0 \text{ on } S\}$. \retTwo\par}

	Proof:\\

But now if $f \in C^r(M, \mathbb{R})$\\ is a function vanishing on $S$, we know that $f \circ \iota \equiv 0$. So:
	
	{\centering$v(f) = \df \iota_p(w)f = w(f \circ \iota) = w(0) = 0$.\retTwo\par}

	\begin{myIndent}\exPPP

\hTwo If $S \subseteq M$ is an embedded $C^r$ submanifold of a manifold $M$, we say a $C^r$ map $\Phi: M \to N$ such that $S$ is a regular level set of $\Phi$ is a \udefine{defining map for $S$}. If $N = \mathbb{R}^{\dim(M) - \dim(S)}$, we also call $\Phi$ a \udefine{defining function}.\retTwo

\ul{Lemma 6.6:} Suppose $A \subseteq M$ and that $\{(U_i, \varphi_i)\}_{i \in I}$ is a collection of smooth charts\\ covering $A$ such that $\varphi_i(A \cap U_i)$ is Lebesgue measurable with measure zero for all $i \in I$. Then $A$ has measure zero in $M$.

\ul{Lemma C:} Suppose $X$ is a second countable and regular (i.e. $T_3$) topological space. Then $X$ is normal (i.e. $T_4$).
\begin{myIndent}\pracTwo
	Proof:\\
	Let $A$ and $B$ be disjoint closed subsets of $X$. For every $x \in A$ we can use regularity to find an open set $U_x$ containing $x$ that is disjoint from $B$. Next, we can use regularity to find two disjoint open sets $U_x^\prime$ and $\widetilde{U_x^\prime}$ such that $x \in U_x^\prime$ and $(U_x)^\comp \subseteq \widetilde{U_x^\prime}$. Finally, we can use\newpage regularity one more time to find an open set $U_x^\pprime \subseteq U_x^\prime$ containing $x$. The result of doing this is that the closure of $U_x^\pprime$ is disjoint from $B$. After all, $B \subseteq (U_x)^\comp \subseteq \widetilde{U_x^\prime}$. So, $(\widetilde{U_x^\prime})^\comp$ is a closed set disjoint from $B$ that contains $U_x^\prime$. And since $U_x^\pprime \subseteq U_x^\prime$, we can conclude there is a closed set containing $U_x^\pprime$ that is disjoint from $B$. So, $\overline{U_x^\pprime} \cap B = \emptyset$.\retTwo

	By identical reasoning, for every $y \in B$ we can find an open set $V_y^\pprime$ containing $y$ such\\ that $\overline{V_y^\pprime} \cap A = \emptyset$.\retTwo

	But now since $X$ is second countable, we know there are sequences $\{x_n\}_{n \in \mathbb{N}} \subseteq A$ and $\{y_n\}_{n \in \mathbb{N}} \subseteq B$ such that $A \subseteq \bigcup_{n \in \mathbb{N}}U_{x_n}^\pprime$ and $B \subseteq \bigcup_{n \in \mathbb{N}}V_{y_n}^\pprime$.\retTwo

	To finish off, let $\widehat{U}_n \coloneqq U_{x_n} - \bigcup_{k=1}^n \overline{V_{y_k}^\pprime}$ and $\widehat{V}_n \coloneqq V_{y_n} - \bigcup_{k=1}^n \overline{U_{x_k}^\pprime}$. Then for every $n$ we have that $\widehat{U}_n$ and $\widehat{V}_n$ are open subsets of $X$. Furthermore, we still have that $A \subseteq \bigcup_{n \in \mathbb{N}}\widehat{U}_n$. After all, every $x \in A$ is contained in some $U_{x_n}$ but is not contained in any of the $\overline{V_{y_n}}$. Similarly, we have that $B \subseteq \bigcup_{n \in \mathbb{N}} \widehat{V}_n$.\retTwo

	Let $U \coloneqq \bigcup_{n \in \mathbb{N}} \widehat{U}_n$ and $V \coloneqq \bigcup_{n \in \mathbb{N}} \widehat{V}_n$. Then $U$ and $V$ are open neighborhoods of $A$ and $B$ respectively. I also claim $U$ and $V$ are disjoint. After all, suppose $x \in U \cap V$. Then $x \in \widehat{U}_j \cap \widehat{V}_k$ for some $j$ and $k$. By symmetry, we can assume without loss of generality that $j \leq k$. But now $x \in V_k^\pprime - \bigcup_{j^\prime=1}^k \overline{U_{j^\prime}^\pprime}$ and $x \in U_j$. This is a contradiction. $\blacksquare$\retTwo
\end{myIndent}

\ul{Lemma D:} Suppose $X$ is a second countable and regular (i.e. $T_3$) topological space. Then given any closed subset $C \subseteq X$ we can find a decreasing sequence of open sets $\{V_k\}_{k \in \mathbb{N}}$ in $X$ such that $\overline{V_{k+1}} \subseteq V_{k}$ for all $k \in \mathbb{N}$ and $C = \bigcap_{k \in \mathbb{N}} V_k$.
\begin{myIndent}\pracTwo
	Proof:\\
	If $C = X$ then this lemma is trivial. Just set $V_k = X$ for all $k$. So from now on we'll suppose $C$ is a proper subset of $X$.\retTwo
	
	Consider any $x \in C$. By regularity we can find open sets $U_x$ and $N_x$ such that $C \subseteq U_x$ and $x \in N_x$, and $U_x \subseteq X - N_x$. But then because $X$ is second countable, we can find a sequence $\{x_k\}_{k \in \mathbb{N}} \subseteq X - C$ such that $X - C = \bigcup_{k \in \mathbb{N}} N_{x_k}$. In turn, we have that $C = \bigcap_{k \in \mathbb{N}}(X - N_{x_k}) \supseteq \bigcap_{k \in \mathbb{N}}(U_{x_k}) \supseteq C$. And now we set $W_k \coloneqq \bigcap_{j=1}^k U_{x_j}$ for all $k \in \mathbb{N}$. By doing this we get a decreasing sequence of open sets $\{W_k\}_{k \in \mathbb{N}}$ in $X$ such that $C = \bigcap_{k \in \mathbb{N}} W_k$.\retTwo
	
	Next, we want to modify our new sets to get the desired closure property. Fortunately, by lemma C we know that $X$ is a normal (i.e. $T_4$) topological space. As a result, given any open set $U$ containing $C$ we can easily find an open set $W$ such that $C \subseteq W \subseteq \overline{W} \subseteq U$.
	\begin{myIndent}\exPPP
		Namely, by Urysohn's lemma we can find a function $f: X \to [0, 1]$ such\\ that $C \subseteq f^{-1}(\{0\})$ and $U^\comp \subseteq f^{-1}(\{1\})$ Then just set $W = f^{-1}([0, \frac{1}{2}))$.\retTwo
	\end{myIndent}

	Set $V_1 = W_1$. Then for every $k > 1$ we inductively define $V_k$ as follows.
	\begin{myIndent}\exPPP
		Let $\Omega_k$ be an open subset of $X$ such that $C \subseteq \Omega_k \subseteq \overline{\Omega_k} \subseteq V_{k-1}$. Then set\\ $V_k \coloneqq \Omega_k \cap W_k$. Thus $\overline{V_k} \subseteq \overline{\Omega_k} \subseteq V_{k-1}$. Also $V_k \subseteq W_k$. And since $C$ is\\ contained in $W_k$ and $\Omega_k$, we have that $C \subseteq V_k$.\newpage
	\end{myIndent}
	
	Because $C = \bigcap_{k \in \mathbb{N}} W_k$, and $C \subseteq V_k \subseteq W_k$ for all $k$, we can conclude that $C = \bigcap_{k \in \mathbb{N}} V_k$ still. $\blacksquare$\retTwo
\end{myIndent}

\ul{Lemma E:} Suppose $(X, \mathcal{M})$ is a measurable space, and $Y$ is a regular (i.e $T_3$) second\\ countable topological space equipped with it's Borel $\sigma$-algebra, and $\{f_n: X \to Y\}_{n \in \mathbb{N}}$\\ is a sequence of functions converging pointwise to $f:X \to Y$. If every $f_n$ is $(\mathcal{M}, \mathcalli{B}_Y)$-\\measurable then so is $f$.
\begin{myIndent}\pracTwo
	Proof:\\
	Since the closed subsets of $Y$ generate the Borel $\sigma$-algebra, it suffices to show that\\ $f^{-1}(C) \in \mathcal{M}$ for any arbitrary closed sets $C \subseteq Y$. Fortunately, by the prior lemma\\ we can find a decreasing sequence of open sets $\{V_k\}_{k \in \mathbb{N}}$ such that $C = \bigcap_{k \in \mathbb{N}} V_k$\\ and $\overline{V_{k+1}} \subseteq V_k$ for all $k \in \mathbb{N}$.\retTwo

	Next note that if $f(x) \in C$ then we know by the definition of a pointwise limit that\\ for every set $V_k$ there exists an integer $N \in \mathbb{N}$ such that $f_n(x) \in V_k$ for all $n \geq N$.\\ In other words:
	
	{\centering$f^{-1}(C) \subseteq \bigcap\limits_{k \in \mathbb{N}} \bigcup\limits_{N \in \mathbb{N}} \bigcap\limits_{n \geq N} f_n^{-1}(V_k)$\retTwo\par}

	On the other hand, if $f(x) \notin C$ then because $C = \bigcap_{k \in \mathbb{N}} \overline{V_k}$ we know that\\ $Y - C = \bigcup_{k \in \mathbb{N}}(Y - \overline{V_k})$ Hence, there exists some $k \in \mathbb{N}$ such that $f(x) \notin \overline{V_k}$.\\ Next, because $Y - \overline{V_k}$ is an open set, we know by the definition of a limit that there\\ exists an integer $N \in \mathbb{N}$ such that $f_n(x) \in X - \overline{V_k}$ for all $n \geq N$. Consequently, for\\ all $N \in \mathbb{N}$ we have that $x \notin \bigcap_{n \geq N}f_n^{-1}(V_k)$. So, $x \notin \bigcup_{N \in \mathbb{N}} (\bigcap_{n\geq N}f_n^{-1}(V_k))$. And\\ this proves that:

	{\centering$f^{-1}(C) \supseteq \bigcap\limits_{k \in \mathbb{N}} \bigcup\limits_{N \in \mathbb{N}} \bigcap\limits_{n \geq N} f_n^{-1}(V_k)$\retTwo\par}

	So, now that we know that $f^{-1}(C) = \bigcap_{k \in \mathbb{N}} (\bigcup_{N \in \mathbb{N}} (\bigcap_{n \geq N} f_n^{-1}(V_k)))$, we can conclude\\ that $f^{-1}(C) \in \mathcal{M}$. After all, $f_n$ being measurable and $V_k$ being open means that $f_n^{-1}(V_k)$ is open for all $k$ and $n$. Plus, $\mathcal{M}$ is closed under countable unions and intersections. $\blacksquare$\retTwo
\end{myIndent}

In particular, if $\mathcalli{H}$ is a separable Hilbert space, then $\mathcalli{H}$ will be second countable and\\ normal (since every metric space is normal and every separable metric space is second\\ countable). So, lemma E applies to $\mathcalli{H}$.\retTwo

\ul{Proposition:} Let $L_\mathcalli{H}$ be the space of all $(\mathcal{M}, \mathcalli{B}_\mathcalli{H})$-measurable maps from $X$ to $\mathcalli{H}$\\ (where $\mathcalli{B}_\mathcalli{H}$ is the Borel $\sigma$-algebra on $\mathcalli{H}$). Then $f: X \to \mathcalli{H}$ is in $L_\mathcalli{H}$ if and only if\\ $f_k(x) \coloneqq \langle f(x), u_k\rangle$ is $(\mathcal{M}, \mathcalli{B}_{\mathbb{C}})$-measurable for all $k \in \mathbb{N}$.
\begin{myIndent}\pracTwo
	$(\Longrightarrow)$\\
	Let $\pi_k: \mathcalli{H} \to \mathbb{C}$ be the map $\pi_k(x) = \langle x, u_k\rangle$. This is a continuous linear functional on $\mathcalli{H}$ and so it is trivially $(\mathcalli{B}_{\mathcalli{H}}, \mathcalli{B}_{\mathbb{C}})$-measurable. Also note that $f_k = \pi_k \circ f$. So if $f$ is $(\mathcal{M}, \mathcalli{B}_\mathcalli{H})$-measurable then we can conclude that $f_k$ is $(\mathcal{M}, \mathcalli{B}_{\mathbb{C}})$-measurable.\retTwo

	$(\Longleftarrow)$\\
	Because $\{u_k\}_{k \in \mathbb{N}}$ is a (complete) orthonormal basis for $\mathcalli{H}$, we know that:
	
	{\centering$f(x) = \sum_{k \in \mathbb{N}} \langle f(x), u_k\rangle u_k = \sum_{k \in \mathbb{N}} f_k(x) u_k$.\newpage\par}

	But now if $m: \mathbb{C} \times \mathcalli{H} \to \mathcalli{H}$ is the scalar multiplication function, we know that $m$ is continuous and in turn $m$ is $(\mathcalli{B}_{\mathbb{C} \times \mathcalli{H}}, \mathcalli{B}_{\mathcalli{H}})$-measurable. But by a propositon at the top of page 14 in my math 240a notes as well as the fact that $\mathbb{C}$ and $\mathcalli{H}$ are separable metric\\ spaces, we know that $\mathcalli{B}_{\mathbb{C} \times \mathcalli{H}} = \mathcalli{B}_{\mathbb{C}} \otimes \mathcalli{B}_{\mathcalli{H}}$. Also, $g_k(x) = (f_k(x), u_k)$ is going to be a $(\mathcal{M}, \mathcalli{B}_\mathbb{C} \otimes \mathcalli{B}_\mathcalli{H})$-measurable function for every $k$. So by composing functions, we get\\ that $x \mapsto f_k(x)u_k$ is a $(\mathcal{M}, \mathcalli{B}_\mathcalli{H})$-measurable function for all $k$.\retTwo

	To finish off, since finite sums of measurable functions from $X$ to $\mathcalli{H}$ are always\\ measurable, we can just apply lemma E to get that $f$ is $(\mathcal{M}, \mathcalli{B}_\mathcalli{H})$-measurable. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition:} Suppose $f \in L^1_{\mathcalli{H}} \coloneqq \{g \in L_{\mathcalli{H}} : \int \|g(x)\|\df \mu(x) < \infty\}$. Then\\ $f_k(x) \coloneqq \langle f(x), u_k\rangle \in L^1(X)$ and $\int f \df \mu = \sum_{k \in \mathbb{N}} (\int f_k \df \mu)u_k$.
\begin{myIndent}\pracTwo
	Proof:\\
	Because of the prior proposition we know that $f_k$ is measurable for all $k$. Also note\\ $|f_k(x)| \leq \|u_k\| \cdot \|f(x)\| = \|f(x)\|$. Therefore $\int |f_k|\df \mu \leq \int \|f\| \df \mu < \infty$ and\\ this proves that $f_k \in L^1(X)$ for all $k \in \mathbb{N}$.\retTwo 

	Next note that by reusing the reasoning in the prior proposition, we know that the\\ map $x \mapsto f_k(x)u_k$ is a $(\mathcal{M}, \mathcalli{B}_{\mathcalli{H}})$-measurable map for all $k$. In turn the partial sums\\ of $f(x) = \sum_{k=1}^\infty f_k(x)u_k$ are also measurable.
	Also note that:
	
	{\centering$\|\sum_{k=1}^n f_k(x)u_k\| = \left(\sum_{k = 1}^n |f_k(x)|^2\right)^{1/2} \leq \left(\sum_{k = 0}^\infty |f_k(x)|^2\right)^{1/2} = \|f(x)\|$\retTwo\par}
	
	So, $\int\|\sum_{k=1}^n f_k(x)u_k\|\df \mu \leq \int \|f(x)\|\df \mu < \infty$, meaning all our partial sums are\\ in $L^1_{\mathcalli{H}}$. And now we can use the dominated convergence theorem (with an upper bound\\ of $g(x) = \|f(x)\|$) to say that:
	
	{\centering$\int f(x)\df \mu = \lim\limits_{n \to \infty} \int \sum\limits_{k=1}^n f_k(x)u_k \df \mu = \lim\limits_{n \to \infty} \sum\limits_{k=1}^n \int f_k(x)u_k \df \mu$.\retTwo\par}
	
	By considering the continuous linear map $T: \mathbb{C} \to \mathcalli{H}$ given by $c_k \mapsto c_k u_k$, we know\\ from \inLinkRap{placeholder}{page 517} that $(\int f_k \df \mu)u_k = T(\int f_k \df \mu) = \int (T \circ f_k)\df \mu = \int f_k(x) u_k \df \mu$. So:

	{\centering$\int f(x)\df \mu = \lim\limits_{n \to \infty} \sum\limits_{k=1}^n \int f_k(x)u_k \df \mu = \lim\limits_{n \to \infty} \sum\limits_{k=1}^n (\int f_k \df \mu)u_k = \sum\limits_{k=1}^\infty (\int f_k \df \mu) u_k$. $\blacksquare$\retTwo\par}
\end{myIndent}

\exTwo\ul{Theorem (Sard Part 2):} Suppose $U \subseteq \mathbb{R}^n$ is an open set and $f: U \to \mathbb{R}^n$ is a $C^1$ map. Then the set of critical values of $f$ is a Borel null set in $\mathbb{R}^n$.
\begin{myIndent}\exThreeP
	Proof:\\
	Recall from proposition 4.1 on \inLinkRap{Proposition 4.1 pages 1054-1056}{Pages 1054-1056} that the set of regular points of $f$ is open. It follows that the set $C$ of critical points of $f$ is closed (relative to $U$). And since $f(C)$ is the set of critical values of $f$, we can conclude from lemma A that $f(C)$ is a Borel subset of $\mathbb{R}^n$.\retTwo

	Next note that since $U$ is second countable, we can cover $U$ with countably many closed\newpage coordinate cubes. In other words, for all $k \in \mathbb{N}$ we can find $x_k = (x_k^1, \ldots, x_k^n) \in U$\\ and $\varepsilon_k > 0$ such that $U = \bigcup_{k \in \mathbb{N}} (\prod_{i=1}^n [x_k^i - \varepsilon, x_k^i + \varepsilon])$. And now upon setting\\ $E_k \coloneqq \prod_{i=1}^n [x_k^i - \varepsilon, x_k^i + \varepsilon]$, it suffices to prove that $f(C \cap E_k)$ is a Borel null set\\ for all $k$.
	\begin{myIndent}\exPPP
		As a side note, $f(C \cap E_k)$ will be Borel measurable for identical reasons that $f(C)$ is Borel measurable.\retTwo
	\end{myIndent}
	
	Fortunately, note that by applying a sufficient affine transformation to the domain of\\ $f$, we can without loss of generality assume $E_k = [0, 1]^n \eqqcolon I^n$. After all, an affine\\ transformation is a $C^\infty$ diffeomorphism and will thus not effect the critical values of\\ $f$ at all. So, we are now left with the slightly simpler task of proving that if $f: I^n \to \mathbb{R}^n$\\ is a $C^1$ map and $C \subseteq I^n$ is the set of critical points of $f$ then $f(C)$ is a Borel null set.
	
	Two advantages of our simplified task though are as follows.
	\begin{enumerate}
		\item $I^n$ a compact subset of $\mathbb{R}^n$.
		\item By proposition C.29 on \inLinkRap{placeholder}{page 291}, we know $f$ is Lipschitz on $I^n$. So let $M \geq 0$ be\\ some constant such that $|f(x) - f(y)| \leq M|x - y|$ for all $x, y \in I^n$.\retTwo
	\end{enumerate}

	Next note for any $x, c \in I^n$ that:
	
	{\centering\begin{tabular}{l}
		$f(x) - f(c) - (\Df f(c))(x - c) = \int_0^1 (\Df f(c + t(x - c)))(x - c) \df t - (\Df f(c))(x - c)$\\ [4pt]
		$\phantom{f(x) - f(c) - (\Df f(c))(x - c)} = \int_0^1 [\Df f(c + t(x - c)) - \Df f(c)](x - c) \df t$.
	\end{tabular}\retTwo\par}

	In turn we have that:

	{\centering\begin{tabular}{l}
		$|f(x) - f(c) - (\Df f(c))(x - c)| \leq \left(\int_0^1 \|\Df f(c + t(x - c)) - \Df f(c)\|_{\opnorm} \df t\right)|x - c|$\\ [4pt]
		$\phantom{|f(x) - f(c) - (\Df f(c))(x - c)|} \leq \left(\sup\limits_{x^\prime \in B_{|x - c|}(c)}\|\Df f(x^\prime) - \Df f(c)\|_\opnorm\right) \cdot |x - c|$
	\end{tabular}\retTwo\par}

	Then as $f$ is a $C^1$ map, we know that $x \mapsto \Df f(x)$ is continuous as a map from $I^n$ to $B(\mathbb{R}^n)$. Next, since $I^n$ is compact we know that $x \mapsto \Df f(x)$ is uniformly continuous as a map from $I^n$ to $B(\mathbb{R}^n)$. So, for any $\varepsilon > 0$ there exists $\delta > 0$ such that $|x^\prime - c| < \delta$ implies that $\|\Df(x^\prime) - \Df(c)\|_\opnorm < \varepsilon$. So in particular, for all $\varepsilon > 0$ we there exists $\delta > 0$ such that for all $x, c \in I^n$ with $|x - c| < \delta$ we have that $|f(x) - f(c) - (\Df f(c))(x - c)| \leq \varepsilon|x - c|$.\retTwo

	Yet now suppose $c \in C$. Then $\myIm(\Df f(c))$ is contained in an $(n - 1)$-dimensional\\ hyperplane $H$. And if $|x - c| < \kappa < \delta$ then:
	
	{\centering$\inf_{x^\prime \in H}|f(x) - f(x^\prime)| \leq |f(x) - f(c) - \Df f(c)(x - c)| \leq \varepsilon \kappa$.\\ [4pt]\par}
	
	Similarly, $|f(x) - f(c)| \leq M\kappa$. So, all in all we can say that there is some $n$-dimensional parallelotope with volume $(2M \kappa)^{n-1}  \cdot (2\varepsilon \kappa) = 2^n \kappa^n M^{n-1} \cdot \varepsilon$ containing $f(B_\kappa(c) \cap I^n)$.\retTwo

	To finish off, we can subdivide $I^n$ into $h^n$ many subcubes of side length $h^{-1}$ (where $h$ is a sufficiently large integer such that $h^{-1}\sqrt{n} < \delta$). Note that the side-lengths of any subcube (say $Q$) are small enough to guarentee that if $c \in Q \cap C$ then $Q \subseteq B_{h^{-1}\sqrt{n}}(c)$. So in turn, by the prior reasoning we'd have that $f(Q) \subseteq f(B_{h^{-1}\sqrt{n}}(c) \cap I^n)$ where the latter is contained in a parallelotope with volume $2^n h^{-n} n^{n/2} M^{n-1} \cdot \varepsilon$. There are only $h^n$ many subcubes though so at most $h^n$ of them actually intersect $C$. Hence, by just taking the unions of the images of the subcubes that intersect $C$, we can guarentee that $f(C)$ is contained in a set of measure at most $h^n(2^n h^{-n} n^{n/2} M^{n-1} \cdot \varepsilon) = 2^n n^{n/2} M^{n-1} \cdot \varepsilon$.\newpage

	But $\varepsilon$ can be any positive real number. So, we just take $\varepsilon \to 0$ to get that $\mu(f(C)) = 0$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Finally, note that if $m > n$, $U \subseteq \mathbb{R}^m$ is an open set, and $f: U \to \mathbb{R}^n$ is a $C^1$ map, then Sard's theorem as was priorly stated does not hold anymore. I don't have the background to explicitely construct a counter example, but here's some vague reasoning for why we wouldn't expect Sard's theorem to hold.\retTwo

Because the Cantor set $C \subseteq \mathbb{R}$ is a null set, it wouldn't contradict the prior theorem that there exists a $C^1$ map $f: \mathbb{R} \to \mathbb{R}$ such that the set of critical values of $f$ includes $C$. (As it turns out, such a function does exist although I don't know how to construct it.)\retTwo

Next define $g: \mathbb{R}^2 \to \mathbb{R}$ by $g(x, y) = f(x) + f(y)$. Then $g$ is a $C^1$ map. But also note that $C + C$ is a subset of the critical values of $f$. And we can fairly easily conclude that $C + C = [0, 2]$.
\begin{myIndent}\pracTwo
	Why?\\
	Since $C \subseteq [0, 1]$ we know that $C + C \subseteq [0, 2]$. On the other hand, suppose $x \in [0, 2]$ and let $y = \frac{x}{2}$. We write the base $3$ representation of $y$ as $\sum_{k=1}^\infty a_k \frac{1}{3^k}$. Now if $a_k = 0$ we write $b_k = c_k = 0$. If $a_k = 1$ we write $b_k = 1$ and $c_k = 0$. And if $a_k = 2$ we write $b_k = c_k = 1$. Then upon setting $z_1 = \sum_{k=1}^\infty b_k \frac{1}{3^k}$ and $z_2 = \sum_{k=1}^\infty c_k \frac{1}{3^k}$, we have that $y = z_1 + z_2$. In turn, $x = 2z_1 + 2z_2$. Also $2z_1, 2z_2 \in C$ since all their ternary-digits are $0$ and $2$. $\blacksquare$\retTwo
\end{myIndent}

So, the set of critical values of $g$ would have positive measure.\retTwo

\begin{myIndent}\exThreeP
	Proof:\\
	We'll do induction on $m$, noting that Sard parts 1 and 2 already proved the base case of when $m = 1$. Also, by identical reasons as with Sard part 2, it suffices to assume that the domain of $f$ is actually the compact set $I^m \coloneqq [0, 1]^m$.\retTwo

Also $C \supseteq C_1 \supseteq \cdots \supseteq C_k$.\retTwo

\ul{Lemma (Another Version of Taylor's Theorem):} Suppose $U \subseteq \mathbb{R}^n$ is open and\\ $g \in C^{k + 1}(U, \mathbb{R})$. Then for any $c \in U$ and convex subset $W \subseteq U$ containing $c$,\\ we have for all $x \in W$ that:

{\centering $g(x) = \sum\limits_{|\alpha| \leq k}\frac{\partial^\alpha g(c)}{\alpha!}(x - c)^\alpha + R_k(x)$ \\ [4pt]\par}

where $R_k(x) = \sum\limits_{|\alpha| = k + 1} \frac{k+1}{\alpha!}(x - c)^\alpha \int_0^1 (1 - t)^k \partial^\alpha g(c + t(x - c))\df t$.
\begin{myIndent}\pracTwo
	Proof:\\
	Let $S_\ell \coloneqq \{1, \ldots, n\}^\ell$. Also let us write $x = (x_1, \ldots, x_n)$ and $c = (c_1, \ldots, c_n)$. Then I'm going to prove that $g(x) = \widetilde{P}_k(x) + \widetilde{R}_k(x)$ where:
	\begin{itemize}
		\item[$\circ$] \begin{tabular}{l}$\widetilde{P}_k(x) \coloneqq g(c) + \sum\limits_{\ell = 1}^k \left(\frac{1}{\ell !}\sum\limits_{(i_1, \ldots, i_\ell) \in S_\ell} \left(\prod\limits_{j=1}^\ell (x_{i_j} - c_{i_j})\right)\frac{\partial^\ell g}{\partial_{x_{i_1}}\cdots \partial_{x_{i_\ell}}}(c)\right)$,\end{tabular}
		\item[$\circ$] \begin{tabular}{l}$\widetilde{R}_k(x) \coloneqq \frac{1}{k!}\hspace{-0.2em}\sum\limits_{(i_1, \ldots, i_{k+1}) \in S_{k+1}}\hspace{-0.5em} \left(\prod\limits_{j=1}^{k+1} (x_{i_j} - c_{i_j})\right)\int_0^1(1 - t)^k \frac{\partial^{k+1}g}{\partial x_{i_1} \cdots \partial_{x_{i_{k+1}}}}(c + t(x - c))\df t$.\end{tabular}
	\end{itemize}

	Note that if $|\alpha| = \ell$ then there are $\frac{\ell!}{\alpha!}$ many orders one can differentiate in to\\ [-3pt] get $\partial^\alpha g$. So by collecting together like terms, we can see that $\widetilde{R}_k = R_k$ and $\widetilde{P}_k = P_k$.\retTwo

	To start off, note that if $k = 0$ then I've already shown this result on \inLinkRap{Page 1035 reference}{page 1035}.\\ So now we proceed by induction on $k$, supposing we've already shown that\\ $g(x) = \widetilde{P}_{k-1}(x) + \widetilde{R}_{k - 1}(x)$\retTwo

	We get via integration by parts that:

	{\centering\begin{tabular}{l}
		$\int_0^1 (1 - t)^{k-1} \frac{\partial^k g}{\partial_{x_{i_1}} \cdots \partial_{x_{i_k}}}(c + t(x - c))\df t$\\ [6pt]
		$\phantom{aa} = \left[\frac{-(1 - t)^k}{k} \frac{\partial^k g}{\partial_{x_{i_1}} \cdots \partial_{x_{i_k}}}(c + t(x - c))\right]_{t = 0}^{t = 1} + \int_0^1 \frac{(1 - t)^k}{k} \cdot \frac{\df }{\df s}[\frac{\partial^k g}{\partial_{x_{i_1}} \cdots \partial_{x_{i_k}}}(c + s(x - c))](t)\df t$\\ [6pt]
		$\phantom{aa} = \frac{1}{k} \frac{\partial^k g}{\partial_{x_{i_1}} \cdots \partial_{x_{i_k}}}(c) + \sum\limits_{j=1}^n \frac{(x_j - c_j)}{k}\int_0^1 (1 - t)^k \frac{\partial^{k+1} g}{\partial_{x_{i_1}} \cdots \partial_{x_{i_k}} \partial_{x_{j}}}(c + t(x - c))\df t$
	\end{tabular}\retTwo\par}

	And now we shift the term without the integral into being part of $\widetilde{P_k}$ and we let the term with the integral be in $\widetilde{R}_k$. $\blacksquare$\retTwo
\end{myIndent}

\ul{Corollary:} Suppose $U \subseteq \mathbb{R}^n$ is open and $g \in C^{k+1}(U, \mathbb{R})$. Given any $c \in U$ let $P_k$ and $R_k$ be from the prior lemma. Then suppose $W \subseteq U$ is a convex subset such that $c \in W$ and all $(k+1)$th partial derivatives of $g$ are bounded in absolute value on $W$ by a constant $M$. Then for all $x \in W$ we have that:

{\centering $|f(x) - P_k(x)| = |R_k(x)| \leq \frac{n^{k+1} M}{(k + 1)!}|x - c|^{k+1}$  \retTwo\par}

\begin{myIndent}\pracTwo
	Proof:\\
	Note that:\newpage
	
	{\centering\begin{tabular}{l}
		$\frac{k+1}{\alpha!}(x - c)^\alpha \int_0^1 (1 - t)^k \partial^\alpha g(c + t(x - c))\df t \leq \frac{M(k+1)}{\alpha!}|x - c|^{|\alpha|} \int_0^1 (1 - t)^k \df t$\\ [4pt]
		$\phantom{\frac{k+1}{\alpha!}(x - c)^\alpha \int_0^1 (1 - t)^k \partial^\alpha g(c + t(x - c))\df t} \leq \frac{M}{\alpha!}|x - c|^{k+1}$
	\end{tabular}\retTwo\par}

	So, we can say that $|R_k(x)| \leq M|x - c|^{k+1}\hspace{-0.4em}\sum\limits_{|\alpha| = k+1}\hspace{-0.4em}\frac{1}{\alpha !}$.\retTwo

	Also note that: $\sum\limits_{|\alpha| = k+1}\hspace{-0.4em}\frac{1}{\alpha !} = \frac{1}{(k + 1)!}\hspace{-0.4em}\sum\limits_{|\alpha| = k+1}\hspace{-0.4em}\frac{(k+1)!}{\alpha !} = \frac{n^{k+1}}{(k + 1)!}$. $\blacksquare$\retTwo
\end{myIndent}

\ul{Corollary 2:} Suppose $U \subseteq \mathbb{R}^m$ is open and $g \in C^k(U, \mathbb{R}^n)$. Also suppose $K \subseteq U$ is a compact convex subset. Let $P_k(x, c) = \sum_{|\alpha| \leq k} \frac{(x- c)^\alpha}{\alpha!}\partial^\alpha g(c)$. Then for any $\varepsilon > 0$ we can find $\delta > 0$ such that $|x - c| < \delta \implies |g(x) - P(x, c)| \leq M\varepsilon|x - c|^k$. (Here $M$ does not depend on $\varepsilon$ and $\delta$.)
\begin{myIndent}\pracTwo
	Proof:\\
	Write $g(x) = (g_1(x), \ldots, g_n(x))$. By the prior lemma, we know that:
	
	{\centering $g_i(x) - \sum\limits_{|\alpha| < k} \frac{\partial^\alpha g_i(c)}{\alpha!}(x - c)^\alpha = \sum\limits_{|\alpha| = k} \frac{k}{\alpha!}(x - c)^\alpha \int_0^1 (1 - t)^{k - 1}\partial^\alpha g_i(c + t(x - c))\df t$ \retTwo\par}

	But now note:
	
	{\centering\begin{tabular}{l}
		$\frac{k}{\alpha!}(x - c)^\alpha\int_0^1 (1 - t)^{k - 1}\partial^\alpha g_i(c + t(x - c))\df t - \frac{1}{\alpha!}(x - c)^\alpha \partial^\alpha g_i(c)$\\ [4pt]
		$\phantom{aaaaaa} = \frac{k}{\alpha!}(x - c)^\alpha\int_0^1 (1 - t)^{k - 1}\partial^\alpha g_i(c + t(x - c))\df t - \frac{k}{\alpha!}(x - c)^\alpha \int_0^1(1 - t)^{k-1}\partial^\alpha g_i(c)\df t$\\ [4pt]
		$\phantom{aaaaaa} = \frac{k}{\alpha!}(x - c)^\alpha\int_0^1 (1 - t)^{k - 1}\left[\partial^\alpha g_i(c + t(x - c)) - \partial^\alpha g_i(c)\right]\df t$\\ [4pt]
	\end{tabular}\retTwo\par}

	So by subtracting all needed terms from both sides, we can get that:

	{\centering\begin{tabular}{l}
		$g_i(x) - \sum_{|\alpha| \leq k} \frac{(x- c)^\alpha}{\alpha!}\partial^\alpha g_i(c) = \sum\limits_{|\alpha| = k}\frac{k}{\alpha!}(x - c)^\alpha\int_0^1 (1 - t)^{k - 1}\left[\partial^\alpha g_i(c + t(x - c)) - \partial^\alpha g_i(c)\right]\df t$.
	\end{tabular}\retTwo\par}
	
	And by putting all the vector components together, we get that:

	{\centering\begin{tabular}{l}
		$g(x) - P(x, c) = \sum\limits_{|\alpha| = k}\frac{k}{\alpha!}(x - c)^\alpha\int_0^1 (1 - t)^{k - 1}\left[\partial^\alpha g(c + t(x - c)) - \partial^\alpha g(c)\right]\df t$.
	\end{tabular}\retTwo\par}

	But now since $\partial^\alpha g$ is a continuous function on $K$ for all $|\alpha| = k$, we can guarentee that each of them are uniformly continuous. So, given any $\varepsilon > 0$ we can find $0 < \delta < \varepsilon$ such that $|x - c| < \delta \implies |\partial^\alpha g(c + t(x - c)) - \partial^\alpha g(c)| < \varepsilon$ for all multi-indices $\alpha$ with $|\alpha| = k$ and all $t \in [0, 1]$. And now:

	{\centering\begin{tabular}{l}
		$|g(x) - P(x, c)| \leq \sum\limits_{|\alpha| = k}\frac{k}{\alpha!}|(x - c)^\alpha| \cdot \left|\int_0^1 (1 - t)^{k - 1}\left[\partial^\alpha g(c + t(x - c)) - \partial^\alpha g(c)\right]\df t\right|$\\ [4pt]
		$\phantom{|g(x) - P(x, c)|} \leq \sum\limits_{|\alpha| = k}\frac{k}{\alpha!}|x - c|^k \cdot \int_0^1 (1 - t)^{k - 1}\left|\partial^\alpha g(c + t(x - c)) - \partial^\alpha g(c)\right|\df t$\\ [4pt]
		$\phantom{|g(x) - P(x, c)|} \leq \sum\limits_{|\alpha| = k}\frac{k}{\alpha!}|x - c|^k \cdot \varepsilon\int_0^1 (1 - t)^{k - 1}\df t = \sum\limits_{|\alpha| = k}\frac{\varepsilon}{\alpha!}|x - c|^k = \frac{m^{k} \varepsilon}{k!}|x - c|^k$\\ [4pt]
	\end{tabular}\retTwo\par}

	So upon setting $M = \frac{m^k}{k!}$ we have that $|g(x) - P(x, c)| \leq M\varepsilon|x - c|^k$.
\end{myIndent}
