\hTwo To address this short coming we instead consider derivations acting on a stalk. Recall\\ the definition of sheaves, stalks, and germs given on \inLinkRap{Page 1026 reference}{pages 1026-1027}. Then given any\\ $C^r$ manifold $M$ (with $r \in \mathbb{Z}_{\geq 1} \cup \{\infty\}$) and any $0 \leq k \leq r$, we consider the sheaf\\ $\mathcalli{F}: \mcateg{Op}(M) \to \mcateg{CAlg}_\mathbb{R}$ such that $\mathcalli{F}(U) = C^k(U, \mathbb{R})$ and for any open sets $V \subseteq U \subseteq M$\\ we have that $r^U_V(f) = f|_V$.
\begin{myTindent}\myComment
	Here $C^0(U, \mathbb{R}) \coloneqq C(U, \mathbb{R})$.\retTwo
\end{myTindent}

For any $p \in M$ we denote the stalk of $\mathcalli{F}$ at $p$ to be the set $C_p^k(M)$. Also, given any open set $U$ and $C^k$ map $f: U \to \mathbb{R}$, we typically denote their germ as either $[(U, f)]$ or $[f]$.\retTwo

Given any germ $[f] \in C^k_p(M)$, it is well-defined to evaluate $f(p)$.
\begin{myIndent}\pracTwo
	This is because if $[(U, f)] = [(V, g)]$ then there is some open set $W \subseteq U \cap V$ containing $p$ such that $f|_W = g|_W$. But in turn we know that $f(p) = g(p)$.\retTwo
\end{myIndent}

Now, we say a \udefine{derivation} on $C^k_p(M)$ is an $\mathbb{R}$-linear map $v: C^k_p(M) \to \mathbb{R}$ which also satisfies that if $[f]$ and $[g]$ are germs then $v([f][g]) = f(p)v([g]) + g(p)v([f])$. The set of all derivations on $C^k_p(M)$ is easily checked to be a real vector space and we denote that vector space by $\mathcalli{D}_p(M)$.\retTwo

% ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

\begin{myIndent}\mySepThree\end{myIndent}

Let $M$ be an $n$-dimensional $C^r$ manifold (where $r \in \mathbb{Z}_{\geq 1} \cup \{\infty\}$). Then let $U \subseteq M$ be an open set, let $p \in M$, and consider any smooth chart $(V, \varphi)$ such that $p \in V \subseteq U$. Note that $\varphi$ defines local coordinates $(x_1, \ldots, x_n)$ on $V$. I'll also write $\varphi(p) = (x_1(p), \ldots, x_n(p))$.\retTwo

\exTwo\ul{Lemma 1:} If $f \in C^k(U, \mathbb{R})$ (where $k \geq 1$) and $f(p) = 0$ then there is some open subset $W \subseteq V$ containing $p$ such that:

{\centering$f|_W = \sum\limits_{i=1}^n (x_i - x_i(p))h_i$,\\ [4pt]\par}

where $h_i \in C^{k-1}(W, \mathbb{R})$ for all $i$.
\begin{myIndent}\exThreeP
	Proof:\\
	To start off, if $\varphi(V)$ is not open in $\mathbb{R}^n$, then by definition $f \circ \varphi^{-1}$ can be extended to\\ being a $C^k$ function $\widetilde{f}$ from an open set $\widetilde{V} \subseteq \mathbb{R}^n$.\retTwo 

	Next, let $0 < r$ be a real number such that $\overline{B_r(\varphi(p))} \subseteq \widetilde{V}$. Then for any\\ $x = (x_1, \ldots, x_n) \in \overline{B_r(\varphi(p))}$ we can write:

	{\centering $\widetilde{f}(x) = \sum\limits_{i=1}^n (x_i - x_i(p))\int_0^1 \frac{\partial}{\partial x_i} \widetilde{f}(\varphi(p) + s(x - \varphi(p)))\df s$ \retTwo\par}

	By Leibniz's rule, we can conclude that $\widetilde{h}_i : B_r(\varphi(p)) \to \mathbb{R}$ given by:
	
	{\center$\widetilde{h_i}(x) = \int_0^1 \frac{\partial}{\partial x_i} \widetilde{f}(\varphi(p) + s(x - \varphi(p)))\df s$\\ [4pt]\par}
	
	is a $C^{k-1}$ map. And now we let $W = \varphi^{-1}(\varphi(V) \cap B_r(\varphi(p)))$ and $h_i: W \to \mathbb{R}$ be defined by $h_i = \widetilde{h}_i \circ \varphi$. And then each $h_i$ will be a $C^{k-1}$ map and $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Now let $(V, \varphi)$ be the smooth chart from before and suppose $f \in C^1(U, \mathbb{R})$ with $f(p) = 0$. If there exists an open set $W \subseteq U \cap V$ such that $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ where each $h_i \in C^{0}(W)$ and $h_i(\varphi(p)) = 0$ for each $i$, we say $f$ \udefine{vanishes to first order at $p$}.\retTwo

\exTwo\ul{Lemma 2:} Let $W, W^\prime \subseteq U \cap V$ both be open sets containing $p$ and then suppose\\ $h_1, \ldots, h_n \in C^{0}(W, \mathbb{R})$ and $\overline{h_1}, \ldots, \overline{h_n} \in C^{0}(W^\prime, \mathbb{R})$ satisfy that:

{\centering$f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ and $f|_{W^\prime} = \sum_{i=1}^n (x_i - x_i(p))\overline{h_i}$.\\ [4pt]\par}

Then $h_i(p) = \overline{h_i}(p)$ for all $i$.
\begin{myIndent}\exThreeP
	Proof:\\
	By considering the set $W \cap W^\prime$ we can assume without loss of generality that $W = W^\prime$. Also, we can assume without loss of generality that $W$ is an open set in $\mathbb{R}_{\geq 0}^n$, $H^n$, or $\mathbb{R}^n$ since $\overline{h_i}(p) = h_i(p) = 0$ iff $(\overline{h_i} \circ \varphi^{-1})(\varphi(p)) = (h_i \circ \varphi^{-1})(\varphi(p)) = 0$. Finally, note that\\ $f = \sum_{i=1}^n (x_i - x_i(p))h_i = \sum_{i=1}^n (x_i - x_i(p))\overline{h}_i$ means that:
	
	{\center$0 \equiv \sum_{i=1}^n (x_i - x_i(p)) \cdot (h_i - \overline{h_i})$.\retTwo\par}
	
	Thus, we will be done once we show that if $0 \equiv \sum_{i=1}^n (x_i - x_i(p)) g_i$ where\\ $g_i \in C^{0}(W, \mathbb{R})$ then $g_i(x(p)) = 0$ for all $i$ (where $x(p) = (x_1(p), \ldots, x_n(p))$).\retTwo

	Fortunately, consider any $s > 0$ such that $x(p) + se_i \in W$. (Here $e_1, \ldots, e_n$ is the standard basis for $\mathbb{R}^n$). Then $0 = \int_0^s 0\df t = \int_0^s tg_i(x(p) + te_i)\df t$. But now if $g_i(x(p)) \neq 0$ then by continuity we know there is some $s > 0$ and $\varepsilon > 0$ such that $|g_i(x(0) + te_i)| > \varepsilon$ when $0 \leq t \leq s$. In turn, we have a contradiction as $\left|\int_0^s tg_i(x(p) + te_i)\df t\right| \geq s\varepsilon$ means that $0 = \int_0^s tg_i(x(p) + te_i)\df t \neq 0$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo More specifically, what the prior lemma shows is that if $f \in C^1(U, \mathbb{R})$ with $f(p) = 0$\\ and $f|_W = \sum_{i=1}^n (x_i - x_i(p))h_i$ for some open set $W \subseteq U \cap V$ containing $p$ and\\ some $C^{0}$ maps $h_1, \ldots, h_n$ then:

{\centering $h_i(p) = \left.\frac{\partial}{\partial x_i}\right|_p f$ for each $i$.\retTwo\par}

After all, the above equation holds for the functions $h_1, \ldots, h_n$ constructed in lemma 1, and lemma 2 then says that all other possible $h_1, \ldots, h_n$ must agree with the functions constructed in lemma 1. As a result:

{\centering $f$ vanishes to first order iff $\left.\frac{\partial}{\partial x_i}\right|_p f = 0$ for all $i$ iff $v(f) = 0$ for all $v \in T_p U$\retTwo\par}

This leads naturally to another result:\retTwo


\exTwo\ul{Lemma 3:} Now suppose $(V^\prime, \psi)$ is another smooth chart containing $p$. We'll let $\psi$ define the local coordinates $(y_1, \ldots, y_n)$ on $V^\prime$. Also I'll write $\psi(p) = y(p) = (y_1(p), \ldots, y_n(p))$. If $W \subseteq U \cap V$ is an open set containing $p$ and there exists functions $h_i \in C^{0}(W, \mathbb{R})$ such that $f|_W = \sum_{i=1}^n(x_i - x_i(p))h_i$ and $h_i(p) = 0$ for each $i$, then for any open set $\Omega \subseteq U \cap V^\prime$ containing $p$ as well as functions $g_i \in C^{0}(\Omega, \mathbb{R})$ such that $f|_\Omega = \sum_{i=1}^n(y_i - y_i(p))g_i$, we know that $g_i(p) = 0$ for each $i$.\newpage
\begin{myIndent}\exThreeP
	Proof:\\
	We can conclude via lemmas 1 and 2 that $g_i(p) = \left.\frac{\partial}{\partial y_i}\right|_p f$ for each $i$. But now note from\\ [-6pt] \inLinkRap{page 1052 reference}{page 1052} that each linear operator $\left.\frac{\partial}{\partial y_i}\right|_p$ is a linear combination of the operators $\left.\frac{\partial}{\partial x_j}\right|_p$.\\ [-6pt] And since $\left.\frac{\partial}{\partial x_j}\right|_p f = h_j(p) = 0$ for each $j$ by assumption, we can conclude $g_i(p) = 0$\\ for each $i$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo As a result of the prior lemmas, we can say that $f$ vanishing to first order at $p$ is independent of our chosen local coordinates or the particular functions in our expression:

{\centering$f|_W = \sum\limits_{i=1}^n (x_i - x_i(p))h_i$.\retTwo\par}

Also as is hopefully obvious, if $[(U, f)] = [(\widetilde{U}, \widetilde{f})]$ in $C_p^1(M)$ and $f$ vanishes to first order at $p$ then so does $\widetilde{f}$. So, we can talk about a germ in $C^1_p(M)$ vanishing to first order.\retTwo

Now here's the new and improved definition of the tangent plane of $p$. For now I'll denote my old construction as $T_p M$. Meanwhile, I'll define a new construction by:

{\centering $\mathcal{T}_p M \coloneqq \{v \in \mathcalli{D}_p(M) : v([f]) = 0 \text{ if } [f] \text{ vanishes to first order}\}$. \retTwo\par}

Clearly $\mathcal{T}_p M$ is a vector subspace of $\mathcalli{D}_p(M)$.\retTwo

\exTwo\ul{Theorem:} There is a bijective linear map $\Phi: T_p M \to \mathcal{T}_p M$ such that $(\Phi(v))([f]) \coloneqq v(f)$ for all $f \in C^k(M, \mathbb{R})$.

\begin{myIndent}\exThreeP
	Proof:\\
	Suppose $[(U, f)]$ is any germ and note that by multiplying on a $C^r$ bump function, we can find a function $g \in C^1(M, \mathbb{R})$ such that $[(M, g)] = [(U, f)]$. So, for any $v \in T_p M$ we let $(\Phi(v))([f]) \coloneqq v(g)$.
	\begin{myIndent}\exPPP
		To see that $\Phi$ is well-defined, note that if $[(M, g)] = [(U, f)] = [(M, \widetilde{g})]$ then $g = f = \widetilde{g}$ on a neighborhood of $p$. So by proposition 3.8 on \inLinkRap{page 1033 reference}{page 1033} we know that $v(g) = v(\widetilde{g})$.\retTwo
	\end{myIndent}

	Showing that $\Phi(v) \in \mathcalli{D}_p(M)$ is then easy. Furthermore, if $[(U, f)]$ vanishes to first\\ order then we know for any $g \in C^k(M, \mathbb{R})$ with $[(M, g)] = [(U, f)]$ that $v(g) = 0$. So,\\ $\Phi(v) \in \mathcal{T}_p M$. Finally, it is easy to verify that $\Phi(c_1 v_1 + c_2 v_2) = c_1 \Phi(v_1) + c_2 \Phi(v_2)$\\ for all $c_1, c_2 \in \mathbb{R}$ and $v_1, v_2 \in T_p M$. So, $\Phi$ is a linear map from $T_p M$ to $\mathcal{T}_p M$.\retTwo

	To show that $\Phi$ is an isomorphism, let $(U, \varphi)$ be a smooth chart containing $p$ and defining\\ local coordiantes $\varphi(m) = x(m) = (x_1(m), \ldots, x_n(m))$. Then recall $\left.\frac{\partial}{\partial x_1}\right|_p, \ldots, \left.\frac{\partial}{\partial x_n}\right|_p$\\ [-4pt] is a basis for $T_p M$. Meanwhile, we can define a basis for $\mathcal{T}_p M$ as follows:
	\begin{myIndent}\exPPP
		Let $[(V, f)] \in C^1_p(M)$. By lemma 1 from a few pages ago can find an open subset $W \subseteq V \cap U$ as well as functions $h_i \in C^{0}(W, \mathbb{R})$ such that:
		
		{\centering$f|_W = f(p) + (x_i - x_i(p))h_i$.\newpage\par}

	Proof:\\

\exTwo\ul{Proposition 4.22:} Suppose $M$ and $N$ are $C^r$ manifolds with or without boundary, and\\ $F: M \to N$ is an injective $C^k$ immersion (where $k \leq r$). Then $F$ is a $C^k$ embedding if any of the following conditions hold:
\begin{enumerate}
	\item[(a)] $F$ is an open or closed map into $N$.
	\begin{myIndent}\exThreeP
		Why?\\
		$F$ being an open map into $N$ trivially means $F$ is an open map into the subspace topology of $F(M) \subseteq N$.\retTwo

		Meanwhile, if $F$ is a closed map then let $U \subseteq M$ be an open set. Then $M - U$ is a closed set so $F(M - U)$ is a closed subset of $N$. It follows that $N - F(M - U)$ is an open subset of $N$. And so $F(M) \cap (N - F(M - U)) = F(U)$ (which holds because $F$ is injective) is an open set in the subspace topology of $F(M) \subseteq N$.
		\begin{myIndent}\myComment
			More generally, this reasoning shows that if $X$ and $Y$ are topological spaces then any injective continuous map $f: X \to Y$ that is also open or closed is a \udefine{topological embedding} (i.e. a homeomorphism from $X$ to $f(X) \subseteq Y$).\retTwo
		\end{myIndent}
	\end{myIndent}

	\item[(b)] $F$ is a proper map.
	\begin{myIndent}\exThreeP
		Why?\\
		Let $E$ be any closed subset of $M$ and then suppose $q$ is any accumulation point of $F(E)$ in $N$. Since $N$ is an LCH space, we can pick a precompact open set $V \subseteq N$\\ [-1pt] containing $q$. And now $\overline{V} \cap F(E)$ also has $q$ as an accumulation point. Yet because\\ $F$ is proper, we know that $F^{-1}(\overline{V})$ is compact. Then since closed subsets of compact sets are also compact, we know that $E \cap F^{-1}(\overline{V})$ is compact. In turn, as $F$ is\\ continuous we have that $F(E \cap F^{-1}(\overline{V})) = F(E) \cap \overline{V}$ is compact in $N$ (and thus closed since $N$ is Hausdorff). So $F(E) \cap \overline{V}$ must contain all its accumulation points and so $q \in F(E)$.\retTwo

		Since $q$ was arbitrary, we've proven that $F(E)$ is a closed subset of $N$. So, $F$ is a closed map into $N$ and we can finish by citing part (a).
		\begin{myIndent}\myComment
			More generally, this reasoning shows that if $X$ is a topological space and $Y$ is an LCH space then any continuous proper map $f: X \to Y$ is also a closed map.\retTwo
		\end{myIndent}
	\end{myIndent}

	\item[(c)] $M$ is a compact manifold.
	\begin{myIndent}\exThreeP
		Why?\\
		$F: M \to F(N)$ is a continuous bijection, $M$ is compact, and $F(N)$ is Hausdorff. Thus $F$ is a homeomorphism.\retTwo

		Suppose $U \subseteq Y$ again satisfies that $f^{-1}(U)$ is an open subset of $X$. Then we know that $X - f^{-1}(U)$ is a closed subset of $X$. So because $f$ is a closed map, we know that $f(X - f^{-1}(U)) = f(X) - U$ is a closed subset of $Y$. Since $f$ is surjective, we in other words know that $Y - U$ is a closed set. Equivalently, $U$ is an open set. $\blacksquare$\retTwo
	\end{myIndent}
\end{itemize}

\exTwo\ul{Theorem 4.25 (Local Embedding Theorem):} Suppose $M$ and $N$ are $C^r$ manifolds where $M$ has no corners and $F: M \to N$ is a $C^k$ map. Then $F$ is a $C^k$ immersion if and only if for all $p \in M$ there exists an open set $U \subseteq M$ such that $F|_U : U \to N$ is a $C^k$ embedding.
\begin{myIndent}\exThreeP
	$(\Longleftarrow)$\\
	This direction is immediate.\retTwo

	$(\Longrightarrow)$\\
	Suppose $F$ is a $C^k$ immersion. Without loss of generality we can assume $M$ and $N$ are equipped with the unique $C^k$ structures containing their original $C^r$ structures since that would have no effect on whether $F$ is a $C^k$ immersion or $F|_U$ is a $C^k$ embedding. And this assumption is useful cause it let's us apply the rank theorem.\newpage

	Firstly we show that any $p \in M$ has a neighborhood $U$ on which $F$ is injective.
	\begin{myIndent}\exPPP
		First suppose $F(p) \notin \partial N$. Then as the set of interior points $\Int(N)$ is an open\\ set and $F$ is continuous, we know that there is an open set  $\Omega \subseteq M$ containing $p$\\ with $F(\Omega) \subseteq \Int(N)$. Then by either the rank theorem on \inLinkRap{The Rank Theorem}{page 1062} or the local\\ immersion theorem on \inLinkRap{Local Immersion Theorem boundary page 1101}{page 1101}, we know there is an open set $U_1 \subseteq M$ of $p$\\ where $F$ has the coordinate representation:
		
		{\centering$(x_1, \ldots, x_m) \mapsto (x_1, \ldots, x_m, 0, \ldots, 0)$.\retTwo\par}
		
		So $F$ is injective on $U_1$.\retTwo

		Meanwhile, suppose $F(p) \in \partial N$ and let $(W, \psi)$ be a smooth chart on $N$\\ containing $F(p)$. Then $U_0 \coloneqq F^{-1}(W)$ is an open subset of $M$ containing $p$.\\ Furthermore, let $\iota: H^n (\text{ or }\mathbb{R}_{\geq 0}^n) \hookrightarrow \mathbb{R}^n$ be the inclusion map (here $n = \dim(N)$).\\ Then $\iota \circ \psi \circ F|_{U_0}$ is a $C^k$ map from $M$ to a $C^k$ manifold without boundary.\\ Also, as $\df \iota_{a}$ and $\df \psi_{b}$ are bijections for all $a \in H^n$ and $b \in W$, we can conclude that $\iota \circ \psi \circ F|_{U_0}$ is a $C^k$-immersion still. Thus, we can apply the earlier reasoning to this map to find an open set $U_1 \subseteq U_0$ containing $p$ such that $(\iota \circ \psi \circ F|_{U_0})|_{U_1}$ is an injection. Equivalently, this implies $F|_{U_1}$ is an injection.\retTwo
	\end{myIndent}

	Now consider any $p \in M$ and let $U_1 \subseteq M$ be an open set containing $p$ on which $F$\\ is injective. Since $M$ is an LCH space, we can find a precompact open set $U$ such that\\ $p \in U \subseteq \overline{U} \subseteq U_1$. But now $F|_{\overline{U}}$ is an injective continuous function from a compact\\ set. Therefore, $F|_{\overline{U}}$ is a topological embedding. Next since restrictions of topological\\ embeddings are still topological embeddings, we know $F|_U$ is a topological embedding. And since $F|_U$ is a $C^k$ immersion, we can conclude that $F|_U: U \to N$ is a $C^k$ embedding. $\blacksquare$\retTwo
\end{myIndent}

Suppose $M$ and $N$ are $C^r$ manifolds. If $\pi: M \to N$ is a continuous map then a \udefine{section for $\pi$} is a continuous right inverse function $\sigma: N \to M$ (i.e. $\pi \circ \sigma = \myId_N$). A \udefine{local section for $\pi$} is a continuous map $\sigma: U \to M$ where $U \subseteq N$ is an open set and $\pi \circ \sigma = \myId_U$.\retTwo

\exTwo\ul{Theorem 4.26 (Local Section Theorem):} Suppose $M$ and $N$ are $C^r$ manifolds without\\ boundaries, and $\pi: M \to N$ is a $C^k$ map (where $k \leq r$). Then $\pi$ is a $C^k$ submersion\\ if and only if every $p \in M$ is in the image of a $C^k$ local section of $\pi$.
\begin{myIndent}\exThreeP
	$(\Longleftarrow)$\\
	Given any $p \in M$, let $\sigma: U \to M$ be a $C^k$ local section such that $q \in U$ and $\sigma(q) = p$. In other words, $q = \pi(\sigma(q)) = \pi(p)$. But now since $\pi \circ \sigma = \myId_U$, we can conclude that $\df \pi_p \circ \df \sigma_q = \myId_{T_q N}$. In turn, we must have that $\df \pi_p$ is surjective. And since $p$ was arbitrary, we can conclude $\pi$ is a submersion.\retTwo

	$(\Longrightarrow)$\\
	Suppose $\pi$ is a $C^k$ submersion. Then without loss of generality we can assume $M$ and $N$ are equipped with the unique $C^k$ structures containing their original $C^r$ structures. After all, this change won't effect whether $\pi$ is a $C^k$ submersion, or whether some local section of $\pi$ is $C^k$ or not. Also, this assumption lets us apply the rank theorem.\newpage

	Pick any $p \in M$ and let $q = \pi(p)$. By the rank theorem, we can choose smooth charts $(V_0, \varphi)$ on $M$ containing $p$ and $(U_0, \psi)$ containing $q$ such that $\varphi(p) = 0$, $\pi(V_0) \subseteq U_0$, and $\widehat{\pi} \coloneqq \psi \circ \pi \circ \varphi^{-1}$ has the form $\widehat{\pi}(x_1, \ldots, x_n, x_{n+1}, \ldots, x_m) = (x_1, \ldots, x_n)$.\retTwo

	For a small enough $\varepsilon > 0$ we have that the coordinate cube
	
	{\centering$C_\varepsilon \coloneqq \{(x_1, \ldots, x_m) : |x_i| < \varepsilon \text{ for all }i\}$\\ [4pt]\par}

	is an open subset of $\varphi(V_0)$ containing $0 = \varphi(p)$. In turn we have that:
	
	{\centering$C_\varepsilon^\prime \coloneqq \{(y_1, \ldots, y_n) : |y_i| < \varepsilon \text{ for all }i\}$\\ [4pt]\par}
	
	is an open subset of $\psi(U_0)$ containing $0 = \widehat{\pi}(0) = \widehat{\pi} \circ \varphi(p) = \psi(q)$ and satisfying that $\widehat{\pi}(C_\varepsilon) = C_\varepsilon^\prime$. Let $U = \psi^{-1}(C_\varepsilon^\prime)$. Then define $\iota: C_\varepsilon^\prime \to C_\varepsilon$ by:
	
	{\centering$\iota(y_1, \ldots, y_n) = (y_1, \ldots, y_n, \underbrace{0, \ldots, 0}_{m - n \text{ zeros}})$.\retTwo\par}
	
	Finally, let $\sigma = \varphi^{-1} \circ \iota \circ \psi|_U$. Then if $q^\prime \in U$ and $\varphi(q^\prime) = (y_1, \ldots, y_n)$ we have that:

	{\center\begin{tabular}{l}
		$\pi \circ \sigma(q^\prime) = \psi^{-1} \circ \widehat{\pi} \circ \varphi \circ (\varphi^{-1} \circ \iota \circ \psi)(q^\prime)$\\ [4pt]
		$\phantom{\pi \circ \sigma(q^\prime)} = \psi^{-1} \circ \widehat{\pi} \circ \iota(y_1, \ldots, y_n) = \psi^{-1} \circ \widehat{\pi}(y_1, \ldots, y_n, 0, \ldots, 0)$\\ [4pt]
		$\phantom{\pi \circ \sigma(q^\prime) = \psi^{-1} \circ \widehat{\pi} \circ \iota(y_1, \ldots, y_n)} = \psi^{-1}(y_1, \ldots, y_n) = q^\prime$\\ [4pt]
	\end{tabular}\retTwo\par}

	So $\pi \circ \sigma = \myId_U$. Also $\sigma$ is a $C^k$ map and 
	
	{\centering$\sigma(q) = \varphi^{-1} \circ \iota(0, \ldots, 0) = \varphi^{-1}(0, \ldots, 0) = p$. $\blacksquare$\retTwo\par}
\end{myIndent}

\hTwo Based on theorems 4.25 and 4.26, we also get the following purely topological definitions. Suppose $X$ and $Y$ are topological spaces and $f: X \to Y$ is a continuous function.
\begin{itemize}
	\item $f$ is a \udefine{topological immersion} if every point $x \in X$ has a neighborhood $U \subseteq X$ such that $f|_U : U \to Y$ is a topological embedding.
	\item $f$ is a \udefine{topological submersion} if every point $x \in X$ is in the image of a (continuous) local section for $f$.\retTwo
\end{itemize}

The prior two theorems say that any $C^k$ immersion / submersion is a topological\\ immersion / submersion. That said, be aware that a topological immersion or submersion that happens to also be a $C^k$ map is not necessarily a "$C^k$ immersion" or "$C^k$ submersion".
\begin{myIndent}\pracTwo
	For example, let $f: \mathbb{R} \to \mathbb{R}$ be defined by $f(t) = t^3$. Then $f$ is a $C^\infty$ map which is also a homeomorphism. And clearly homeomorphisms are trivially topological immersions and submersions. That said, as $\df f_0$ is the trivial linear map as $f^\prime(0) = 0$, we know $f$ is not a $C^\infty$ immersion or submersion.\retTwo
\end{myIndent}

\exTwo\ul{Theorem 4.27:} Let $M$ and $N$ be $C^r$ manifolds without boundaries, and suppose\\ $\pi: M \to N$ is a $C^k$ submersion. Then $\pi$ is an open map. Also consequently, if\\ $\pi$ is surjective then $\pi$ is a quotient map.
\begin{myIndent}\exThreeP
	Proof:\\
	Suppose $W \subseteq M$ is an open set and $q \in \pi(W) \subseteq N$. We can pick $p \in W$ with $\pi(p) = q$. And then by the local section theorem, we know there is an open subset $U \subseteq N$ containing $q$ as well as a $C^k$ map $\sigma: U \to M$ such that $\sigma(q) = p$ and $\pi \circ \sigma = \myId_U$. Yet now note\newpage that $\sigma^{-1}(W)$ is an open subset of $U$ (since $\sigma$ is continuous) and it contains $q$. Also note that if $y \in \sigma^{-1}(W)$ then $y = \pi(\sigma(y)) \in \pi(W)$ Therefore, we've shown that $q$ is in the topological interior of $\pi(W)$. And since $q$ was arbitrary, this proves that $\pi(W)$ is an open set. $\blacksquare$\retTwo
\end{myIndent}

\Hstatement\blab{Exercise 4-6:} Let $M$ be a nonempty compact $C^r$ manifold without a boundary, and let $N$ be a connected non-compact $C^r$ without a boundary. Then there is no $C^k$ submersion $\pi: M \to N$ (where $k \leq r$).
\begin{myIndent}\HexOne
	Proof:\\
	By the prior theorem we must have that such a function $\pi$ is an open map. Thus in particular we can conclude that $\pi(M)$ is an open subset of $N$. That said, as $M$ is compact and $\pi$ is continuous, we must also have that $\pi(M)$ is a compact subset of $N$. Since $N$ is Hausdorff, we can conclude $\pi(M)$ is a clopen set. And since $N$ is connected, we thus must have that $\pi(M)$ is either the empty set or all of $N$.\retTwo

	$\pi(M)$ can't be the empty set however because $M \neq \emptyset$. Simultaneously, since $N$ is not compact we can't have that $\pi(M) = N$. Thus we have a contradiction. $\blacksquare$\retTwo
\end{myIndent}

Suppose $M$ is a $C^r$ manifold. A \udefine{$C^r$ embedded submanifold of $M$} is a subset $S \subseteq M$ which is a topological manifold when equipped with the subspace topology and which has a $C^r$ structure such that the inclusion map $\iota: S \to M$ is a $C^r$ embedding.\retTwo

\myComment Remarks:\newpage
\begin{itemize}
	\item If $k \leq r$ then $M$ has a unique $C^k$ structure containing its $C^r$ structure. Based on this structure on $M$, if $S$ is a $C^k$ manifold then we can ask if $\iota: S \to M$ is a $C^k$ embedding. And if so we call $S$ a $C^k$ embedded submanifold of $M$.
	\item Some authors also refer to embedded submanifolds as \udefine{regular submanifolds}.\retTwo
\end{itemize}

\hTwo If $M$ is a $C^r$ manifold and $S \subseteq M$ is a $C^k$ embedded submanifold of $M$ then the\\ \udefine{codimension of $S$ in $M$} is defined to be the difference $\dim(M) - \dim(S)$. Here $M$ is\\ called the \udefine{ambient manifold for $S$}. If $S$ has codimension $1$ we say $S$ is an \udefine{embedded\\ hypersurface}.\retTwo

\exTwo\ul{Proposition 5.1:} Suppose $M$ is a $C^r$ manifold without a boundary. Then the embedded $C^k$\\ submanifolds of codimension $0$ (without boundaries) in $M$ are precisely the open\\ submanifolds of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	If $U \subseteq M$ is an open submanfold then we know $\dim(U) = \dim(M)$. Furthermore, $U$ is equipped with the subspace topology and we know from before that the inclusion map $\iota: U \hookrightarrow M$ is a $C^k$ immersion and submersion. It follows that $\iota$ is a $C^k$ embedding. And as $M$ has no boundary, neither does $U$.\retTwo

	Conversely, suppose $U \subseteq M$ is an embedded $C^k$ submanifold in $M$ of codimension $0$. By definition the inclusion map $\iota: U \hookrightarrow M$ is a $C^k$ embedding. Yet as $\dim(M) = \dim(U)$ and $\iota$ is an immersion, we know from proposition 4.8(b) on \inLinkRap{Proposition 4.8 page 1061}{page 1061} that $\iota$ is a local $C^k$-diffeomorphism. In turn, by proposition 4.6(c) on \inLinkRap{Properties of local $C^k$-diffeomorphisms page 1060}{page 1060} we know $\iota$ is an open map. So, $\iota(U) = U$ is an open subset of $M$.\retTwo

	Finally, suppose $(V, \varphi)$ is a smooth chart on $M$ (equipped with its unique $C^k$ structure). Then $U \cap V$ is an open subset of $U$ and I claim that $(U \cap V, \varphi)$ is a smooth map on $U$. After all, suppose $(W, \psi)$ is any other smooth chart on $U$ as an embedded submanifold. Then as $\iota: U \hookrightarrow M$ is a $C^k$ map we know that $\varphi \circ \iota \circ \psi^{-1} = \varphi \circ \psi^{-1}$ is $C^k$ on $\psi((U \cap V) \cap W)$.\retTwo

	Meanwhile, as $\iota$ is a local $C^k$-diffeomorphism, we know for every $p \in (U \cap V) \cap W$ that there is some open set $\Omega \subseteq U$ containing $p$ such that $\iota$ is a $C^k$-diffeomorphism from $\Omega$ (as an open submanifold of $U$) to $\Omega$ (as an open submanifold of $M$). In turn, $\psi \circ \varphi^{-1}$ is a $C^k$ map on $\psi((U \cap V) \cap W \cap \Omega)$. So, $\psi \circ \varphi^{-1}$ is $C^k$ at every $p \in (U \cap V) \cap W$. And with that, we've proven that the $C^k$ structure on $U$ as an embedded submanifold must be the same as the $C^k$ structure on $U$ as an open submanifold. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition 5.2:} Suppose $M$ and $N$ are $C^r$ manifolds, and $F: N \to M$ is a $C^r$ embedding. Let $S = F(N)$. With the subspace topology $S$ is a topological manifold. Also $S$ has a unique $C^r$ structure making it into an embedded submanifold of $M$ and satisfying that $F$ is a $C^r$-diffeomorphism from $N$ to $S$.
\begin{myIndent}\exThreeP
	Proof:\\
	$S$ being a subspace of $M$ and $M$ being second countable and Hausdorff means that $S$ is also second countable and Hausdorff. Also by assumption $F$ is a homeomorphism from $N$ to $S$. Therefore, $N$ being locally Euclidean of dimension $\dim(N)$ means that $S$ is as well.\newpage

	Now we equip $S$ with the $C^r$ structure given by the charts $(F(U), \varphi \circ F^{-1})$ where $(U, \varphi)$ is any smooth chart on $N$. Them being $C^r$ compatible with each other is a result of the smooth charts on $N$ being $C^r$ compatible with each other. So, $S$ is a $C^r$ manifold. It's also easy to see now that $F$ is a $C^r$-diffeomorphism. After all, for any $p \in S$ we can pick a smooth chart $(U, \varphi)$ on $N$ containing $F^{-1}(p)$. Then $(F(U), \varphi \circ F^{-1})$ is a smooth chart on $S$ containing $p$, and:

	{\centering $\varphi \circ F \circ (\varphi \circ F^{-1})^{-1} = \myId_{\varphi(U)}$ is a $C^r$ map.\\ [4pt]\par}

	So, $F^{-1}: S \to N$ is a $C^r$ map.\retTwo

	Next, to show uniqueness just note that if $F$ is to be a diffeomorphism from $N$ to $S$ then for any smooth chart $(U, \varphi)$ on $N$ we must have that $(F(U), \varphi \circ F^{-1}$) is a smooth chart on $S$. Similarly, for every smooth chart $(V, \psi)$ on $S$ we must have that $(F^{-1}(V), \psi \circ F)$ is a smooth chart on $N$. So the atlas described in the prior paragraph is maximal and $F$ being a $C^r$-diffeomorphism from $N$ to $S$ automatically implies that $S$ is equipped with that maximal structure.\retTwo
	
	Finally, I need to show that the map $\iota: S \to M$ is a $C^r$-embedding. Fortunately, consider that $\iota$ is given by the composition:

	\begin{myIndent}\myComment

\ul{Proposition 5.3:} Suppose $M$ and $N$ are $C^r$ manifolds. For each $p \in N$ the subset\\ $M \times \{p\} \subseteq M \times N$ (called a \udefine{slice} of the product manifold) is an embedded submanifold of $M \times N$ diffeomorphic to $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	The map $F: M \to M \times N$ given by $F(m) \coloneqq (m, p)$ is a $C^r$ embedding. After all, it's a $C^r$ map that is a topological embedding. Furthermore, recall from \inLinkRap{Proposition 3.14 page 1059}{page 1059} how we standardly identify $T_{(m, p)}(M \times N)$ with $T_m M \oplus T_p N$.\retTwo

	\ul{Lemma:} Suppose $N, M_1, \ldots, M_k$ are $C^r$ manifolds. Then let $M \coloneqq \prod_{i=1}^k M_i$ and\\ for each $i$ let $\pi_i: M \to M_i$ be the projection map. Given any $(p_1, \ldots, p_k) \in M$ we\\ identify $T_{(p_1, \ldots, p_k)}M$ with $\bigoplus_{i=1}^k T_{p_i}M_i$ via the standard identification. Now suppose for\\ each $1 \leq i \leq k$ that $F_i: N \to M_i$ is a $C^1$ map. Then let $F: N \to M$ be defined\\ by $F(q) = (F_1(q), \ldots, F_k(q))$. Now for any $q \in N$ we have that:
	
	{\centering$\df F_q(v) = (\df (F_1)_q (v),\phantom{.} \ldots,\phantom{.} \df (F_k)_q (v))$.\\ [4pt]\par}
	\begin{myIndent}\exPPP
		Why?\\
		Suppose $v \in T_q N$ and $f: M \to \mathbb{R}$ is a $C^1$ map. Then $\df F_q(v) f$ is identified with:
		
		{\centering\begin{tabular}{l}
			$(\df (\pi_1)_{F(q)} \circ \df F_q(v), \ldots \df (\pi_k)_{F(q)} \circ \df F_q(v)) = (\df (\pi_1 \circ F)_q (v), \ldots, \df (\pi_k \circ F)_q (v))$\\ [4pt]
			$\phantom{(\df (\pi_1)_{F(q)} \circ \df F_q(v), \ldots \df (\pi_k)_{F(q)} \circ \df F_q(v))} = (\df (F_1)_q (v),\phantom{.} \ldots,\phantom{.} \df (F_k)_q (v))$
		\end{tabular}\retTwo\par}
	\end{myIndent}

	Based on the prior lemma, we can see for any $v \in T_m M$ that:
	
	{\centering$\df F_m(v) = (v, 0) \in T_m M \oplus T_p N$.\retTwo\par}

	It follows that $\df F_m$ is injective for all $m \in M$. So, $F$ is an immersion. $\blacksquare$\retTwo
\end{myIndent}

\exTwo\ul{Proposition 5.4:} Suppose $M$ is a $C^r$ $m$-dimensional manifold, $N$ is a $C^r$ $n$-dimensional manifold, $U \subseteq M$ is open, and $f: U \to N$ is a $C^k$ map. Let $\Gamma(f) \subseteq M \times N$ denote the graph of $f$:

{\centering $\Gamma(f) \coloneqq \{(m, f(M)) \in M \times N : m \in U\}$ \\ [4pt]\par}

Then $\Gamma(f)$ is a $C^k$ $m$-dimensional embedded submanifold of $M \times N$.
\begin{myIndent}\exThreeP
	Proof:\\
	Define a map $\gamma_f: U \to M \times N$ by $\gamma_f(m) = (m, f(m))$. This will be a $C^k$ map because the inclusion map $\iota: U \to M$ is $C^r$ and similarly the map $f: U \to N$ is $C^k$. Also let $\pi_M: M \times N \to M$ be the projection map. Then $\myId_{U} = \pi_M \circ \gamma_f$ implies that $\gamma_f$ injective. Also for any $m \in U$ we have $\df (\pi_M)_{(m, f(m))} \circ \df (\gamma_{f})_m = \df (\myId_{U})_m = \myId_{T_m U}$. So, for every $m \in U$ we have that $\df (\gamma_{f})_m$ is injective. In other words, $\gamma_{f}$ is an immersion. Finally, $\gamma_f$ is a topological embedding since its inverse function is given by $\pi_M |_{\Gamma(f)}$.\retTwo
	
	And now we've shown that $\Gamma(f)$ is the image of a $C^k$ embedding $\gamma_f$ from $U$. $\blacksquare$\retTwo
\end{myIndent}

\hTwo An embedded submanifold $S \subseteq M$ is said to be \udefine{properly embedded} if the inclusion map $S \hookrightarrow M$ is a proper map.\retTwo

\exTwo\ul{Proposition 5.5:} Suppose $M$ is a $C^r$ manifold and $S \subseteq M$ is an embedded submanifold. $S$ is properly embedded if and only if $S$ is a closed subset of $M$.
\begin{myIndent}\exThreeP
	Proof:\\
	Suppose the inclusion map $\iota: S \to M$ is a proper map. Then we know that it is a closed map (see \inLinkRap{Page 1104 reference}{page 1104}). In turn, $S = \iota(S)$ must be a closed subset of $M$.\retTwo

	Conversely, note that $\iota$ being a topological embedding with a closed image implies that $\iota$ is proper.
	\begin{myIndent}\exPPP
		This is because if $K \subseteq M$ is a compact set, then $S \cap K$ will also be compact since $S$ is closed. And now as $\iota^{-1}$ is a continuous map, we know that $\iota^{-1}(K) = \iota^{-1}(S \cap K)$ is compact. So, $\iota$ is proper. \exThreeP $\blacksquare$\retTwo
	\end{myIndent}
\end{myIndent}

\ul{Corollary 5.6:} Every compact embedded submanifold is properly embedded.
\begin{myIndent}\exThreeP
	Proof:\\
	Compact subsets of Haudorff spaces are closed. $\blacksquare$\retTwo
\end{myIndent}

\ul{Proposition 5.7:} Suppose $M$ and $N$ are $C^r$ manifolds, and $f: M \to N$ is a $C^k$ map. Then after equipping $\Gamma(f) \coloneqq \{(m, f(M)) \in M \times N : m \in M\}$ with the smooth structure defined in proposition 5.4, we have that $\Gamma(f)$ is properly embedded.\newpage
\begin{myIndent}\exThreeP
	Proof:\\
	Note that $\gamma_f : M \to M \times N$ given by $\gamma_f(m) = (m, f(m))$ has the map\\ $\pi_M : M \times N \to M$ given by $\pi_M(m, n) = m$ as a continuous left inverse. In other words, $\pi_M \circ \gamma_f = \myId_M$. As a result we can conclude that $\gamma_f : M \to M \times N$ is a proper map. To see why, I claim the following:\retTwo

	\ul{Lemma:} If $X, Y$ are topological spaces and $g: X \to Y$ and $h: Y \to X$ are continuous functions such that $h \circ g = \myId_X$, then $g$ must be proper.
	\begin{myIndent}\exPPP
		Why?\\
		Suppose $K \subseteq Y$ is a compact set. Since $h$ is continuous, we know that $h(K) \subseteq X$ is a compact set. But also let $F \coloneqq g^{-1}(K)$. Then:
		
		{\centering\begin{tabular}{l}
			$F = \myId_X(F) = (h \circ g)(F) = h(K \cap g(X))$\\ [4pt]
			$\phantom{F = \myId_X(F) = (h \circ g)(F)} = h(K) \cap (h \circ g)(X)$\\ [4pt]
			$\phantom{F = \myId_X(F) = (h \circ g)(F)} = h(K) \cap \myId_X(X) = h(K) \cap X = h(K)$.
		\end{tabular}\retTwo\par}

		So, we've shown that $g^{-1}(K)$ is compact.\retTwo
	\end{myIndent}

	Since $\gamma_f$ is a continuous proper map though and $M \times N$ is an LCH space, we can conclude that $\gamma_f$ is a closed map. So $\gamma_f(M) = \Gamma(f)$ is a closed subset of $M \times N$. And by proposition 5.5, we are done. $\blacksquare$\retTwo
\end{myIndent}
