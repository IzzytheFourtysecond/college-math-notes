Unfortunately, there is still a little more infrastructure that needs to be built up before I can get to constructing interesting $R$-algebras satisfying certain universal properties.\retTwo

\exTwo\ul{Theorem 2.1:} Let $M_1, \ldots, M_p$ be $R$-modules (where $p > 2$) and suppose $1 \leq q < p$. Then there is an $R$-module isomorphism from $(\bigotimes_{i=1}^q M_i) \otimes (\bigotimes_{i=q + 1}^p M_i)$ to $\bigotimes_{i=1}^p M_i$ such that $(m_1 \otimes \cdots \otimes m_q) \otimes (m_{q +1} \otimes \cdots \otimes m_p) \mapsto m_1 \otimes \cdots \otimes m_p$ for all $m_i \in M_i$.
\begin{myIndent}\exThreeP
	Proof:\\
	I shall proceed by induction on $p$. For our base case of $p = 3$, recall from \inLinkRap{Page 1067 Lemma A}{pages 1067-1069} that my original construction of $M_1 \otimes M_2 \otimes M_3$ was the set $M_1 \otimes (M_2 \otimes M_3)$ where $m_1 \otimes m_2 \otimes m_3$ is equal to $m_1 \otimes (m_2 \otimes m_3)$. Meanwhile, I showed on \inLinkRap{placeholder}{pages 730-731} that $M_1 \otimes (M_2 \otimes M_3)$ is isomorphic to $(M_1 \otimes M_2) \otimes M_3$ via an isomorphism mapping $m_1 \otimes (m_2 \otimes m_3)$ to $(m_1 \otimes m_2) \otimes m_3$. Hence, our base case is entirely dealt with via prior results.\retTwo

	Next suppose $p > 3$.
	\begin{itemize}
		\item If $q = 1$ then recall that $\bigotimes_{i=2}^p M_i$ is isomorphic to $M_2 \otimes (\cdots \otimes (M_{p-1} \otimes M_p )\cdots)$ via an isomorphism $\phi$ such that:
		
		{\centering\begin{tabular}{l}
			$m_2 \otimes \cdots \otimes m_{p-1} \otimes m_p \mapsto m_2 \otimes (\cdots \otimes(m_{p-1} \otimes m_p)\cdots)$.
		\end{tabular}\retTwo\par}

		In turn, $\myId_{M_1} \otimes \phi$ is an isomorphism from $M_1 \otimes (\bigotimes_{i=2}^p M_i)$ to\\ $M_1 \otimes (M_2 \otimes (\cdots \otimes (M_{p-1} \otimes M_p )\cdots))$ such that:
		
		{\centering$m_1 \otimes (m_2 \otimes \cdots \otimes m_p)\mapsto m_1 \otimes (m_2 \otimes (\cdots \otimes(m_{p-1} \otimes m_p)\cdots))$.\retTwo\par}

		And via the reasoning on \inLinkRap{placeholder}{pages 1067-1069}, we can compose that map with an\\ isomorphism $\psi: M_1 \otimes (M_2 \otimes (\cdots \otimes (M_{p-1} \otimes M_p )\cdots)) \to \bigotimes_{i=1}^p M_i$\\ sending $m_1 \otimes (m_2 \otimes (\cdots \otimes(m_{p-1} \otimes m_p)\cdots))$ to $m_1 \otimes \cdots \otimes m_p$.\retTwo 

		\item Meanwhile, if $q > 1$ then we can conclude by our induction hypothesis that $\bigotimes_{i=1}^q M_i$ is isomorphic to $M_1 \otimes \bigotimes_{i=2}^q M_i$ via an isomorphism $\phi$ mapping $m_1 \otimes \cdots \otimes m_q$ to $m_1 \otimes (m_2 \otimes \cdots \otimes m_q)$.\retTwo
		
		In turn, $\phi \otimes \myId_{\bigotimes_{i=q+1}^p M_i}$ is an isomorphism from $(\bigotimes_{i=1}^q M_i) \otimes (\bigotimes_{i=q+1}^p M_i)$ to\\ $(M_1 \otimes (\bigotimes_{i=2}^q M_i)) \otimes (\bigotimes_{i=q+1}^p M_i)$ such that $(m_1 \otimes \cdots \otimes m_q) \otimes (m_{q+1} \otimes \cdots \otimes m_p)$ is mapped to $(m_1 \otimes (m_2 \otimes \cdots \otimes m_q)) \otimes (m_{q+1} \otimes \cdots \otimes m_p)$.\retTwo

		Next, by the exercise on \inLinkRap{placeholder}{pages 730-731} we have an isomorphism from\\ $(M_1 \otimes (\bigotimes_{i=2}^q M_i)) \otimes (\bigotimes_{i=q+1}^p M_i)$ to $M_1 \otimes ((\bigotimes_{i=2}^q M_i) \otimes (\bigotimes_{i=q+1}^p M_i))$\\ sending $(m_1 \otimes (m_2 \otimes \cdots \otimes m_q)) \otimes (m_{q+1} \otimes \cdots \otimes m_p)$ to\\ $m_1 \otimes ((m_2 \otimes \cdots \otimes m_q) \otimes (m_{q+1} \otimes \cdots \otimes m_p))$.\retTwo

		Then by our induction hypothesis, $(\bigotimes_{i=2}^q M_i) \otimes (\bigotimes_{i=q+1}^p M_i)$ is isomorphic to\\ $\bigotimes_{i=2}^p M_i$ via an isomorphism $\psi$ mapping $(m_2 \otimes \cdots \otimes m_q) \otimes (m_{q+1} \otimes \cdots \otimes m_p)$ to\\ $m_2 \otimes \cdots \otimes m_p$.\retTwo
		
		In turn, $\myId_{M_1} \otimes \psi$ is an isomorphism from $M_1 \otimes ((\bigotimes_{i=2}^q M_i) \otimes (\bigotimes_{i=q+1}^p M_i))$ to $M_1 \otimes (\bigotimes_{i=2}^p M_i)$ sending $m_1 \otimes ((m_2 \otimes \cdots \otimes m_q) \otimes (m_{q+1} \otimes \cdots \otimes m_p))$ to\\ $m_1 \otimes (m_2 \otimes \cdots \otimes m_p)$.\newpage

		Finally, we can now use our reasoning for the case when $q = 1$ to finish up. $\blacksquare$\retTwo
	\end{itemize}
\end{myIndent}

\hTwo Next let $A_1, \ldots, A_p$ be $R$-algebras. I already mentioned how $\bigotimes_{i=1}^p A_i$ can naturally be turned into an $R$-algebra itself. However, here is another interpretation of this natural product operation on $\bigotimes_{i=1}^p A_i$.\\ [-8pt]
\begin{myIndent}\pracTwo
	Consider the $2p$-linear map $T: A_1 \times \cdots \times A_p \times A_1 \times \cdots \times A_p \to \bigotimes_{i=1}^p A_i$ given by:
	
	{\centering$T(a_1, \ldots, a_p, a_1^\prime, \ldots, a_p^\prime) \coloneqq (a_1a_1^\prime, \ldots, a_p a_p^\prime)$.\retTwo\par}

	By the universal property of tensor products we thus know there is a unique $R$-module homomorphism $\phi: A_1 \otimes \cdots \otimes A_p \otimes A_1 \otimes \cdots \otimes A_p \to \bigotimes_{i=1}^p A_i$ such that:

	{\centering$\phi(a_1 \otimes \cdots \otimes a_p \otimes a_1^\prime \otimes \cdots \otimes a_p^\prime) \coloneqq (a_1a_1^\prime, \ldots, a_p a_p^\prime)$.\retTwo\par}

	Then by the prior theorem, we can also say that there is a unique $R$-module\\ homomorphism $\widetilde{\phi}: (\bigotimes_{i=1}^p A_i) \otimes (\bigotimes_{i=1}^p A_i) \to \bigotimes_{i=1}^p A_i$ such that:

	{\centering$\widetilde{\phi}((a_1 \otimes \cdots \otimes a_p) \otimes (a_1^\prime \otimes \cdots \otimes a_p^\prime)) \coloneqq (a_1a_1^\prime, \ldots, a_p a_p^\prime)$.\retTwo\par}

	And finally, we can conclude that there is a unique bilinear map\\ $\mu: (\bigotimes_{i=1}^p A_i) \times (\bigotimes_{i=1}^p A_i) \to \bigotimes_{i=1}^p A_i$ such that:
	
	{\centering$\mu((a_1 \otimes \cdots \otimes a_p), (a_1^\prime \otimes \cdots \otimes a_p^\prime)) \coloneqq (a_1a_1^\prime, \ldots, a_p a_p^\prime)$.\retTwo\par}

	\ul{Lemma:} \hypertarget{Lemma middle of page 1081}{Suppose} $M$ is an $R$-module generated by a set $B$, and $\mu: M \times M \to M$ is a bilinear map such that $\mu(\mu(b_1, b_2), b_3) = \mu(b_1, \mu(b_2, b_3))$ for all $b_1, b_2, b_3 \in B$. Then $\mu(\mu(m_1, m_2), m_3) = \mu(m_1, \mu(m_2, m_3))$ for all $m_1, m_2, m_3 \in M$.
	\begin{myIndent}\exPPP
		Proof:\\
		Write $m_1 = \sum_{i=1}^n c_i b_i$,\hspace{0.1em} $m_2 = \sum_{j=1}^m c_j^\prime b_j^\prime$,\hspace{0.1em} and $m_3 = \sum_{k=1}^\ell c_k^\pprime b_k^\pprime$. Then:
		\begin{itemize}
			\item $\mu(\mu(m_1, m_2), m_3) = \sum_{i=1}^n \sum_{j=1}^m \sum_{k=1}^\ell c_i c_j^\prime c_k^\pprime \mu(\mu(b_i, b_j^\prime), b_k^\pprime)$,
			\item $\mu(m_1, \mu(m_2, m_3)) = \sum_{i=1}^n \sum_{j=1}^m \sum_{k=1}^\ell c_i c_j^\prime c_k^\pprime \mu(b_i, \mu(b_j^\prime, b_k^\pprime))$.\retTwo
		\end{itemize}

		And now we just invoke that $\mu(\mu(b_i, b_j^\prime), b_k^\pprime) = \mu(b_i, \mu(b_j^\prime, b_k^\pprime))$. $\blacksquare$\retTwo
	\end{myIndent}

	Based on that lemma, its now clear that $\mu$ defines an associative operation on $M$. Clearly, $\mu$ is the standard multiplication map on $\bigotimes_{i=1}^p A_i$.\retTwo
\end{myIndent}

\exTwo\ul{Lemma:} Suppose $A_1, A_2$ are $R$-algebras, $A_1$ is generated as an $R$-module by a subset $B$, $f: A_1 \to A_2$ is an $R$-module homomorphism, and $f(b b^\prime) = f(b)f(b^\prime)$ for all $b, b^\prime \in B$. Then $f$ is an $R$-algebra homomorphism.
\begin{myIndent}\exThreeP
	Proof:\\
	We need to show for any $a, a^\prime \in A_1$ that $f(aa^\prime) = f(a)f(a^\prime)$. Fortunately, we can write $a = \sum_{i=1}^n c_i b_i$ and $a^\prime = \sum_{j=1}^m c_j^\prime b_j^\prime$. Then:

	{\centering \begin{tabular}{l}
		$f(aa^\prime) = f((\sum_{i=1}^n c_i b_i)(\sum_{j=1}^m c_j^\prime b_j^\prime)) = \sum_{i=1}^n \sum_{j=1}^m c_i c_j^\prime f(b_i b_j^\prime)$\\ [4pt]
		$\phantom{f(aa^\prime) = f((\sum_{i=1}^n c_i b_i)(\sum_{j=1}^m c_j^\prime b_j^\prime))} = \sum_{i=1}^n \sum_{j=1}^m c_i c_j^\prime f(b_i)f(b_j^\prime)$\\ [4pt]
		$\phantom{f(aa^\prime) = f((\sum_{i=1}^n c_i b_i)(\sum_{j=1}^m c_j^\prime b_j^\prime))} = \left(\sum_{i=1}^n c_i f(b_i)\vphantom{|_j^j}\right)\left(\sum_{j=1}^m c_j^\prime f(b_j^\prime)\right)$\\ [8pt]
		$\phantom{f(aa^\prime) = f((\sum_{i=1}^n c_i b_i)(\sum_{j=1}^m c_j^\prime b_j^\prime))} = f(\sum_{i=1}^n c_i b_i)f(\sum_{j=1}^m c_j^\prime b_j^\prime) = f(a)f(a^\prime)$. $\blacksquare$
	\end{tabular}\newpage\par}
\end{myIndent}

\exTwo\ul{Theorem 3.2:} Let $A_1, \ldots, A_p$ be $R$-algebras (where $p > 2$) and suppose $1 \leq q < p$. Then the $R$-module isomorphism $f : (\bigotimes_{i=1}^q A_i) \otimes (\bigotimes_{i=q+1}^p A_i) \to \bigotimes_{i=1}^p A_i$ given by theorem 2.1 is also an $R$-algebra homomorphism.

\begin{myIndent}\exThreeP
	Why?\\
	Just apply the prior lemma where:
	
	{\centering$B = \{(a_1 \otimes \cdots \otimes a_q) \otimes (a_{q + 1} \otimes \cdots \otimes a_p) : a_1, \ldots, a_p \in A\}$.\retTwo\par}
\end{myIndent}

\exTwo\ul{Theorem 3.4:} Let $A$ be an $R$-algebra and note that $R$ itself is trivially an $R$-algebra. Thus, we can consider the standard $R$-algebra structure on $R \otimes A$.  I claim that the $R$-module isomorphism $\phi: R \otimes A \to A$ satisfying that $\phi(c \otimes a) = ca$ is also an $R$-algebra\\ homomorphism.
\begin{myIndent}\exThreeP
	Proof:\\
	Let $B = \{(c \otimes a) : c \in R \text{ and } a \in A\}$. Then:
	
	{\centering$\phi((c \otimes a) \cdot (c^\prime \otimes a^\prime)) = \phi(cc^\prime \otimes aa^\prime) = cc^\prime aa^\prime = ca c^\prime a^\prime = \phi(c \otimes a) \phi(c^\prime \otimes a^\prime)$.\retTwo\par}

	So, we can apply the lemma at the bottom of the prior page. $\blacksquare$\retTwo
\end{myIndent}

\hTwo Obviously a similar thing holds if we replace $R \otimes A$ with $A \otimes R$.\retTwo

Let $M$ be an $R$-module. Then suppose $A$ is an $R$-algebra and $\phi: M \to A$ is an $R$-\\module homomorphism. If $B$ is a second $R$-algebra and $h: A \to B$ is a unital $R$-algebra\\ homomorphism, we know that $h \circ \phi$ is an $R$-module homomorphism from $M$ to $B$.\\ This brings up the following question:\\ [-6pt]
\begin{myIndent}
	Does there exist a unital $R$-algebra $A$ and an $R$-module homomorphism $\phi: M \to A$\\ such that for any unital $R$-algebra $B$ and any $R$-module homomorphism $f: M \to B$\\ there exists a unique unital $R$-algebra homomorphism $h: A \to B$ with $h \circ \phi = f$?

		\arrow["\phi", from=2-1, to=1-2]

	\end{tikzcd}\]

	If such an $R$-algebra $A$ and $R$-module homomorphism $\phi$ exists then we say $A$\\ along with $\phi$ is the \udefine{tensor algebra of $M$}. Furthermore, if $(A_1, \phi_1)$ and $(A_2, \phi_2)$ both satisfy the above universal property then it is easy to see that there exists inverse $R$-algebra isomorphisms $f: A_1 \to A_2$ and $g: A_2 \to A_1$ such that $f \circ \phi_1 = \phi_2$ and $g \circ \phi_2 = \phi_1$. So, assuming a model of the tensor algebra of $M$ exists, it is unique up to isomorphism.\newpage
\end{myIndent}

\exTwo\ul{Theorem 4.1:} Such a unital $R$-algebra $A$ and $R$-module homomorphism $\phi: M \to A$ exists. Furthermore:
\begin{itemize}
	\item[(a)] $\phi: M \to A$ is an injection.
	\item[(b)] $A$ is generated as an $R$-algebra by $\phi(M)$.
	\item[(c)] There is a grading on $\{A_n\}_{n \in \mathbb{Z}}$ on $A$ such that $A_1 = \phi(M)$.
	\item[(d)] The grading $\{A_n\}_{n \in \mathbb{Z}}$ is nonnegative and $A_0 = R1_A$.
	\item[(e)] For $p \geq 1$, $T_p(M)$ is isomorphic to $A_p$ via the unique $R$-module homomorphism\\ sending $m_1 \otimes \cdots \otimes m_p$ to $\phi(m_1)\cdots \phi(m_p)$.
\end{itemize}

\begin{myIndent}\exThreeP
	Proof:\\

Clearly $A$ has an $R$-module structure. But we still need to define a multiplication operation on $A$ such that $A$ becomes an algebra.\retTwo

	Fortunately for any $p, q \in \mathbb{Z}_{\geq 1}$ we know by theorem 2.1 that there is a (unique) $R$-\\module homomorphism $\phi_{p,q}: T_p(M) \otimes T_q(M) \to T_{p + q}(M)$ such that:
	
	{\centering$\phi_{p, q}((m_1 \otimes \cdots \otimes m_p) \otimes (m_1^\prime \otimes \cdots \otimes m_q^\prime)) = m_1 \otimes \cdots \otimes m_p \otimes m_1^\prime \otimes \cdots \otimes m_q^\prime$\retTwo\par}

	In turn, we know there is a (unique) bilinear map $\mu_{p, q} : T_p(M) \times T_q(M) \to T_{p + q}(M)$\\ such that:
	
	{\centering$\mu_{p, q}((m_1 \otimes \cdots \otimes m_p), (m_1^\prime \otimes \cdots \otimes m_q^\prime)) = m_1 \otimes \cdots \otimes m_p \otimes m_1^\prime \otimes \cdots \otimes m_q^\prime$\retTwo\par}

	Similarly for any $p \in \mathbb{Z}_{\geq 1}$ we know there is a (unique) $R$-module homomorphism\\ $\phi_{0, q} : R \otimes T_p(M) \to T_p(M)$ such that:
	
	{\centering$\phi_{0, q}(c \otimes (m_1 \otimes \cdots \otimes m_p)) = c(m_1 \otimes \cdots \otimes m_p)$.\retTwo\par}

One notable consequence of having a natural way of viewing $M$ as a submodule of $T(M)$ is as follows:
\begin{myIndent}\pracTwo
	Suppose $M$ and $N$ are $R$-modules and $f: M \to N$ is an $R$-algebra homomorphism. By viewing $N$ as an $R$-submodule of $T(N)$, we can then say by the universal property of $T(M)$ that there is a unique unital $R$-algebra homomorphism $T(f): T(M) \to T(N)$ such that $T(f)|_M = f$.\retTwo

	Furthermore, note that $x \in T(M)$ is homogeneous of degree $1$ iff $x \in M$. Yet then $(T(f))(x) = f(x) \in N$. So, $(T(f))(x) \in T(N)$ is homogeneous of degree $1$ as well and so we can conclude $(T(f))(T_1(M)) \subseteq T_1(N)$.\retTwo
	
	By lemma 3.1 on \hspace{0.1em}\inLinkRap{Lemma 3.1 page 1071}{page 1071}\hspace{0.2em} it then follows that $T(f)$ maps a generating set for\\ $T_p(M)$ (as an $R$-submodule) into $T_p(N)$ for all $p \geq 2$. Plus, it's obvious that:
	
	{\centering$(T(f))(T_0(M)) = R(T(f))(1_{T(M)}) = R1_{T(N)} = T_0(N)$.\retTwo\par}
	
	So, $T(f): T(M) \to T(N)$ is a graded $R$-algebra homomorphism.\retTwo
\end{myIndent}

Another way of describing the prior consequence is as follows. Let $\mcateg{GrAlg}_R$ denote the category whose objects are graded unital $R$-algebras and whose morphisms are (unital) graded algebra homomorphisms.\retTwo

We can define a functor $T: R$-$\mcateg{mod} \to \mcateg{GrAlg}_R$ mapping any $R$-module $M$ to its tensor algebra and mapping any $R$-module homomorphism $f: M \to N$ to the graded $R$-algebra homomorphism $T(f) : T(M) \to T(N)$ described earlier.
\begin{myIndent}\pracTwo
	Note that $T(\myId_M) = \myId_{T(M)}$ for all $R$-modules $M$. After all, $\myId_{T(M)}: T(M) \to T(M)$\\ is a graded $R$-algebra homomorphism whose restriction to $M$ is just the inclusion map\\ $M \hookrightarrow T(M)$. Yet by the univeral property of tensor algebras there is only one unital $R$-algebra homomorphism $T(\myId_M)$ with that property.\retTwo

	In a similar vein, if $f: M \to N$ and $g: N \to K$ are $R$-module homomorphisms, then $T(g \circ f) = T(g) \circ T(f)$. After all, the restriction of either map to $M$ is just the map $g \circ f$. So, both maps must be the same unique map satisfying that. And this proves that $T: R$-$\mcateg{mod} \to \mcateg{GrAlg}_R$ is a functor.\retTwo
\end{myIndent}

One other note is that every element of $T_1(M) = M$ is homogeneous in $T(M)$. Therefore, if $K \subseteq M$ is a submodule then we know from lemma 3.2 on \inLinkRap{Lemma 3.2 page 1071}{page 1071} that the two-sided ideal $I(K) \subseteq T(M)$ generated by $K$ is homogeneous.\retTwo

	Proof:\\

So, $T(f)$ is surjective.\retTwo

	Next note that it's obvious that $I(K) \subseteq \ker(T(f))$ since $K \subseteq \ker(T(f))$ and we can\\ easily verify that the kernel of an $R$-algebra homomorphism is an ideal and $R$-module.\retTwo
	
	As for the other inclusion, consider the quotient map $\pi: T(M) \to T(M)/I(K)$. We know from \inLinkRap{Page 1072 quotients of graded algebras}{page 1072} that $T(M)/I(K)$ has a natural grading $\{A_n\}_{n \in \mathbb{N}}$ satisfying that $M/K$ is isomorphic to $A_1$ as an $R$-module via the map $\phi(m + K) \coloneqq m + I(K)$. Also note by the first isomorphism theorem that $f: M \to N$ induces an $R$-module isomorphism $\widetilde{f}: M/K \to N$.\retTwo
	
	So, $g \coloneqq \phi \circ \widetilde{f}^{-1}$ is an $R$-module homorphism from $N$ to $A_1 \subseteq T(M)/I(K)$ satisfying that $g(n) = m + I(K)$ where $m \in M$ is any element satisfying that $f(m) = n$. 	{\centering$n = \myId_N(n) = v(u(n)) = v(m) = 0$. So $m = u(0) = 0$.\retTwo\par}

\ul{Theorem 4.5:} Let $P$ be an $R$-module and let $T(P)$ be the tensor algebra of $P$. Then $P$ is a projective $R$-module if and only if $T(P)$ is a projective $R$-module.
\begin{myIndent}\exThreeP
	$(\Longleftarrow)$\\
	Note that $T(P) \cong P \oplus (\bigoplus_{\substack{p \geq 0 \\ p \neq 1}}T_p(P))$ as $R$-modules.\retTwo
	
	But now note that $T(P)$ is projective if and only if there is a free $R$-module $\mathcal{F}$ and some $R$-module $K$ such that $T(P) \oplus K \cong \mathcal{F}$. In turn:
	
	{\centering$\mathcal{F} \cong (P \oplus (\bigoplus_{\substack{p \geq 0 \\ p \neq 1}}T_p(P))) \oplus K \cong P \oplus ((\bigoplus_{\substack{p \geq 0 \\ p \neq 1}}T_p(P)) \oplus K)$\retTwo\par}

	So, $P$ is also a direct summand of a free $R$-module and is thus projective.
	\begin{myIndent}\myComment

	\end{myIndent}

	$(\Longrightarrow)$\\

Then we let $\iota: P \hookrightarrow \mathcal{F}$ be the direct sum embedding and we let\\ $\pi: \mathcal{F} \rightarrowdbl P$ be the direct sum projection. By the universal property of $T(P)$ and $T(\mathcal{F})$, we know that $\iota$ and $\pi$ uniquely extend to $R$-algebra homomorphisms:
	
	{\centering$T(\iota): T(P) \to T(\mathcal{F})$ and $T(\pi) : T(\mathcal{F}) \to T(P)$.\retTwo\par}

	But now as $\pi \circ \iota = \myId_P$, we can conclude that $T(\pi) \circ T(\iota) = T(\myId_P) = \myId_{T(P)}$. So by the prior lemma, we can conclude that $T(\mathcal{F}) \cong T(P) \oplus \ker(T(\pi))$. Yet also note that since $\mathcal{F}$ is a free $R$-module, we know $T(\mathcal{F})$ is also a free $R$-module. So, $T(P)$ is a direct summand of a free $R$-module and is thus projective. $\blacksquare$\retTwo
\end{myIndent}
